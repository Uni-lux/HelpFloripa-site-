    // ===================================================
    // STORIES (24 horas): stories/{id} — foto (no documento ou no Storage) ou vídeo (Storage).
    // Somem para todos depois de 24 h (o site filtra; o autor apaga os vencidos). vistos/{uid}: quem viu.
    // ===================================================
    // Lembrete da contagem regressiva de um flash (como o "Lembrar-me" do Instagram): a pessoa pede para ser avisada
    // quando a contagem acabar. id = {flash}_{uid}. Guarda título e fim (o flash sai do ar em 24 h, a contagem não).
    // A pessoa cria e apaga o dela; ela e o autor do flash leem (o autor vê quem pediu).
    match /contagem_lembretes/{lid} {
      allow create: if logado() && request.resource.data.uid == request.auth.uid
                    && lid == request.resource.data.flashId + '_' + request.auth.uid
                    && request.resource.data.keys().hasOnly(['flashId', 'autorId', 'uid', 'titulo', 'fim', 'criadoEm'])
                    && request.resource.data.criadoEm == request.time
                    && get(/databases/$(database)/documents/stories/$(request.resource.data.flashId)).data.autorId == request.resource.data.autorId
                    && get(/databases/$(database)/documents/stories/$(request.resource.data.flashId)).data.contagem.fim == request.resource.data.fim
                    && get(/databases/$(database)/documents/stories/$(request.resource.data.flashId)).data.contagem.titulo == request.resource.data.titulo
                    && request.resource.data.autorId != request.auth.uid;
      allow read: if logado() && (resource.data.uid == request.auth.uid || resource.data.autorId == request.auth.uid);
      allow delete: if logado() && resource.data.uid == request.auth.uid;
      allow update: if false;
    }
    match /stories/{id} {
      // Depois das 24 h o flash fica no ARQUIVO do autor (só ele vê, como no Instagram). Os outros leem o que está no ar
      // (26 h, com folga para relógio atrasado) e o que o autor pôs num destaque (destacado: true).
      allow read: if logado() && (resource.data.autorId == request.auth.uid
                    || resource.data.criadoEm > request.time - duration.value(26, 'h')
                    || resource.data.get('destacado', false) == true
                    || podeModerar());
      // Anti-spam: até 50 flashes por dia (ritmo/{uid}, gravado junto na mesma operação).
      allow create: if logado() && emailOk() && flashNovoOk(request.resource.data) && !sancionado(request.auth.uid) && ritmoOk('flash');
      allow delete: if logado() && (resource.data.autorId == request.auth.uid || podeModerar());
      // O autor só marca/desmarca "destacado" (o flash está num destaque do perfil); o resto não muda depois de publicado.
      allow update: if logado() && resource.data.autorId == request.auth.uid
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['destacado'])
                    && request.resource.data.destacado is bool;
      match /vistos/{uid} {
        allow create: if logado() && uid == request.auth.uid
                      && request.resource.data.keys().hasOnly(['criadoEm'])
                      && request.resource.data.criadoEm == request.time;
        allow read: if logado() && (uid == request.auth.uid
                      || get(/databases/$(database)/documents/stories/$(id)).data.autorId == request.auth.uid);
        allow update, delete: if false;
      }
      // reacoes/{uid}: a estrela (curtir o flash) e o último emoji de quem viu, para o autor ver na lista de
      // visualizações (como o coração e as reações do Instagram). Só a própria pessoa grava; só ela e o autor leem.
      match /reacoes/{uid} {
        allow create, update: if logado() && uid == request.auth.uid
                      && request.resource.data.keys().hasOnly(['curtiu', 'emoji', 'em'])
                      && request.resource.data.em == request.time
                      && (!('curtiu' in request.resource.data) || request.resource.data.curtiu is bool)
                      && (!('emoji' in request.resource.data) || request.resource.data.emoji in ['🔥', '😍', '😂', '😮', '👏', '💯', '⭐'])
                      && get(/databases/$(database)/documents/stories/$(id)).data.autorId != request.auth.uid;
        allow read: if logado() && (uid == request.auth.uid
                      || get(/databases/$(database)/documents/stories/$(id)).data.autorId == request.auth.uid);
        allow delete: if logado() && uid == request.auth.uid;
      }
    }

    // ===================================================
    // DESTAQUES do Flash: destaques/{id} {uid, nome, ids, capa, capaMini, criadoEm, atualizadoEm} — ficam no perfil,
    // como os destaques do Instagram. Quem está logado lê (os flashes em si já são lidos assim); só o dono cria,
    // muda e apaga (a moderação também apaga). capaMini = miniatura JPEG da capa (para o perfil não baixar os flashes).
    // ===================================================
    match /destaques/{id} {
      function destaqueOk(d) {
        return d.keys().hasOnly(['uid', 'nome', 'ids', 'capa', 'capaMini', 'criadoEm', 'atualizadoEm'])
          && d.nome is string && d.nome.size() >= 1 && d.nome.size() <= 30
          && d.ids is list && d.ids.size() >= 1 && d.ids.size() <= 100
          && d.atualizadoEm == request.time
          && (!('capa' in d) || (d.capa is string && d.capa.size() <= 64 && d.capa in d.ids))
          && (!('capaMini' in d) || (d.capaMini is string && d.capaMini.size() <= 30000 && d.capaMini.matches('^data:image/jpeg;base64,[A-Za-z0-9+/=]+$')));
      }
      allow read: if logado();
      allow create: if logado() && emailOk() && request.resource.data.uid == request.auth.uid
                    && request.resource.data.criadoEm == request.time && destaqueOk(request.resource.data);
      allow update: if logado() && resource.data.uid == request.auth.uid && request.resource.data.uid == request.auth.uid
                    && request.resource.data.criadoEm == resource.data.criadoEm && destaqueOk(request.resource.data);
      allow delete: if logado() && (resource.data.uid == request.auth.uid || podeModerar());
    }

    // ===================================================
    // RASCUNHOS: rascunhos/{uid}/itens/{id} — só a própria pessoa (texto e fotos ainda não publicados).
    // ===================================================
    // ===================================================
    // VOTOS DAS ENQUETES: enquete_votos/{alvo}_{uid} — alvo = tipo + "_" + itemId (tipo d = publicação, f = flash).
    // fimEm (opcional) = o mesmo fim da enquete; votar só até o fim.
    // Um voto por pessoa, sem trocar. A contagem fica dividida em 10 pedaços (enquete_contas/{alvo}_{fatia},
    // ver abaixo): o voto leva a fatia (0–9) e o pedaço dela sobe +1 na MESMA gravação.
    // ===================================================
    // A contagem da opção do voto sobe exatamente 1 no pedaço escolhido (votos públicos e anônimos).
    function fatiaSobe(d) {
      let pedaco = /databases/$(database)/documents/enquete_contas/$(d.alvo + '_' + string(d.fatia));
      let k = 'n' + string(d.opcao);
      return d.fatia is int && d.fatia >= 0 && d.fatia < 10
          && getAfter(pedaco).data.get(k, 0) == (exists(pedaco) ? get(pedaco).data.get(k, 0) : 0) + 1;
    }
    match /enquete_votos/{id} {
      allow read: if logado();
      function votoOk() {
        let d = request.resource.data;
        let doc = d.tipo == 'd'
          ? /databases/$(database)/documents/diario/$(d.itemId)
          : /databases/$(database)/documents/stories/$(d.itemId);
        let e = get(doc).data.enquete;
        return d.keys().hasOnly(['alvo', 'tipo', 'itemId', 'uid', 'opcao', 'fatia', 'fimEm', 'criadoEm'])
            && d.uid == request.auth.uid
            && d.tipo in ['d', 'f']
            && d.itemId is string && d.itemId.matches('^[A-Za-z0-9]{1,64}$')
            && d.alvo == d.tipo + '_' + d.itemId
            && id == d.alvo + '_' + request.auth.uid
            && d.opcao is int && d.opcao >= 0
            && exists(doc) && d.opcao < e.opcoes.size()
            && e.get('anonima', false) == false
            // depois do fim não dá mais para votar; o voto leva o fim junto (as notificações do resultado usam)
            && (!('fimEm' in e) || request.time < e.fimEm)
            && (!('fimEm' in d) || ('fimEm' in e && d.fimEm == e.fimEm))
            && fatiaSobe(d);
      }
      // Voto único: depois de votar não dá para trocar nem tirar (a limpeza de contas apaga pelo servidor).
      allow create: if logado() && votoOk() && request.resource.data.criadoEm == request.time && !sancionado(request.auth.uid);
      allow update, delete: if false;
    }

    // ===================================================
    // ENQUETE ANÔNIMA: o voto fica em enquete_anon/{alvo}_{uid} (ninguém lista; só a própria pessoa lê o dela)
    // e a contagem nos mesmos pedaços enquete_contas/{alvo}_{fatia}, que sobem +1 na MESMA gravação do voto.
    // Assim nem o autor da enquete sabe quem votou em quê.
    // ===================================================
    match /enquete_anon/{id} {
      function votoAnonOk() {
        let d = request.resource.data;
        let doc = d.tipo == 'd'
          ? /databases/$(database)/documents/diario/$(d.itemId)
          : /databases/$(database)/documents/stories/$(d.itemId);
        let e = get(doc).data.enquete;
        return d.keys().hasOnly(['alvo', 'tipo', 'itemId', 'uid', 'opcao', 'fatia', 'fimEm', 'criadoEm'])
            && d.uid == request.auth.uid && d.criadoEm == request.time
            && d.tipo in ['d', 'f']
            && d.itemId is string && d.itemId.matches('^[A-Za-z0-9]{1,64}$')
            && d.alvo == d.tipo + '_' + d.itemId
            && id == d.alvo + '_' + request.auth.uid
            && d.opcao is int && d.opcao >= 0 && d.opcao < e.opcoes.size()
            && e.get('anonima', false) == true
            && (!('fimEm' in e) || request.time < e.fimEm)
            && (!('fimEm' in d) || ('fimEm' in e && d.fimEm == e.fimEm))
            // a contagem da opção sobe exatamente 1 junto com o voto
            && fatiaSobe(d);
      }
      allow get: if logado() && id.matches('.*_' + request.auth.uid);
      // listar só os próprios votos (o resultado chega nas notificações de quem votou)
      allow list: if logado() && resource.data.uid == request.auth.uid;
      allow create: if logado() && !sancionado(request.auth.uid) && votoAnonOk();
      allow update, delete: if false;
    }
    // CONTAGEM DAS ENQUETES em 10 pedaços: enquete_contas/{alvo}_{fatia} = { n0..n3 } (fatia 0–9). Cada voto sobe
    // 1 pedaço sorteado: um documento só aguentava ~1 voto por segundo; assim a enquete aguenta ~10× mais, e ler o
    // resultado custa no máximo 10 leituras, com quantos votos for. Só sobe junto com o voto de quem grava
    // (público em enquete_votos ou anônimo em enquete_anon), na fatia do voto, +1 na opção dele.
    match /enquete_contas/{cid} {
      allow read: if logado();
      function partes() { return cid.split('_'); }
      function votoEm(colecao) { return /databases/$(database)/documents/$(colecao)/$(partes()[0] + '_' + partes()[1] + '_' + request.auth.uid); }
      function votoDeAgora() {
        return !exists(votoEm('enquete_anon')) && existsAfter(votoEm('enquete_anon')) ? votoEm('enquete_anon') : votoEm('enquete_votos');
      }
      function chaveDoVoto() { return 'n' + string(getAfter(votoDeAgora()).data.opcao); }
      function pedacoDoVoto() {
        return cid.matches('^[df]_[A-Za-z0-9]{1,64}_[0-9]$')
            && !exists(votoDeAgora()) && existsAfter(votoDeAgora())
            && string(getAfter(votoDeAgora()).data.fatia) == partes()[2];
      }
      allow create: if logado() && pedacoDoVoto()
                    && request.resource.data.keys().hasOnly([chaveDoVoto()])
                    && request.resource.data[chaveDoVoto()] == 1;
      allow update: if logado() && pedacoDoVoto()
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly([chaveDoVoto()])
                    && request.resource.data[chaveDoVoto()] == resource.data.get(chaveDoVoto(), 0) + 1;
      allow delete: if false;
    }
