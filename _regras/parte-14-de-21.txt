    // =====================================================
    // PEDIDOS DE EXCLUSÃO DE CONTA: exclusoes/{uid}
    // A pessoa pede a exclusão; durante 30 dias pode recuperar a conta (apaga este
    // documento). Depois disso a limpeza automática (Admin SDK, fora das regras)
    // apaga a conta e os dados e anonimiza avaliações e reclamações.
    // =====================================================
    match /exclusoes/{uid} {
      allow get: if logado() && request.auth.uid == uid;
      allow read: if admin();
      allow create: if logado()
                    && request.auth.uid == uid
                    && request.resource.data.keys().hasOnly(['pedidaEm', 'motivo'])
                    && request.resource.data.pedidaEm == request.time
                    && valorTextoAte(request.resource.data.get('motivo', ''), 300);
      allow update: if false;
      allow delete: if logado() && request.auth.uid == uid;
    }


    // =====================================================
    // ESTRELAS NAS PUBLICAÇÕES: curtidas/{postId}_{uid}
    // Nota de 1 a 5, uma por pessoa em cada publicação. A pessoa pode
    // trocar a nota ou tirar. Ninguém dá estrelas na própria publicação.
    // =====================================================
    match /curtidas/{id} {

      allow read: if logado();

      function notaOk(d) {
        return d.nota is int && d.nota >= 1 && d.nota <= 5;
      }
      // Cada nota muda junto o resumo da publicação (diario_notas/{postId}); as regras do resumo conferem a conta.
      function resumoJunto(postId) {
        let r = /databases/$(database)/documents/diario_notas/$(postId);
        return existsAfter(r)
            && getAfter(r).data.get('soma', 0) != (exists(r) ? get(r).data.get('soma', 0) : 0);
      }

      allow create: if logado()
                    && emailOk()
                    && request.resource.data.keys().hasOnly(['postId', 'uid', 'postAutorId', 'nota', 'criadoEm', 'origem'])
                    && (!('origem' in request.resource.data) || request.resource.data.origem in ['principais', 'seguindo', 'amigos', 'perfil', 'link', 'tag', 'salvos'])
                    && request.resource.data.uid == request.auth.uid
                    && id == request.resource.data.postId + '_' + request.auth.uid
                    && exists(/databases/$(database)/documents/diario/$(request.resource.data.postId))
                    && get(/databases/$(database)/documents/diario/$(request.resource.data.postId)).data.autorId == request.resource.data.postAutorId
                    // pode dar estrelas na própria publicação ("se curtir")
                    && notaOk(request.resource.data)
                    && !bloqueioEntre(request.auth.uid, request.resource.data.postAutorId)
                    && request.resource.data.criadoEm == request.time
                    && resumoJunto(request.resource.data.postId);

      allow update: if logado()
                    && emailOk()
                    && resource.data.uid == request.auth.uid
                    && mudou().hasOnly(['nota', 'atualizadoEm'])
                    && notaOk(request.resource.data)
                    && request.resource.data.atualizadoEm == request.time
                    && resumoJunto(resource.data.postId);

      // Tirar a nota: tira junto do resumo. Publicação que já não existe: apaga sem resumo (limpeza de conta).
      allow delete: if logado() && resource.data.uid == request.auth.uid
                    && (resumoJunto(resource.data.postId)
                        || !exists(/databases/$(database)/documents/diario/$(resource.data.postId)));
    }

    // ===================================================
    // SALVAMENTOS: salvamentos/{postId}_{uid} — só para o autor contar quantos salvaram (não vê quem).
    // A lista de salvos de cada um continua em usuarios/{uid}.salvos (privada).
    // ===================================================
    match /salvamentos/{id} {
      allow create: if logado()
                    && request.resource.data.keys().hasOnly(['postId', 'uid', 'postAutorId', 'criadoEm'])
                    && request.resource.data.uid == request.auth.uid
                    && id == request.resource.data.postId + '_' + request.auth.uid
                    && get(/databases/$(database)/documents/diario/$(request.resource.data.postId)).data.autorId == request.resource.data.postAutorId
                    && request.resource.data.criadoEm == request.time;
      allow read: if logado() && (resource.data.uid == request.auth.uid || resource.data.postAutorId == request.auth.uid);
      allow delete: if logado() && resource.data.uid == request.auth.uid;
      allow update: if false;
    }

    // Republicar (como o repost do Instagram): republicacoes/{postId}_{uid}. A publicação original aparece para quem
    // segue quem republicou, com "Fulano republicou". Não dá para republicar a própria nem com bloqueio entre os dois.
    match /republicacoes/{id} {
      allow read: if logado();
      allow create: if logado() && emailOk()
                    && request.resource.data.keys().hasOnly(['postId', 'uid', 'postAutorId', 'nome', 'criadoEm'])
                    && request.resource.data.uid == request.auth.uid
                    && request.resource.data.postId is string && request.resource.data.postId.size() <= 64
                    && id == request.resource.data.postId + '_' + request.auth.uid
                    && request.resource.data.postAutorId != request.auth.uid
                    && (!('nome' in request.resource.data) || (request.resource.data.nome is string && request.resource.data.nome.size() <= 80))
                    && get(/databases/$(database)/documents/diario/$(request.resource.data.postId)).data.autorId == request.resource.data.postAutorId
                    && !bloqueioEntre(request.auth.uid, request.resource.data.postAutorId)
                    && request.resource.data.criadoEm == request.time;
      allow delete: if logado() && (resource.data.uid == request.auth.uid || podeModerar());
      allow update: if false;
    }

    // ===================================================
    // STORIES (24 horas): stories/{id} — foto (no documento ou no Storage) ou vídeo (Storage).
    // Somem para todos depois de 24 h (o site filtra; o autor apaga os vencidos). vistos/{uid}: quem viu.
    // ===================================================
    // Lembrete da contagem regressiva de um flash (como o "Lembrar-me" do Instagram): a pessoa pede para ser avisada
    // quando a contagem acabar. id = {flash}_{uid}. Guarda título e fim (o flash sai do ar em 24 h, a contagem não).
    // A pessoa cria e apaga o dela; ela e o autor do flash leem (o autor vê quem pediu).
    match /contagem_lembretes/{lid} {
      allow create: if logado() && request.resource.data.uid == request.auth.uid
                    && lid == request.resource.data.flashId + '_' + request.auth.uid
                    && request.resource.data.keys().hasOnly(['flashId', 'autorId', 'uid', 'titulo', 'fim', 'criadoEm'])
                    && request.resource.data.criadoEm == request.time
                    && get(/databases/$(database)/documents/stories/$(request.resource.data.flashId)).data.autorId == request.resource.data.autorId
                    && get(/databases/$(database)/documents/stories/$(request.resource.data.flashId)).data.contagem.fim == request.resource.data.fim
                    && get(/databases/$(database)/documents/stories/$(request.resource.data.flashId)).data.contagem.titulo == request.resource.data.titulo
                    && request.resource.data.autorId != request.auth.uid;
      allow read: if logado() && (resource.data.uid == request.auth.uid || resource.data.autorId == request.auth.uid);
      allow delete: if logado() && resource.data.uid == request.auth.uid;
      allow update: if false;
    }
    match /stories/{id} {
      // Depois das 24 h o flash fica no ARQUIVO do autor (só ele vê, como no Instagram). Os outros leem o que está no ar
      // (26 h, com folga para relógio atrasado) e o que o autor pôs num destaque (destacado: true).
      // Quem pode ver (13/11/2026; antes qualquer pessoa logada lia, e só o app escondia): para todos (restrito false);
      // Amigos próximos só quem está na lista; com gente ocultada, todos menos ela (esses a fileira acha pelo
      // stories_restritos e abre um por um). Flash antigo, sem o campo, vale pelo publico/ocultoPara dele.
      // ATENÇÃO: numa consulta (list) o campo que ela não filtra vale o padrão do .get(campo, padrão) — por isso a consulta
      // só passa com "restrito == false" ou "lista array-contains eu"; a conferência de "ocultoPara" é só ao abrir (get).
      allow get: if logado() && (resource.data.autorId == request.auth.uid || podeModerar()
                    || ((resource.data.criadoEm > request.time - duration.value(26, 'h')
                         || resource.data.get('destacado', false) == true)
                        && (resource.data.get('publico', 'todos') == 'amigos'
                            ? request.auth.uid in resource.data.lista
                            : !(request.auth.uid in resource.data.get('ocultoPara', [])))));
      allow list: if logado() && (resource.data.autorId == request.auth.uid || podeModerar()
                    || (resource.data.criadoEm > request.time - duration.value(26, 'h')
                        && (resource.data.restrito == false || request.auth.uid in resource.data.lista)));
      // Anti-spam: até 50 flashes por dia (ritmo/{uid}, gravado junto na mesma operação).
      allow create: if logado() && emailOk() && flashNovoOk(request.resource.data) && !sancionado(request.auth.uid) && ritmoOk('flash');
      allow delete: if logado() && (resource.data.autorId == request.auth.uid || podeModerar());
      // O autor só marca/desmarca "destacado" (o flash está num destaque do perfil); o resto não muda depois de publicado.
      allow update: if logado() && resource.data.autorId == request.auth.uid
                    && ((request.resource.data.diff(resource.data).affectedKeys().hasOnly(['destacado'])
                         && request.resource.data.destacado is bool)
                        // Flash de antes do campo "restrito": o autor completa (só com o valor certo) e ele volta à fileira.
                        || (!('restrito' in resource.data)
                            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['restrito'])
                            && request.resource.data.restrito == (resource.data.get('publico', 'todos') == 'amigos'
                                                                  || resource.data.get('ocultoPara', []).size() > 0)));
      match /vistos/{uid} {
        allow create: if logado() && uid == request.auth.uid
                      && request.resource.data.keys().hasOnly(['criadoEm'])
                      && request.resource.data.criadoEm == request.time;
        allow read: if logado() && (uid == request.auth.uid
                      || get(/databases/$(database)/documents/stories/$(id)).data.autorId == request.auth.uid);
        allow update, delete: if false;
      }
      // reacoes/{uid}: a estrela (curtir o flash) e o último emoji de quem viu, para o autor ver na lista de
      // visualizações (como o coração e as reações do Instagram). Só a própria pessoa grava; só ela e o autor leem.
      match /reacoes/{uid} {
        allow create, update: if logado() && uid == request.auth.uid
                      && request.resource.data.keys().hasOnly(['curtiu', 'emoji', 'em'])
                      && request.resource.data.em == request.time
                      && (!('curtiu' in request.resource.data) || request.resource.data.curtiu is bool)
                      && (!('emoji' in request.resource.data) || request.resource.data.emoji in ['🔥', '😍', '😂', '😮', '👏', '💯', '⭐'])
                      && get(/databases/$(database)/documents/stories/$(id)).data.autorId != request.auth.uid;
        allow read: if logado() && (uid == request.auth.uid
                      || get(/databases/$(database)/documents/stories/$(id)).data.autorId == request.auth.uid);
        allow delete: if logado() && uid == request.auth.uid;
      }
    }
