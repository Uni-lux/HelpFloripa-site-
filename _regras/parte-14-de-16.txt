      allow create: if logado() && pedacoDoVoto()
                    && request.resource.data.keys().hasOnly([chaveDoVoto()])
                    && request.resource.data[chaveDoVoto()] == 1;
      allow update: if logado() && pedacoDoVoto()
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly([chaveDoVoto()])
                    && request.resource.data[chaveDoVoto()] == resource.data.get(chaveDoVoto(), 0) + 1;
      allow delete: if false;
    }

    //   uid, nome, foto (só se NÃO for anônima), texto (até 300), anonima, criadoEm }.
    // Anônima: sem uid/nome/foto; quem respondeu fica em caixa_autoria/{id} (só a moderação lê).
    // O dono lê e apaga as respostas; quem respondeu (não anônima) lê e apaga a sua.
    // ===================================================
    match /caixa_respostas/{id} {
      function caixaDoItem(d) {
        return get(d.tipo == 'd'
          ? /databases/$(database)/documents/diario/$(d.itemId)
          : /databases/$(database)/documents/stories/$(d.itemId)).data;
      }
      function respostaOk() {
        let d = request.resource.data;
        let item = caixaDoItem(d);
        let autoria = /databases/$(database)/documents/caixa_autoria/$(id);
        return d.keys().hasOnly(['alvo', 'tipo', 'itemId', 'autorId', 'uid', 'nome', 'foto', 'texto', 'anonima', 'criadoEm'])
            && d.tipo in ['d', 'f']
            && d.itemId is string && d.itemId.matches('^[A-Za-z0-9]{1,64}$')
            && d.alvo == d.tipo + '_' + d.itemId
            && d.texto is string && d.texto.size() >= 1 && d.texto.size() <= 300
            && d.criadoEm == request.time
            && 'caixa' in item && d.autorId == item.autorId
            && d.anonima == item.caixa.anonima
            && (d.anonima
                ? (!d.keys().hasAny(['uid', 'nome', 'foto']) && existsAfter(autoria) && getAfter(autoria).data.uid == request.auth.uid)
                : (d.uid == request.auth.uid && textoAte('nome', 80) && textoAte('foto', 2000)));
      }
      allow create: if logado() && emailOk() && !sancionado(request.auth.uid) && respostaOk();
      allow read, delete: if logado() && (resource.data.autorId == request.auth.uid
                                       || resource.data.get('uid', '') == request.auth.uid
                                       || podeModerar());
      allow update: if false;
    }
    match /caixa_autoria/{id} {
      allow read, delete: if podeModerar();
      allow create: if logado()
                    && request.resource.data.keys().hasOnly(['uid', 'criadoEm'])
                    && request.resource.data.uid == request.auth.uid
                    && request.resource.data.criadoEm == request.time
                    && existsAfter(/databases/$(database)/documents/caixa_respostas/$(id));
      allow update: if false;
    }

    // (quem pode marcar/mencionar a pessoa). Só o dono grava; qualquer pessoa logada lê.
    // ===================================================
    match /pref_publicas/{uid} {
      allow read: if logado();
      allow create, update: if logado() && uid == request.auth.uid
                    && request.resource.data.keys().hasOnly(['permiteMarcar', 'permiteFlash', 'atualizadoEm'])
                    && (!('permiteMarcar' in request.resource.data) || request.resource.data.permiteMarcar in ['todos', 'seguindo', 'ninguem'])
                    && (!('permiteFlash' in request.resource.data) || request.resource.data.permiteFlash is bool);
      allow delete: if logado() && uid == request.auth.uid;
    }

      allow read, delete: if logado() && uid == request.auth.uid;
      allow create, update: if logado() && uid == request.auth.uid
                    && request.resource.data.keys().hasOnly(['texto', 'fotos', 'atualizadoEm'])
                    && valorTextoAte(request.resource.data.get('texto', ''), 1000)
                    && (!('fotos' in request.resource.data) || (request.resource.data.fotos is list && request.resource.data.fotos.size() <= 4));
    }

    // ===================================================
    // RESUMO DAS ESTRELAS DE UMA PUBLICAÇÃO: diario_notas/{postId} { total, soma, n1..n5 }
    // Uma leitura mostra a média e quantas pessoas deram 5, 4, 3, 2 e 1 estrela, com mil ou um milhão
    // de notas. Só muda junto com a nota de quem grava (curtidas/{postId}_{uid}), na conta exata:
    // a nota de antes sai e a nova entra. A moderação pode recalcular.
    // ===================================================
    match /diario_notas/{postId} {
      allow read: if logado();

      function minhaNota() { return /databases/$(database)/documents/curtidas/$(postId + '_' + request.auth.uid); }
      function contaCerta() {
        let a = exists(minhaNota()) ? get(minhaNota()).data.nota : 0;
        let d = existsAfter(minhaNota()) ? getAfter(minhaNota()).data.nota : 0;
        let r = resource == null ? {} : resource.data;
        let n = request.resource.data;
        return a != d
            && n.keys().hasOnly(['total', 'soma', 'n1', 'n2', 'n3', 'n4', 'n5'])
            && n.get('total', 0) == r.get('total', 0) + (d > 0 ? 1 : 0) - (a > 0 ? 1 : 0)
            && n.get('soma', 0) == r.get('soma', 0) + d - a
            && n.get('n1', 0) == r.get('n1', 0) + (d == 1 ? 1 : 0) - (a == 1 ? 1 : 0)
            && n.get('n2', 0) == r.get('n2', 0) + (d == 2 ? 1 : 0) - (a == 2 ? 1 : 0)
            && n.get('n3', 0) == r.get('n3', 0) + (d == 3 ? 1 : 0) - (a == 3 ? 1 : 0)
            && n.get('n4', 0) == r.get('n4', 0) + (d == 4 ? 1 : 0) - (a == 4 ? 1 : 0)
            && n.get('n5', 0) == r.get('n5', 0) + (d == 5 ? 1 : 0) - (a == 5 ? 1 : 0);
      }
      allow create, update: if logado() && contaCerta();
      // Recalcular (moderação): números inteiros, sem campo estranho.
      allow create, update: if podeModerar()
                    && request.resource.data.keys().hasOnly(['total', 'soma', 'n1', 'n2', 'n3', 'n4', 'n5'])
                    && request.resource.data.total is int && request.resource.data.soma is int
                    && request.resource.data.n1 is int && request.resource.data.n2 is int && request.resource.data.n3 is int
                    && request.resource.data.n4 is int && request.resource.data.n5 is int;
      allow delete: if podeModerar();
    }

    // =====================================================
    // COMENTÁRIOS: comentarios/{id}
    // - Resposta: respostaA = id do comentário respondido (mesma publicação)
    //   e respostaAutorId = autor dele (para o aviso).
    // - Quem comentou edita o texto. O dono da publicação oculta um comentário.
    // - Quem comentou ou o dono da publicação apaga.
    // - Publicação com comentários ocultos não recebe comentários novos.
    // =====================================================
    match /comentarios/{id} {

      allow read: if logado();

      function textoComentarioOk(d) {
        return d.texto is string && d.texto.size() >= 1 && d.texto.size() <= 500;
      }

      // Quem pode comentar (escolha do autor): o autor sempre pode.
      function segue(a, b) { return exists(/databases/$(database)/documents/relacoes/$(a + '_' + b)); }
      function podeComentar(p) {
        let q = p.get('quemComenta', 'todos');
        return p.autorId == request.auth.uid
            || q == 'todos'
            || (q == 'seguidores' && segue(request.auth.uid, p.autorId))
            || (q == 'mutuos' && segue(request.auth.uid, p.autorId) && segue(p.autorId, request.auth.uid));
      }

      function respostaOk(d) {
        return (!('respostaA' in d) && !('respostaAutorId' in d))
            || (d.respostaA is string
                && exists(/databases/$(database)/documents/comentarios/$(d.respostaA))
                && get(/databases/$(database)/documents/comentarios/$(d.respostaA)).data.postId == d.postId
                && d.respostaAutorId == get(/databases/$(database)/documents/comentarios/$(d.respostaA)).data.autorId);
      }

      allow create: if logado()
                    && emailOk()
                    && request.resource.data.keys().hasOnly(['postId', 'postAutorId', 'autorId', 'nome', 'foto', 'texto', 'respostaA', 'respostaAutorId', 'criadoEm'])
                    && request.resource.data.autorId == request.auth.uid
                    && textoComentarioOk(request.resource.data)
                    && valorTextoAte(request.resource.data.get('nome', ''), 80)
                    && fotoOk(request.resource.data.get('foto', ''))
                    && exists(/databases/$(database)/documents/diario/$(request.resource.data.postId))
                    && get(/databases/$(database)/documents/diario/$(request.resource.data.postId)).data.autorId == request.resource.data.postAutorId
                    && get(/databases/$(database)/documents/diario/$(request.resource.data.postId)).data.get('comentariosOcultos', false) != true
                    && podeComentar(get(/databases/$(database)/documents/diario/$(request.resource.data.postId)).data)
                    && respostaOk(request.resource.data)
                    && !bloqueioEntre(request.auth.uid, request.resource.data.postAutorId)
                    && ritmoOk('com')
                    && request.resource.data.criadoEm == request.time;

      allow update: if logado()
                    && emailOk()
                    && (
                      // quem comentou corrige o texto
                      (resource.data.autorId == request.auth.uid
                        && mudou().hasOnly(['texto', 'editadoEm'])
                        && textoComentarioOk(request.resource.data)
                        && request.resource.data.editadoEm == request.time)
                      ||
                      // o dono da publicação oculta ou mostra o comentário
                      (resource.data.postAutorId == request.auth.uid
                        && mudou().hasOnly(['oculto'])
                        && request.resource.data.oculto is bool)
                    );

      allow delete: if logado()
                    && (resource.data.autorId == request.auth.uid || resource.data.postAutorId == request.auth.uid);
      allow delete: if podeModerar();
    }

    // =====================================================
    // CURTIDAS EM COMENTÁRIOS: curtidas_comentarios/{comentarioId}_{uid}
    // =====================================================
    match /curtidas_comentarios/{id} {

      allow read: if logado();

      // Reação (emoji); sem o campo = curtida antiga (conta como ❤️).
      function reacaoOk(d) { return !('reacao' in d) || d.reacao in ['❤️', '👍', '😂', '😮', '😢', '👏']; }

      allow create: if logado()
                    && emailOk()
                    && request.resource.data.keys().hasOnly(['comentarioId', 'uid', 'criadoEm', 'reacao', 'comentarioAutorId'])
                    // quem escreveu o comentário (para avisar da reação)
                    && (!('comentarioAutorId' in request.resource.data)
                        || request.resource.data.comentarioAutorId == get(/databases/$(database)/documents/comentarios/$(request.resource.data.comentarioId)).data.autorId)
                    && request.resource.data.uid == request.auth.uid
                    && id == request.resource.data.comentarioId + '_' + request.auth.uid
                    && exists(/databases/$(database)/documents/comentarios/$(request.resource.data.comentarioId))
                    && reacaoOk(request.resource.data)
                    && request.resource.data.criadoEm == request.time;

      // Trocar a reação (só o emoji).
      allow update: if logado()
                    && resource.data.uid == request.auth.uid
                    && mudou().hasOnly(['reacao'])
                    && reacaoOk(request.resource.data);
      allow delete: if logado() && resource.data.uid == request.auth.uid;
    }
