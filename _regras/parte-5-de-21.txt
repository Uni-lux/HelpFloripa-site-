      function comunidadeOk() {
        let d = request.resource.data;
        let texto = d.get('titulo', '') + ' ' + d.get('descricao', '');
        return d.categoria in ['doacao', 'troca', 'pedido', 'quarto', 'achados']
            && d.titulo is string && d.titulo.size() >= 3 && d.titulo.size() <= 80
            && textoAte('descricao', 1000)
            && textoAte('tipoItem', 20)
            && textoAte('cidade', 60)
            && localOk()
            && textoAte('bairro', 60)
            && geoOk(d.get('geo', null))
            && listaAte('fotos', 4)
            && listaAte('busca', 100)
            && d.get('estado', 'usado') in ['novo', 'seminovo', 'usado', 'reparo']
            && d.get('entrega', 'combinar') in ['combinar', 'retirar', 'entrego']
            && d.status in ['disponivel', 'reservado', 'concluido']
            && d.aceiteRegras == true
            && (d.categoria == 'quarto' ? quartoOk(d.get('quarto', null)) : !('quarto' in d))
            && (d.categoria == 'achados' ? achadoOk(d.get('achado', null)) : !('achado' in d))
            // Sem venda: nada de "vendo/venda/vende..." e, fora do quarto, nada de "R$".
            && !texto.matches('(?i)(?s).*\\b(vendo|venda|vende|vender|vendem|vendas)\\b.*')
            && (d.categoria == 'quarto' || !texto.matches('(?s).*[Rr][$].*'))
            && dataOuNulo('ocultoAte');
      }
      allow read: if logado();
      allow create: if logado() && emailOk()
                    && request.resource.data.donoId == request.auth.uid
                    && request.resource.data.keys().hasOnly(camposComunidade())
                    && !('ocultoPelaEquipe' in request.resource.data)
                    && !banidoDoPrograma(request.auth.uid, 'comunidade')
                    && !sancionado(request.auth.uid)
                    && request.resource.data.criadoEm == request.time
                    && comunidadeOk();
      allow update: if logado() && emailOk()
                    && resource.data.donoId == request.auth.uid
                    && request.resource.data.donoId == resource.data.donoId
                    && request.resource.data.criadoEm == resource.data.criadoEm
                    && gravou().hasOnly(camposComunidade())
                    && travaModeracao()
                    && !banidoDoPrograma(request.auth.uid, 'comunidade')
                    && comunidadeOk();
      allow delete: if logado() && (resource.data.donoId == request.auth.uid || podeModerar());
      // A equipe esconde (anúncio com venda, golpe, conta suspensa).
      allow update: if podeModerar() && moderacaoOk();
    }

    // Banimento de um programa (um tipo de perfil de negócio ou a Comunidade). Só a moderação cria/apaga;
    // a pessoa lê o próprio (para ver o motivo e até quando).
    match /programa_banidos/{id} {
      allow get: if logado() && (id.split('_')[0] == request.auth.uid || admin());
      allow list: if admin();
      allow create, update: if podeModerar()
                    && request.resource.data.keys().hasOnly(['uid', 'tipo', 'motivo', 'ate', 'por', 'em'])
                    && request.resource.data.tipo in ['servicos', 'delivery', 'lojinha', 'imoveis', 'saude', 'automotivo', 'comunidade', 'entregas']
                    && id == request.resource.data.uid + '_' + request.resource.data.tipo
                    && request.resource.data.motivo is string && request.resource.data.motivo.size() >= 3 && request.resource.data.motivo.size() <= 300
                    && (request.resource.data.get('ate', null) == null || request.resource.data.ate is timestamp)
                    && request.resource.data.por == request.auth.uid
                    && request.resource.data.em == request.time;
      allow delete: if podeModerar();
    }

    // =====================================================
    // CONVERSAS
    // Documento: conversas/{uidMenor}_{uidMaior}
    // =====================================================
    // =====================================================
    // NOTAS DO NEGÓCIO SOBRE O CLIENTE: notas_clientes/{donoId}_{clienteId}
    // Etiquetas e anotação particular. Só quem escreveu lê; o cliente não vê.
    // =====================================================
    match /notas_clientes/{id} {
      allow get: if logado() && id.split('_')[0] == request.auth.uid;
      allow list: if logado() && resource.data.donoId == request.auth.uid;
      allow create, update: if logado()
                    && request.resource.data.keys().hasOnly(['donoId', 'clienteId', 'etiquetas', 'nota', 'atualizadoEm'])
                    && request.resource.data.donoId == request.auth.uid
                    && request.resource.data.clienteId is string
                    && id == request.auth.uid + '_' + request.resource.data.clienteId
                    && request.resource.data.etiquetas is list && request.resource.data.etiquetas.size() <= 8
                    && valorTextoAte(request.resource.data.get('nota', ''), 1000)
                    && request.resource.data.atualizadoEm == request.time;
      allow delete: if logado() && resource.data.donoId == request.auth.uid;
    }

    // Ligações de voz e vídeo (chamadas.js): só entre as duas pessoas de uma conversa que já existe (do site ou do chat
    // da equipe) e sem bloqueio.
    // A chamada guarda a "campainha" (oferta/resposta do WebRTC) e o resultado; áudio e vídeo nunca passam por aqui.
    match /chamadas/{id} {
      function daChamada() { return logado() && (request.auth.uid == resource.data.de || request.auth.uid == resource.data.para); }
      function sdpOk(x) { return x is map && x.keys().hasOnly(['type', 'sdp']) && x.type is string && x.sdp is string && x.sdp.size() <= 30000; }
      allow get, list: if daChamada();
      allow create: if logado()
                    && request.resource.data.keys().hasOnly(['de', 'para', 'conv', 'tipo', 'status', 'oferta', 'criadoEm'])
                    && request.resource.data.de == request.auth.uid
                    && request.resource.data.para is string && request.resource.data.para != request.auth.uid
                    && request.resource.data.conv == (request.auth.uid < request.resource.data.para
                         ? request.auth.uid + '_' + request.resource.data.para : request.resource.data.para + '_' + request.auth.uid)
                    // conversa do site ou conversa individual do chat da equipe (equipe_dm)
                    && (exists(/databases/$(database)/documents/conversas/$(request.resource.data.conv))
                        || exists(/databases/$(database)/documents/equipe_dm/$(request.resource.data.conv)))
                    && !bloqueioEntre(request.auth.uid, request.resource.data.para)
                    // solicitação de mensagem ainda não aceita: ninguém liga (12/11/2026)
                    && (!exists(/databases/$(database)/documents/conversas/$(request.resource.data.conv))
                        || get(/databases/$(database)/documents/conversas/$(request.resource.data.conv)).data.get('pedido', null) == null
                        || get(/databases/$(database)/documents/conversas/$(request.resource.data.conv)).data.pedido.status == 'aceito')
                    && request.resource.data.tipo in ['audio', 'video']
                    && request.resource.data.status == 'chamando'
                    && sdpOk(request.resource.data.oferta)
                    && request.resource.data.criadoEm == request.time;
      // Quem recebe atende (com a resposta) enquanto toca.
      allow update: if daChamada() && request.auth.uid == resource.data.para
                    && resource.data.status == 'chamando'
                    && mudou().hasOnly(['status', 'resposta', 'aceitaEm'])
                    && request.resource.data.status == 'aceita'
                    && sdpOk(request.resource.data.resposta)
                    && request.resource.data.aceitaEm == request.time;
      // Qualquer um dos dois termina: recusar, cancelar, perder (não atendeu) ou encerrar.
      allow update: if daChamada()
                    && resource.data.status in ['chamando', 'aceita']
                    && mudou().hasOnly(['status', 'fimEm', 'encerradaPor'])
                    && request.resource.data.status in ['recusada', 'cancelada', 'perdida', 'encerrada']
                    && request.resource.data.encerradaPor == request.auth.uid
                    && request.resource.data.fimEm == request.time;
      allow delete: if false;
      match /candidatos/{c} {
        function chamada() { return get(/databases/$(database)/documents/chamadas/$(id)).data; }
        allow read: if logado() && request.auth.uid in [chamada().de, chamada().para];
        allow create: if logado() && request.auth.uid in [chamada().de, chamada().para]
                      && chamada().status in ['chamando', 'aceita']
                      && request.resource.data.keys().hasOnly(['de', 'c', 'em'])
                      && request.resource.data.de == request.auth.uid
                      && request.resource.data.c is map && request.resource.data.c.keys().hasOnly(['candidate', 'sdpMid', 'sdpMLineIndex'])
                      && request.resource.data.c.candidate is string && request.resource.data.c.candidate.size() <= 1000
                      && request.resource.data.em == request.time;
        allow update, delete: if false;
      }
    }

    match /conversas/{conversaId} {

      // Cada pessoa só mexe na própria marca de leitura / "digitando".
      function soMinhaChave(campo) {
        return !mudou().hasAny([campo])
            || (request.resource.data[campo] is map
                && request.resource.data[campo].diff(resource.data.get(campo, {})).affectedKeys().hasOnly([request.auth.uid]));
      }

      function contextoOk() {
        let c = request.resource.data.get('contexto', null);
        return c == null
            || (c is map && c.keys().hasOnly(['tipo', 'titulo'])
                && c.tipo in ['servicos', 'delivery', 'lojinha', 'imoveis', 'saude', 'automotivo', 'comunidade']
                && valorTextoAte(c.get('titulo', ''), 120));
      }

      // Quem pode mandar mensagem (12/11/2026, conversa-acesso.js). A conversa NOVA vai direto quando quem recebe aceita
      // todos (pref_publicas.msgPessoal/msgNegocios), segue quem escreve ou tem vínculo aceito com ele; senão ela nasce
      // como SOLICITAÇÃO (pedido {de, status: pendente, em}) — e com "fechado" nem a solicitação. Negócio (contexto do
      // tipo de um perfil que quem recebe tem) usa a escolha dos negócios (padrão todos); pessoal, padrão "seguindo".
      // Conversa de serviço (servico {tipo, id}: corrida, SOS, pet perdido, animal achado) vai direto quando o documento
      // do serviço mostra que os dois fazem parte dele.
      function servicoOk(sv, p) {
        return sv is map && sv.keys().hasOnly(['tipo', 'id']) && sv.id is string && sv.id.size() <= 64
            && (sv.tipo == 'corrida'
                  ? [get(/databases/$(database)/documents/corridas/$(sv.id)).data.clienteId,
                     get(/databases/$(database)/documents/corridas/$(sv.id)).data.get('entregadorId', ''),
                     get(/databases/$(database)/documents/corridas/$(sv.id)).data.get('destinatarioId', ''),
                     get(/databases/$(database)/documents/corridas/$(sv.id)).data.get('pagadorId', '')].hasAll(p)
              : sv.tipo == 'sos'
                  ? get(/databases/$(database)/documents/sos/$(sv.id)).data.uid in p
                    && get(/databases/$(database)/documents/sos/$(sv.id)).data.status in ['aberto', 'atendido']
              : sv.tipo == 'pet'
                  ? get(/databases/$(database)/documents/pets/$(sv.id)).data.donoId in p
                    && get(/databases/$(database)/documents/pets/$(sv.id)).data.status == 'perdido'
              : sv.tipo == 'achado'
                  && get(/databases/$(database)/documents/pets_achados/$(sv.id)).data.autorId in p
                  && get(/databases/$(database)/documents/pets_achados/$(sv.id)).data.status == 'ativo');
      }
