function hashDaPlaca(p) {
let q = p[4:5].matches('[0-9]') ? p[0:4] + 'ABCDEFGHIJ'[int(p[4:5]):int(p[4:5]) + 1] + p[5:7] : p;
return hashing.sha256('hf-veiculo:' + q).toHexString().lower();
}
