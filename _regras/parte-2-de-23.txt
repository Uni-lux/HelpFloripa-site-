    // Flash novo (stories/{id}). Numa função com "d" (em vez de repetir request.resource.data) para gastar menos do
    // limite de 1.000 expressões: o flash com TUDO junto (repost, link, negócios, enquete, caixa, menções, figurinhas
    // tocáveis) estava no limite.
    function flashNovoOk(d) {
      let k = d.keys();
      return k.hasOnly(['autorId', 'nome', 'tipo', 'midia', 'texto', 'duracao', 'enquete', 'caixa', 'link', 'repostDe', 'postDe', 'negocios', 'perfil', 'mencoes', 'respostas', 'publico', 'lista', 'ocultoPara', 'criadoEm', 'toques', 'mencoesOcultas', 'contagem', 'alvo', 'restrito'])
          // caixa de perguntas, link para negócio, repost, perfis de negócio, figurinhas tocáveis e menção oculta
          // (+ contagem regressiva e público do flash, 02/11/2026)
          && (!k.hasAny(['caixa', 'link', 'repostDe', 'postDe', 'negocios', 'perfil', 'toques', 'mencoesOcultas', 'contagem', 'alvo']) || extrasFlashOk(d))
          // enquete (no flash acaba junto com ele: até 24 h, com 10 min de folga), menções e escolhas do autor
          && (!('enquete' in k) || enqueteOk(d.enquete, request.time, 1450))
          && (!('mencoes' in k) || uidsOk(d.mencoes, 10))
          && d.get('respostas', 'todos') in ['todos', 'seguindo', 'ninguem']
          && d.get('publico', 'todos') in ['todos', 'amigos']
          && (!('lista' in k) || uidsOk(d.lista, 150))
          && (!('ocultoPara' in k) || uidsOk(d.ocultoPara, 100))
          // Privacidade de verdade (13/11/2026): "restrito" = Amigos próximos ou com gente ocultada. Amigos próximos leva
          // só a lista (quem foi ocultado já sai dela); para todos não leva lista. A leitura usa isso (match /stories).
          && d.restrito == (d.get('publico', 'todos') == 'amigos' || d.get('ocultoPara', []).size() > 0)
          && (d.get('publico', 'todos') == 'amigos' ? ('lista' in k && !('ocultoPara' in k)) : !('lista' in k))
          && d.autorId == request.auth.uid
          && d.criadoEm == request.time
          && d.tipo in ['imagem', 'video']
          && d.midia is string && d.midia.size() <= 950000
          && d.midia.matches(d.tipo == 'imagem' ? '^(data:image/(jpeg|png|webp);base64,|https://firebasestorage[.]googleapis[.]com/).*' : '^https://firebasestorage[.]googleapis[.]com/.*')
          && valorTextoAte(d.get('texto', ''), 150)
          && valorTextoAte(d.get('nome', ''), 80)
          && (!('duracao' in k) || (d.duracao is number && d.duracao <= 60));
    }
    // Contagem regressiva (título e quando termina: de 5 min a 1 ano) e público do flash (direcionamento, conferido de
    // leve: quem vê filtra no aparelho; a forma completa é a mesma do público da publicação).
    function flashNovos2Ok(d) {
      // (cor e posição não são conferidas: o site só usa as cores da lista e prende a posição no cartão ao desenhar)
      return (!('contagem' in d) || (d.contagem.keys().hasOnly(['titulo', 'fim', 'cor', 'x', 'y', 'k'])
                && d.contagem.titulo.size() >= 1 && d.contagem.titulo.size() <= 40
                && d.contagem.fim > request.time + duration.value(4, 'm') && d.contagem.fim <= request.time + duration.value(366, 'd')))
          && (!('alvo' in d) || d.alvo.keys().hasOnly(['idadeMin', 'idadeMax', 'genero', 'local', 'ufs', 'cidades', 'perto', 'seguidores']));
    }
    // Lista de uids (até max), conferida com uma expressão só.
    function uidsOk(l, max) {
      return l is list && l.size() <= max
          && (l.size() == 0 || l.join(' ').matches('^[A-Za-z0-9]{1,64}( [A-Za-z0-9]{1,64})*$'));
    }

    // Local para o Brasil todo: estado (UF) e código IBGE da cidade (7 dígitos; os dois
    // primeiros são o código do estado, então cidade e estado precisam combinar).
    // Feito com expressões regulares (uma tabela de 27 estados estourava o limite de
    // 1.000 expressões por gravação do Firestore).
    function localOk() {
      let d = request.resource.data;
      let uf = d.get('uf', null);
      let ibge = d.get('cidadeIbge', null);
      return (uf == null || uf == ''
              || (uf is string && uf.matches('^(AC|AL|AP|AM|BA|CE|DF|ES|GO|MA|MT|MS|MG|PA|PB|PR|PE|PI|RJ|RN|RS|RO|RR|SC|SP|SE|TO)$')))
          && (ibge == null
              || (ibge is int && uf is string
                  && (uf + string(ibge)).matches('^(RO11|AC12|AM13|RR14|PA15|AP16|TO17|MA21|PI22|CE23|RN24|PB25|PE26|AL27|SE28|BA29|MG31|ES32|RJ33|SP35|PR41|SC42|RS43|MS50|MT51|GO52|DF53)[0-9]{5}$')));
    }
    // Endereço detalhado opcional (negócios e anúncios): CEP com 8 números e textos curtos.
    function enderecoOk() {
      let cep = request.resource.data.get('cep', null);
      return (cep == null || cep == '' || (cep is string && cep.matches('^[0-9]{8}$')))
          && textoAte('bairro', 60)
          && textoAte('rua', 100)
          && textoAte('numero', 10)
          && textoAte('complemento', 60);
    }

    // Lista opcional com no máximo N itens.
    function listaAte(campo, max) {
      return listaValorAte(request.resource.data.get(campo, null), max);
    }
    function listaValorAte(v, max) {
      return v == null || (v is list && v.size() <= max);
    }

    // Foto só pode vir do próprio site: imagem dentro do documento (sem Storage),
    // Firebase Storage do projeto ou foto da conta Google. Bloqueia imagens de
    // sites de terceiros (rastreamento por IP, conteúdo impróprio).
    // Ponto do endereço (negócio ou anúncio) para "Perto de você": só lat/lng dentro do Brasil.
    function geoOk(g) {
      return g == null
          || (g is map && g.keys().hasOnly(['lat', 'lng'])
              && g.lat is number && g.lat >= -34 && g.lat <= 6
              && g.lng is number && g.lng >= -75 && g.lng <= -28);
    }
    function fotoOk(v) {
      return v == null || v == ''
          || (v is string && v.matches('^(data:image/(jpeg|png|webp);base64,|https://firebasestorage[.]googleapis[.]com/|https://lh3[.]googleusercontent[.]com/).*'));
    }

    function campoFotoOk(campo) {
      return !(campo in request.resource.data) || fotoOk(request.resource.data[campo]);
    }

    // Só os campos que mudaram (para não travar documentos antigos com campos a mais).
    function mudou() {
      return request.resource.data.diff(resource.data).affectedKeys();
    }

    // Escondido pela equipe (moderação): o dono não pode mostrar de novo nem tirar a marca.
    function travaModeracao() {
      return !mudou().hasAny(['ocultoPelaEquipe'])
          && !(resource.data.get('ocultoPelaEquipe', false) == true && mudou().hasAny(['ocultoAte']));
    }

    // Escondido agora (pela equipe ou com a conta do dono desativada: ocultoAte no futuro) ou agendado para depois.
    function abertoAgora(d) {
      return !(d.get('ocultoAte', null) is timestamp && d.ocultoAte > request.time)
          && !(d.get('agendadoPara', null) is timestamp && d.agendadoPara > request.time);
    }
    // Abrir UM documento (link direto): escondido ou agendado, só o dono e a equipe. Documento que não existe
    // continua respondendo "não existe" (o site mostra "não está mais disponível").
    // As listas continuam abertas (as buscas não filtram por ocultoAte; o site esconde na tela).
    function podeAbrir(campoDono) {
      return resource == null || abertoAgora(resource.data)
          || resource.data.get(campoDono, '') == request.auth.uid || admin();
    }

    // ---------- conteúdo escondido de verdade (guardar.js, 13/11/2026) ----------
    // Esconder = o conteúdo vai para guardados/{colecao}_{id} e no lugar fica um ESQUELETO vazio
    // { dono, tipo?, ocultoAte, ocultoPelaEquipe, guardado: true }: nem numa lista dá para ler o que foi escondido.
    // Quem pode esconder/devolver é conferido no guardados (create/delete); aqui só a forma do documento.
    function guardadoRef(col, id) {
      return /databases/$(database)/documents/guardados/$(col + '_' + id);
    }
    function ehEsqueleto() {
      return resource.data.get('guardado', false) == true;
    }
    // Documento cheio -> esqueleto, com o guardado sendo criado na mesma gravação.
    function viraEsqueleto(col, id, campo) {
      let n = request.resource.data;
      return n.get('guardado', false) == true
          && !ehEsqueleto()
          && n.keys().hasOnly([campo, 'tipo', 'ocultoAte', 'ocultoPelaEquipe', 'guardado'])
          && n[campo] == resource.data[campo]
          && n.get('tipo', null) == resource.data.get('tipo', null)
          && n.ocultoAte is timestamp && n.ocultoAte > request.time
          && !exists(guardadoRef(col, id))
          && n.ocultoPelaEquipe == getAfter(guardadoRef(col, id)).data.porEquipe;
    }
    // Esqueleto -> documento de volta, exatamente como estava guardado (o guardado sai na mesma gravação).
    function voltaDoGuardado(col, id) {
      return ehEsqueleto()
          && !existsAfter(guardadoRef(col, id))
          && request.resource.data == get(guardadoRef(col, id)).data.dados;
    }

    // A equipe esconde/mostra: só ocultoAte e a marca ocultoPelaEquipe.
    function moderacaoOk() {
      return request.resource.data.diff(resource.data).affectedKeys().hasOnly(['ocultoAte', 'ocultoPelaEquipe'])
          && dataOuNulo('ocultoAte')
          && request.resource.data.get('ocultoPelaEquipe', false) is bool;
    }

    // Campos novos ou alterados (campos antigos removidos não contam).
    function gravou() {
      let d = request.resource.data.diff(resource.data);
      return d.addedKeys().union(d.changedKeys());
    }

    // Bloqueio: bloqueios/{quemBloqueou}_{quemFoiBloqueado}
    function bloqueou(a, b) {
      return exists(/databases/$(database)/documents/bloqueios/$(a + '_' + b));
    }

    function bloqueioEntre(a, b) {
      return bloqueou(a, b) || bloqueou(b, a);
    }

    // ---------- avaliações e reclamações justas ----------
    // Id da conversa / do vínculo entre duas pessoas (ids em ordem).
    function par(a, b) {
      return a < b ? a + '_' + b : b + '_' + a;
    }

    // Conta com pelo menos 3 dias (usuarios.criadoEm é gravado no cadastro e não muda).
    function contaAntiga(uid) {
      let u = /databases/$(database)/documents/usuarios/$(uid);
      return exists(u)
          && get(u).data.get('criadoEm', null) is timestamp
          && get(u).data.criadoEm < request.time - duration.value(3, 'd');
    }

    // Quem já escreveu na conversa entre a e b (conversas.falaram, cada um marca só o seu).
    function falou(a, b, quem) {
      let c = /databases/$(database)/documents/conversas/$(par(a, b));
      return exists(c) && get(c).data.get('falaram', {}).get(quem, false) == true;
    }

    // Conta desativada (ou com exclusão pedida): perfis_publicos.desativadaAte no futuro.
    function desativada(uid) {
      let p = /databases/$(database)/documents/perfis_publicos/$(uid);
      return exists(p)
          && get(p).data.get('desativadaAte', null) is timestamp
          && get(p).data.desativadaAte > request.time;
    }

    // Campo opcional de data (ex.: ocultoAte): ausente, nulo ou uma data.
    function dataOuNulo(campo) {
      return !(campo in request.resource.data)
          || request.resource.data[campo] == null
          || request.resource.data[campo] is timestamp;
    }

    // Quem da equipe assumiu um item: alguém da equipe (ou ninguém), com nome curto.
    function responsavelOk() {
      let r = request.resource.data.get('responsavel', null);
      return (r == null || (r is string && exists(/databases/$(database)/documents/admins/$(r))))
          && valorTextoAte(request.resource.data.get('responsavelNome', null), 80)
          && (!mudou().hasAny(['responsavel']) || request.resource.data.get('assumidoEm', null) == request.time);
    }

    // Anti-spam: cada mensagem, publicação ou comentário conta no limite diário da pessoa
    // (ritmo/{uid}, gravado junto na mesma operação).
    function ritmoOk(tipo) {
      let r = getAfter(/databases/$(database)/documents/ritmo/$(request.auth.uid)).data;
      return r.ultimaEm == request.time && r.ultimoTipo == tipo;
    }

    function donoDoNegocio(negocioId) {
      return get(/databases/$(database)/documents/negocios/$(negocioId)).data.donoId;
    }
