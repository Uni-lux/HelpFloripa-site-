      // Edição de n_ / c_ pelo próprio autor: só nota, comentário e (uma vez) o pedido confirmado.
      allow update: if logado()
                    && emailOk()
                    && resource.data.autorId == request.auth.uid
                    && (id.matches('^n_.*') || id.matches('^c_.*'))
                    && mudou().hasOnly(['nota', 'comentario', 'atualizadoEm', 'pedidoId'])
                    && request.resource.data.nota is int && request.resource.data.nota >= 1 && request.resource.data.nota <= 5
                    && valorTextoAte(request.resource.data.get('comentario', ''), 500)
                    && request.resource.data.atualizadoEm == request.time
                    && (!mudou().hasAny(['pedidoId'])
                        || (!('pedidoId' in resource.data)
                            && pedidoOk(request.resource.data, id.matches('^n_.*') ? request.auth.uid : resource.data.alvoId, resource.data.negocioId)))
                    && !bloqueioEntre(request.auth.uid, resource.data.alvoId)
                    // nota mudou: o resumo precisa trocar a nota antiga pela nova NESTA mesma gravação
                    && (request.resource.data.nota == resource.data.nota
                        || (getAfter(/databases/$(database)/documents/notas/$(resource.data.chaves[0])).data.ultima == id
                            && getAfter(/databases/$(database)/documents/notas/$(resource.data.chaves[0])).data.soma
                               == get(/databases/$(database)/documents/notas/$(resource.data.chaves[0])).data.soma
                                  - resource.data.nota + request.resource.data.nota));

      allow delete: if false;
    }


    // =====================================================
    // RESUMO DAS NOTAS (média e quantas de cada estrela)
    //   neg_{negocioId} · cli_{uid} · pub_{uid} · post_{postId}
    // Só aceita somar exatamente a avaliação nova gravada junto.
    // =====================================================
    match /notas/{chave} {

      function caminhoNova() {
        return /databases/$(database)/documents/avaliacoes/$(request.resource.data.ultima);
      }

      function avaliacaoNova() {
        return request.resource.data.ultima is string
            && !exists(caminhoNova())
            && existsAfter(caminhoNova())
            && getAfter(caminhoNova()).data.autorId == request.auth.uid
            && chave in getAfter(caminhoNova()).data.chaves;
      }

      function estrela() {
        return 'n' + string(getAfter(caminhoNova()).data.nota);
      }

      allow read: if logado();

      allow create: if logado()
                    && avaliacaoNova()
                    && request.resource.data.keys().hasOnly(['total', 'soma', 'n1', 'n2', 'n3', 'n4', 'n5', 'ultima', 'atualizadoEm'])
                    && request.resource.data.keys().hasAll(['total', 'soma', 'n1', 'n2', 'n3', 'n4', 'n5', 'ultima'])
                    && request.resource.data.total == 1
                    && request.resource.data.soma == getAfter(caminhoNova()).data.nota
                    && request.resource.data[estrela()] == 1
                    && request.resource.data.n1 + request.resource.data.n2 + request.resource.data.n3
                       + request.resource.data.n4 + request.resource.data.n5 == 1;

      allow update: if logado()
                    && avaliacaoNova()
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['total', 'soma', 'ultima', 'atualizadoEm', estrela()])
                    && request.resource.data.total == resource.data.total + 1
                    && request.resource.data.soma == resource.data.soma + getAfter(caminhoNova()).data.nota
                    && request.resource.data[estrela()] == resource.data.get(estrela(), 0) + 1;

      // Edição de uma avaliação n_/c_: troca a nota antiga pela nova (o total não muda).
      function edicao() {
        let antes = get(caminhoNova()).data;
        let depois = getAfter(caminhoNova()).data;
        return request.resource.data.ultima is string
            && exists(caminhoNova())
            && depois.autorId == request.auth.uid
            && chave in depois.chaves
            && antes.nota != depois.nota
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['soma', 'ultima', 'atualizadoEm', 'n' + string(antes.nota), 'n' + string(depois.nota)])
            && request.resource.data.total == resource.data.total
            && request.resource.data.soma == resource.data.soma - antes.nota + depois.nota
            && resource.data.get('n' + string(antes.nota), 0) >= 1
            && request.resource.data['n' + string(antes.nota)] == resource.data.get('n' + string(antes.nota), 0) - 1
            && request.resource.data['n' + string(depois.nota)] == resource.data.get('n' + string(depois.nota), 0) + 1;
      }

      allow update: if logado() && edicao();

      allow delete: if false;
    }


    // =====================================================
    // RECLAMAÇÕES: resposta pública do negócio
    // Reclamação = avaliação de negócio com nota 1 ou 2 (avaliacoes/{id}).
    // reclamacoes/{id da avaliação}: o negócio avaliado responde uma vez
    // (pode editar o texto) e o cliente que reclamou marca como resolvida.
    // =====================================================
    match /reclamacoes/{avalId} {

      allow read: if logado();

      function aval() {
        return get(/databases/$(database)/documents/avaliacoes/$(avalId)).data;
      }

      function respostaOk(d) {
        return d.texto is string && d.texto.size() >= 2 && d.texto.size() <= 1000;
      }

      allow create: if logado()
                    && emailOk()
                    && exists(/databases/$(database)/documents/avaliacoes/$(avalId))
                    && aval().tipo == 'negocio'
                    && aval().nota <= 2
                    && aval().alvoId == request.auth.uid
                    && request.resource.data.keys().hasOnly(['texto', 'autorId', 'criadoEm'])
                    && request.resource.data.autorId == request.auth.uid
                    && respostaOk(request.resource.data)
                    && request.resource.data.criadoEm == request.time;

      allow update: if logado()
                    && emailOk()
                    && request.resource.data.autorId == resource.data.autorId
                    && (
                      // o negócio corrige a resposta
                      (resource.data.autorId == request.auth.uid
                        && mudou().hasOnly(['texto', 'editadoEm'])
                        && respostaOk(request.resource.data)
                        && request.resource.data.editadoEm == request.time)
                      ||
                      // o cliente que reclamou marca como resolvida (ou não)
                      (aval().autorId == request.auth.uid
                        && mudou().hasOnly(['resolvido', 'resolvidoEm'])
                        && request.resource.data.resolvido is bool
                        && request.resource.data.resolvidoEm == request.time)
                    );

      allow delete: if false;
    }


    // =====================================================
    // RECLAMAÇÕES ABERTAS PELO CLIENTE (separadas da nota)
    // queixas/{negocioId}_{cliente}: uma por cliente em cada negócio.
    //   - abrir: e-mail confirmado, conta com 3 dias, já mandou mensagem ao negócio
    //     (o negócio não precisa ter respondido) e dentro da cota (cotas/{uid}).
    //   - o dono do negócio responde (pode corrigir a resposta);
    //   - só o cliente marca como resolvida; resolvida, pode reabrir (conta na cota).
    //   - ninguém apaga.
    // cotas/{uid}: no máximo 3 reclamações a cada 30 dias (janela que começa na primeira).
    // =====================================================
    match /queixas/{id} {

      function motivoOk(m) {
        return m in ['atraso', 'qualidade', 'cobranca', 'ausencia', 'atendimento', 'golpe', 'outro'];
      }

      function relatoOk(d) {
        return motivoOk(d.motivo)
            && d.texto is string && d.texto.size() >= 20 && d.texto.size() <= 1000
            && fotoOk(d.get('foto', ''));
      }

      // A cota foi atualizada nesta mesma gravação.
      function contou() {
        return getAfter(/databases/$(database)/documents/cotas/$(request.auth.uid)).data.ultimaEm == request.time;
      }

      allow read: if logado();

      allow create: if logado()
                    && emailOk()
                    && request.resource.data.keys().hasOnly(['negocioId', 'alvoId', 'autorId', 'autorNome', 'negocioNome', 'tipo', 'motivo', 'texto', 'foto',
                                                              'status', 'abertaEm', 'criadoEm', 'aberturas', 'resposta', 'respondidaEm', 'resolvidaEm'])
                    && request.resource.data.autorId == request.auth.uid
                    && id == request.resource.data.negocioId + '_' + request.auth.uid
                    && request.resource.data.alvoId == donoDoNegocio(request.resource.data.negocioId)
                    && request.resource.data.alvoId != request.auth.uid
                    && request.resource.data.tipo in ['servicos', 'delivery', 'lojinha', 'imoveis', 'saude']
                    && relatoOk(request.resource.data)
                    && valorTextoAte(request.resource.data.get('autorNome', ''), 80)
                    && valorTextoAte(request.resource.data.get('negocioNome', ''), 80)
                    && request.resource.data.status == 'aberta'
                    && request.resource.data.aberturas == 1
                    && request.resource.data.get('resposta', null) == null
                    && request.resource.data.criadoEm == request.time
                    && request.resource.data.abertaEm == request.time
                    && contaAntiga(request.auth.uid)
                    && falou(request.auth.uid, request.resource.data.alvoId, request.auth.uid)
                    && contou();

      allow update: if logado()
                    && emailOk()
                    && request.resource.data.autorId == resource.data.autorId
                    && request.resource.data.alvoId == resource.data.alvoId
                    && request.resource.data.negocioId == resource.data.negocioId
                    && (
                      // o negócio responde ou corrige a resposta (enquanto está aberta)
                      (resource.data.alvoId == request.auth.uid
                        && resource.data.status == 'aberta'
                        && mudou().hasOnly(['resposta', 'respondidaEm'])
                        && request.resource.data.resposta is string
                        && request.resource.data.resposta.size() >= 2 && request.resource.data.resposta.size() <= 1000
                        && request.resource.data.respondidaEm == request.time)
                      ||
                      // o cliente marca como resolvida
                      (resource.data.autorId == request.auth.uid
                        && resource.data.status == 'aberta'
                        && mudou().hasOnly(['status', 'resolvidaEm'])
                        && request.resource.data.status == 'resolvida'
                        && request.resource.data.resolvidaEm == request.time)
                      ||
                      // o cliente reabre (com relato novo), contando na cota
                      (resource.data.autorId == request.auth.uid
                        && resource.data.status == 'resolvida'
                        && mudou().hasOnly(['motivo', 'texto', 'foto', 'status', 'abertaEm', 'aberturas', 'resposta', 'respondidaEm', 'resolvidaEm'])
                        && relatoOk(request.resource.data)
                        && request.resource.data.status == 'aberta'
                        && request.resource.data.abertaEm == request.time
                        && request.resource.data.aberturas == resource.data.get('aberturas', 1) + 1
                        && request.resource.data.get('resposta', null) == null
                        && contou())
                    );

      allow delete: if false;
    }
