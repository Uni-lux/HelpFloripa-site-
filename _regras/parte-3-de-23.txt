      allow read: if logado()
                  && request.auth.uid == userId;
      // Dados pessoais (e-mail, telefone, nascimento): na equipe, só dono e moderação (LGPD: o mínimo necessário).
      allow read: if podeModerar();
      // A equipe aprova ou recusa o selo de verificado.
      allow update: if podeModerar() && (userId != request.auth.uid || papel() == 'dono')
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['verificado', 'solicitacaoVerificacao', 'verificacaoRespondidaEm'])
                    && seloLiberado(userId);

      // criadoEm é a data real do cadastro (conta antiga = pode avaliar/reclamar).
      allow create: if logado()
                    && request.auth.uid == userId
                    && request.resource.data.uid == userId
                    && request.resource.data.aceiteTermos == true
                    && (!('criadoEm' in request.resource.data) || request.resource.data.criadoEm == request.time)
                    && emailConfirmadoOk()
                    && emailDaContaOk()
                    && naoTocaCamposProtegidosNoCreate();

      // E-mail confirmado (o painel mostra): a conta só grava o valor verdadeiro, o que o login diz (token).
      function emailConfirmadoOk() {
        return !('emailConfirmado' in request.resource.data)
            || request.resource.data.emailConfirmado == (request.auth.token.email_verified == true);
      }
      // E-mail gravado na conta = o do login (o painel mostra e busca por ele; antes dava para gravar o e-mail de outra
      // pessoa). Maiúsculas não contam (o cadastro grava em minúsculas).
      function emailDaContaOk() {
        return !('email' in request.resource.data)
            || request.resource.data.email == null || request.resource.data.email == ''
            || request.resource.data.email.lower() == request.auth.token.email.lower();
      }

      // criadoEm não muda depois de gravado (cadastro antigo sem a data só pode gravar a data de agora).
      allow update: if logado()
                    && request.auth.uid == userId
                    && request.resource.data.uid == resource.data.uid
                    && (!mudou().hasAny(['criadoEm'])
                        || (!('criadoEm' in resource.data) && request.resource.data.criadoEm == request.time))
                    && (!mudou().hasAny(['emailConfirmado']) || emailConfirmadoOk())
                    && (!mudou().hasAny(['email']) || emailDaContaOk())
                    && naoTocaCamposProtegidosNoUpdate();

      allow delete: if logado()
                    && request.auth.uid == userId;
    }


    // =====================================================
    // PERFIL PÚBLICO
    // Usado somente para busca e exibição pública.
    // NÃO deve conter e-mail, telefone ou data de nascimento.
    // =====================================================
    match /perfis_publicos/{userId} {

      // O @ exibido no perfil precisa estar reservado em nicknames/ pelo
      // próprio usuário. getAfter enxerga a reserva feita na mesma transação
      // (cadastro) ou numa escrita anterior (edição do perfil).
      function nickDoDono() {
        let d = request.resource.data;
        return !('nickname' in d)
            || d.nickname == null
            || d.nickname == ''
            || getAfter(/databases/$(database)/documents/nicknames/$(d.nickname)).data.uid == request.auth.uid;
      }

      // Trocou de @: o antigo tem que ser liberado na mesma gravação (ninguém guarda vários @).
      function liberouNickAntigo() {
        let antigo = resource.data.get('nickname', '');
        return antigo == null || antigo == '' || antigo == request.resource.data.get('nickname', '')
            || !existsAfter(/databases/$(database)/documents/nicknames/$(antigo));
      }

      function tamanhosOk() {
        return textoAte('nome', 80)
            && textoAte('bio', 400)
            && textoAte('cidade', 80)
            && localOk()
            && listaAte('busca', 12)
            && dataOuNulo('desativadaAte')
            // "online agora": só a hora de agora (ou nada); antes dava para gravar uma data no futuro e ficar sempre online
            && (request.resource.data.get('ultimoAcesso', null) == null
                || request.resource.data.ultimoAcesso == request.time
                || (resource != null && request.resource.data.ultimoAcesso == resource.data.get('ultimoAcesso', null)));
      }

      // Campos que o site grava no perfil público. Qualquer outro é recusado.
      function camposPerfil() {
        return ['uid', 'nome', 'nickname', 'cidade', 'uf', 'cidadeIbge', 'bio', 'fotoPerfil', 'fotoCapa',
                'socialVisibilidade', 'duoCapa', 'ultimoAcesso', 'criadoEm', 'atualizadoEm',
                'busca',   // palavras do nome e do @ (minúsculas, sem acento) para a busca de pessoas
                'desativadaAte',   // conta desativada até esta data (ou exclusão pedida)
                'atendimento'];    // negócios: horário, modo ausente e tempo médio de resposta
      }

      function opcoesOk() {
        let d = request.resource.data;
        return (!('socialVisibilidade' in d) || d.socialVisibilidade in ['todos', 'seguidores', 'amigos', 'ocultar'])
            && (!('duoCapa' in d) || d.duoCapa in ['clicavel', 'foto', 'rosa', 'ocultar'])
            && (!('nickname' in d) || valorTextoAte(d.nickname, 20))
            && atendimentoOk();
      }

      // Faixa de horário "HH:MM-HH:MM" (00:00 a 23:59).
      function faixaOk(v) {
        return v is string && v.matches('^([01][0-9]|2[0-3]):[0-5][0-9]-([01][0-9]|2[0-3]):[0-5][0-9]$');
      }
      // Horário de cada dia da semana: lista com 7 textos, de domingo (0) a sábado (6),
      // "HH:MM-HH:MM" ou "" (fechado). Uma expressão regular só confere a lista toda
      // (uma por dia estourava o limite de 1.000 expressões por gravação).
      function semanaOk(s) {
        return s == null
            || (s is list && s.size() == 7
                && s.join(';').matches('^(([01][0-9]|2[0-3]):[0-5][0-9]-([01][0-9]|2[0-3]):[0-5][0-9])?(;(([01][0-9]|2[0-3]):[0-5][0-9]-([01][0-9]|2[0-3]):[0-5][0-9])?){6}$'));
      }

      // Atendimento (negócios): horário (resumo e por dia), pausa de almoço, aviso de
      // ausência com data de volta e tempo médio de resposta.
      function atendimentoOk() {
        let a = request.resource.data.get('atendimento', null);
        return a == null
            || (a is map
                && a.keys().hasOnly(['horario', 'semana', 'almoco', 'ausente', 'mensagem', 'volta', 'resposta'])
                && (!('semana' in a) || semanaOk(a.semana))
                && (!('almoco' in a) || a.almoco == null || faixaOk(a.almoco))
                && (!('volta' in a) || a.volta == null || a.volta is timestamp)
                && (!('ausente' in a) || a.ausente is bool)
                && (!('mensagem' in a) || (a.mensagem is string && a.mensagem.size() <= 200))
                && (!('horario' in a) || a.horario == null
                    || (a.horario is map && a.horario.keys().hasOnly(['dias', 'inicio', 'fim'])
                        && a.horario.dias is list && a.horario.dias.size() <= 7
                        && a.horario.inicio is string && a.horario.inicio.size() == 5
                        && a.horario.fim is string && a.horario.fim.size() == 5))
                && (!('resposta' in a)
                    || (a.resposta is map && a.resposta.keys().hasOnly(['media', 'n'])
                        && a.resposta.media is number && a.resposta.media >= 0 && a.resposta.media <= 1440
                        && a.resposta.n is int && a.resposta.n >= 0 && a.resposta.n <= 1000)));
      }

      // Foto nova precisa vir do próprio site (fotos antigas não travam a edição).
      function fotosOk() {
        return ((resource != null && !mudou().hasAny(['fotoPerfil'])) || campoFotoOk('fotoPerfil'))
            && ((resource != null && !mudou().hasAny(['fotoCapa'])) || campoFotoOk('fotoCapa'));
      }

      // Quem foi bloqueado não abre o perfil de quem bloqueou (a busca em lista continua
      // aberta a quem está logado, mas o perfil em si não carrega).
      allow get: if logado()
                 && (request.auth.uid == userId || !bloqueou(userId, request.auth.uid) || admin());
      allow list: if logado();
      // A equipe marca o selo e esconde contas suspensas/banidas.
      allow update: if podeModerar()
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['verificado', 'desativadaAte'])
                    && dataOuNulo('desativadaAte')
                    // o selo público só acende se a conta já tem o selo (dado pelas regras de usuarios)
                    && (request.resource.data.get('verificado', false) != true
                        || get(/databases/$(database)/documents/usuarios/$(userId)).data.get('verificado', false) == true);

      allow create: if logado()
                    && request.auth.uid == userId
                    && request.resource.data.uid == userId
                    && naoTocaCamposProtegidosNoCreate()
                    && request.resource.data.keys().hasOnly(camposPerfil())
                    && nickDoDono()
                    && tamanhosOk()
                    && opcoesOk()
                    && fotosOk();

      allow update: if logado()
                    && request.auth.uid == userId
                    && request.resource.data.uid == resource.data.uid
                    && naoTocaCamposProtegidosNoUpdate()
                    && (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['nickname'])
                        || (nickDoDono() && liberouNickAntigo()))
                    && gravou().hasOnly(camposPerfil())
                    && (!mudou().hasAny(['desativadaAte']) || !sancionado(request.auth.uid))
                    && tamanhosOk()
                    && opcoesOk()
                    && fotosOk();

      // Apagar o perfil público libera o @ na mesma gravação (antes o @ ficava preso para sempre).
      allow delete: if logado()
                    && request.auth.uid == userId
                    && (resource.data.get('nickname', '') == ''
                        || !existsAfter(/databases/$(database)/documents/nicknames/$(resource.data.nickname)));
    }


    // =====================================================
    // RESERVA DE NICKNAMES
    // =====================================================
    match /nicknames/{nickname} {

      // Sem login só confere UM @ (o cadastro vê se está livre; link.html leva ao perfil). Listar todos os @ (e os uids)
      // só com a conta entrada.
      allow get: if true;
      allow list: if logado();

      // Só reserva o @ que o próprio perfil passa a usar na mesma gravação (sem guardar @ sobrando).
      allow create: if logado()
                    && request.auth.uid == request.resource.data.uid
                    && nickname.matches('^[a-z0-9._]{3,20}$')
                    && getAfter(/databases/$(database)/documents/perfis_publicos/$(request.auth.uid)).data.get('nickname', '') == nickname;

      allow update: if false;

      allow delete: if logado()
                    && request.auth.uid == resource.data.uid;
    }


    // =====================================================
    // SEGUIDORES
    // Documento: relacoes/{seguidorId}_{alvoId}
    // =====================================================
    match /relacoes/{id} {

      allow read: if logado();

      allow create: if logado()
                    && emailOk()
                    && request.resource.data.keys().hasOnly(['tipo', 'seguidorId', 'alvoId', 'criadoEm'])
                    && request.resource.data.tipo == 'seguir'
                    && request.resource.data.seguidorId == request.auth.uid
                    && request.resource.data.alvoId is string
                    && request.resource.data.alvoId != request.auth.uid
                    && id == request.auth.uid + '_' + request.resource.data.alvoId
                    && request.resource.data.criadoEm == request.time
                    && !bloqueioEntre(request.auth.uid, request.resource.data.alvoId);

      allow update: if false;

      // Quem segue pode deixar de seguir; quem é seguido pode remover o seguidor.
      allow delete: if logado()
                    && (resource.data.seguidorId == request.auth.uid
                        || resource.data.alvoId == request.auth.uid);
    }
