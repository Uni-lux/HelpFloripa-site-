      // Janela nova (primeira vez ou a anterior já passou de 30 dias): começa agora com 1.
      // Dentro da janela: soma 1, até 3.
      allow create: if logado()
                    && request.auth.uid == uid
                    && request.resource.data.keys().hasOnly(['inicio', 'n', 'ultimaEm'])
                    && request.resource.data.inicio == request.time
                    && request.resource.data.n == 1
                    && request.resource.data.ultimaEm == request.time;

      allow update: if logado()
                    && request.auth.uid == uid
                    && request.resource.data.keys().hasOnly(['inicio', 'n', 'ultimaEm'])
                    && request.resource.data.ultimaEm == request.time
                    && ((resource.data.inicio < request.time - duration.value(30, 'd')
                         && request.resource.data.inicio == request.time
                         && request.resource.data.n == 1)
                        ||
                        (resource.data.inicio >= request.time - duration.value(30, 'd')
                         && request.resource.data.inicio == resource.data.inicio
                         && request.resource.data.n == resource.data.n + 1
                         && request.resource.data.n <= 3));

      allow delete: if false;
    }


    // =====================================================
    // DENÚNCIAS: denuncias/{tipo}_{item}_{quemDenunciou}
    // Uma por pessoa em cada item. Só quem denunciou consegue ler a própria
    // denúncia (para saber que já enviou). A equipe analisa pelo painel.
    // O item precisa ser mesmo de quem foi denunciado.
    // =====================================================
    match /denuncias/{id} {

      function itemDoAlvo(d) {
        return (d.tipo == 'perfil' && d.itemId == d.alvoId)
            || (d.tipo == 'publicacao' && get(/databases/$(database)/documents/diario/$(d.itemId)).data.autorId == d.alvoId)
            || (d.tipo == 'comentario' && get(/databases/$(database)/documents/comentarios/$(d.itemId)).data.autorId == d.alvoId)
            || (d.tipo == 'negocio' && get(/databases/$(database)/documents/negocios/$(d.itemId)).data.donoId == d.alvoId)
            || (d.tipo == 'mensagem'
                && d.conversaId is string
                && request.auth.uid in get(/databases/$(database)/documents/conversas/$(d.conversaId)).data.participantes
                && get(/databases/$(database)/documents/conversas/$(d.conversaId)/mensagens/$(d.itemId)).data.remetenteId == d.alvoId);
      }

      allow get: if logado() && resource.data.autorId == request.auth.uid;
      allow read: if admin();
      // A equipe registra a análise e quem assumiu o caso (para ninguém fazer o mesmo trabalho).
      allow update: if podeModerar()
                    && mudou().hasOnly(['status', 'acao', 'nota', 'analisadaEm', 'analisadaPor', 'responsavel', 'responsavelNome', 'assumidoEm'])
                    && request.resource.data.status in ['nova', 'em_analise', 'resolvida', 'descartada']
                    && (!mudou().hasAny(['status', 'acao', 'nota', 'analisadaEm', 'analisadaPor']) || request.resource.data.analisadaPor == request.auth.uid)
                    && responsavelOk()
                    && valorTextoAte(request.resource.data.get('nota', ''), 1000);

      allow create: if logado()
                    && emailOk()
                    && request.resource.data.keys().hasOnly(['tipo', 'alvoId', 'itemId', 'conversaId', 'autorId', 'motivo', 'detalhe', 'trecho', 'contexto', 'status', 'criadoEm'])
                    && request.resource.data.autorId == request.auth.uid
                    && request.resource.data.alvoId is string
                    && request.resource.data.alvoId != request.auth.uid
                    && request.resource.data.itemId is string
                    && request.resource.data.tipo in ['perfil', 'publicacao', 'comentario', 'mensagem', 'negocio']
                    && id == request.resource.data.tipo + '_' + request.resource.data.itemId + '_' + request.auth.uid
                    && request.resource.data.motivo in ['golpe', 'falso', 'assedio', 'odio', 'improprio', 'spam', 'menor', 'outro']
                    && valorTextoAte(request.resource.data.get('detalhe', ''), 500)
                    && valorTextoAte(request.resource.data.get('trecho', ''), 300)
                    // mensagem: ids de algumas mensagens antes e depois (a equipe abre uma por uma)
                    && (!('contexto' in request.resource.data)
                        || (request.resource.data.tipo == 'mensagem'
                            && request.resource.data.contexto is list
                            && request.resource.data.contexto.size() <= 12))
                    && request.resource.data.status == 'nova'
                    && request.resource.data.criadoEm == request.time
                    && itemDoAlvo(request.resource.data);

      allow delete: if false;
    }


    // =====================================================
    // ADMINISTRAÇÃO
    // admins/{uid}      : quem é da equipe (criado à mão no console do Firebase)
    // sancoes/{uid}     : suspensão (até uma data) ou banimento
    // avisos/{id}       : aviso da equipe para uma pessoa
    // comunicados/{id}  : faixa para todos (ex.: manutenção, novidade)
    // suporte/{id}      : chamados de suporte (+ mensagens)
    // parcerias/{id}    : interessados em ser sócio, investidor ou parceiro
    // admin_log/{id}    : registro de tudo o que a equipe faz
    // estatisticas/{dia}, relatorios/{id}: gravados pelo relatório automático (Admin SDK)
    // =====================================================
    match /admins/{uid} {
      allow get: if logado() && (request.auth.uid == uid || admin());
      allow list: if admin();
      allow write: if false;
    }

    match /sancoes/{uid} {
      allow get: if logado() && request.auth.uid == uid;
      allow read: if admin();
      allow create, update: if podeModerar()
                    && request.resource.data.keys().hasOnly(['tipo', 'motivo', 'ate', 'por', 'em', 'historico'])
                    && request.resource.data.tipo in ['suspensao', 'banimento']
                    && valorTextoAte(request.resource.data.get('motivo', ''), 500)
                    && (request.resource.data.tipo == 'banimento' || request.resource.data.ate is timestamp)
                    && request.resource.data.por == request.auth.uid
                    && request.resource.data.em == request.time
                    && listaAte('historico', 50);
      allow delete: if podeModerar();
    }

    match /avisos/{id} {
      allow read: if logado() && resource.data.uid == request.auth.uid;
      allow read: if admin();
      allow create: if admin()
                    && request.resource.data.keys().hasOnly(['uid', 'titulo', 'texto', 'tipo', 'por', 'em', 'lidoEm'])
                    && request.resource.data.uid is string
                    && valorTextoAte(request.resource.data.get('titulo', ''), 120)
                    && request.resource.data.texto is string && request.resource.data.texto.size() <= 2000
                    && request.resource.data.tipo in ['info', 'alerta', 'grave']
                    && request.resource.data.por == request.auth.uid
                    && request.resource.data.em == request.time
                    && request.resource.data.lidoEm == null;
      // quem recebeu marca como lido
      allow update: if logado()
                    && resource.data.uid == request.auth.uid
                    && mudou().hasOnly(['lidoEm'])
                    && request.resource.data.lidoEm == request.time;
      allow delete: if false;
    }

    match /comunicados/{id} {
      allow read: if logado();
      allow create, update: if podeModerar()
                    && request.resource.data.keys().hasOnly(['titulo', 'texto', 'tipo', 'ativo', 'link', 'por', 'em', 'ate'])
                    && valorTextoAte(request.resource.data.get('titulo', ''), 120)
                    && valorTextoAte(request.resource.data.get('texto', ''), 600)
                    && request.resource.data.tipo in ['info', 'novidade', 'alerta']
                    && request.resource.data.ativo is bool
                    && request.resource.data.por == request.auth.uid;
      allow delete: if podeModerar();
    }

    match /suporte/{id} {
      function camposChamado() {
        return ['uid', 'nome', 'email', 'assunto', 'categoria', 'status', 'criadoEm', 'atualizadoEm', 'ultimaDe', 'atendidoPor', 'responsavel', 'responsavelNome', 'assumidoEm'];
      }
      allow read: if logado() && resource.data.uid == request.auth.uid;
      allow read: if admin();
      // Abrir chamado: não exige e-mail confirmado nem conta livre (quem foi suspenso pode contestar).
      allow create: if logado()
                    && request.resource.data.keys().hasOnly(camposChamado())
                    && request.resource.data.uid == request.auth.uid
                    && request.resource.data.assunto is string && request.resource.data.assunto.size() >= 3 && request.resource.data.assunto.size() <= 120
                    && request.resource.data.categoria in ['conta', 'pagamento', 'golpe', 'denuncia', 'contestacao', 'negocio', 'sugestao', 'outro']
                    && valorTextoAte(request.resource.data.get('nome', ''), 80)
                    && valorTextoAte(request.resource.data.get('email', ''), 120)
                    && request.resource.data.status == 'aberto'
                    && request.resource.data.criadoEm == request.time
                    && request.resource.data.atualizadoEm == request.time
                    && request.resource.data.ultimaDe == 'usuario'
                    && !request.resource.data.keys().hasAny(['atendidoPor', 'responsavel', 'responsavelNome', 'assumidoEm']);
      // A equipe abre uma conversa de suporte com qualquer pessoa ("Suporte Help Floripa"):
      // o chamado já nasce respondido, com a primeira mensagem da equipe gravada junto.
      allow create: if admin()
                    && request.resource.data.keys().hasOnly(camposChamado().concat(['abertoPor']))
                    && request.resource.data.uid is string
                    && exists(/databases/$(database)/documents/usuarios/$(request.resource.data.uid))
                    && request.resource.data.abertoPor == request.auth.uid
                    && request.resource.data.assunto is string && request.resource.data.assunto.size() >= 3 && request.resource.data.assunto.size() <= 120
                    && request.resource.data.categoria in ['conta', 'pagamento', 'golpe', 'denuncia', 'contestacao', 'negocio', 'sugestao', 'outro']
                    && valorTextoAte(request.resource.data.get('nome', ''), 80)
                    && valorTextoAte(request.resource.data.get('email', ''), 120)
                    && request.resource.data.status == 'respondido'
                    && request.resource.data.criadoEm == request.time
                    && request.resource.data.atualizadoEm == request.time
                    && request.resource.data.ultimaDe == 'equipe'
                    // quem abre pode já ficar como responsável pelo chamado
                    && (!('responsavel' in request.resource.data)
                        || (request.resource.data.responsavel == request.auth.uid
                            && request.resource.data.get('assumidoEm', null) == request.time
                            && valorTextoAte(request.resource.data.get('responsavelNome', ''), 80)))
                    && existsAfter(/databases/$(database)/documents/suporte/$(id)/mensagens/inicio);
      allow update: if logado()
                    && ((resource.data.uid == request.auth.uid
                         && mudou().hasOnly(['status', 'atualizadoEm', 'ultimaDe'])
                         && request.resource.data.status in ['aberto', 'fechado']
                         && request.resource.data.ultimaDe == 'usuario')
                        || (admin()
                         && mudou().hasOnly(['status', 'atualizadoEm', 'ultimaDe', 'atendidoPor', 'responsavel', 'responsavelNome', 'assumidoEm'])
                         && request.resource.data.status in ['aberto', 'respondido', 'fechado']
                         && responsavelOk()))
                    && request.resource.data.atualizadoEm == request.time;
      // A equipe só assume ou passa o chamado (sem mexer na data da conversa).
      allow update: if admin()
                    && mudou().hasOnly(['responsavel', 'responsavelNome', 'assumidoEm'])
                    && responsavelOk();
      allow delete: if false;
