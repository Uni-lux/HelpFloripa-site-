      // Com bloqueio entre os dois, ninguém mexe na conversa (evita "recado"
      // pelo resumo da última mensagem).
      allow update: if logado()
                    && request.auth.uid in resource.data.participantes
                    && request.resource.data.participantes == resource.data.participantes
                    && !bloqueioEntre(resource.data.participantes[0], resource.data.participantes[1])
                    && mudou().hasOnly(['ultimaMensagem', 'ultimaMensagemRemetenteId', 'atualizadoEm', 'lidoEm', 'vistoEm', 'digitando', 'contexto', 'arquivadaPara', 'ocultaPara', 'falaram', 'fixadaPara', 'silenciadaPara', 'mensagemFixada'])
                    // o resumo da última mensagem só pode ser escrito por quem mandou
                    && (!mudou().hasAny(['ultimaMensagem', 'ultimaMensagemRemetenteId'])
                        || request.resource.data.get('ultimaMensagemRemetenteId', '') == request.auth.uid)
                    && soMinhaChave('lidoEm')
                    && soMinhaChave('vistoEm')
                    && soMinhaChave('digitando')
                    // arquivar / apagar "só para mim": cada um mexe só na própria marca
                    && soMinhaChave('arquivadaPara')
                    && soMinhaChave('ocultaPara')
                    // fixar no topo / silenciar avisos: cada um só a própria marca
                    && soMinhaChave('fixadaPara')
                    && soMinhaChave('silenciadaPara')
                    // falaram: cada um marca só que ELE escreveu (usado para liberar avaliação e reclamação)
                    && soMinhaChave('falaram')
                    && (!mudou().hasAny(['falaram']) || request.resource.data.falaram[request.auth.uid] == true)
                    && (!mudou().hasAny(['contexto']) || contextoOk())
                    // mensagem fixada no topo da conversa (qualquer um dos dois fixa ou tira)
                    && (!mudou().hasAny(['mensagemFixada'])
                        || request.resource.data.mensagemFixada == null
                        || (request.resource.data.mensagemFixada is map
                            && request.resource.data.mensagemFixada.keys().hasOnly(['id', 'texto', 'por', 'em'])
                            && request.resource.data.mensagemFixada.id is string
                            && request.resource.data.mensagemFixada.por == request.auth.uid
                            && valorTextoAte(request.resource.data.mensagemFixada.get('texto', ''), 200)))
                    && textoAte('ultimaMensagem', 200);

      allow delete: if logado()
                    && request.auth.uid in resource.data.participantes;


      // ===================================================
      // MENSAGENS DA CONVERSA
      // ===================================================
      match /mensagens/{mensagemId} {

        function participante() {
          return request.auth.uid in
                 get(/databases/$(database)/documents/conversas/$(conversaId)).data.participantes;
        }

        function semBloqueio() {
          let p = get(/databases/$(database)/documents/conversas/$(conversaId)).data.participantes;
          return !bloqueioEntre(p[0], p[1]);
        }

        // A outra pessoa da conversa não pode estar com a conta desativada.
        function outroAtivo() {
          let p = get(/databases/$(database)/documents/conversas/$(conversaId)).data.participantes;
          return !desativada(p[0] == request.auth.uid ? p[1] : p[0]);
        }

        // Resposta (citação): precisa apontar para uma mensagem real desta conversa,
        // com o autor verdadeiro. Impede citar algo que a outra pessoa nunca disse.
        function respostaOk(r) {
          let caminho = /databases/$(database)/documents/conversas/$(conversaId)/mensagens/$(r.id);
          return r is map
              && r.keys().hasOnly(['id', 'texto', 'remetenteId'])
              && r.id is string
              && valorTextoAte(r.get('texto', ''), 200)
              && exists(caminho)
              && get(caminho).data.remetenteId == r.remetenteId;
        }

        function mensagemOk() {
          let d = request.resource.data;
          return d.keys().hasOnly(['remetenteId', 'texto', 'criadoEm', 'resposta', 'tipoArquivo', 'arquivoUrl', 'nomeArquivo', 'cartao', 'avaliacao'])
              && (!('tipoArquivo' in d) || d.tipoArquivo in ['imagem', 'video', 'audio', 'cartao', 'avaliacao'])
              // pedido de avaliação: aponta para um atendimento criado por quem envia
              && (!('avaliacao' in d)
                  || (d.avaliacao is map && d.get('tipoArquivo', '') == 'avaliacao'
                      && d.avaliacao.keys().hasOnly(['pedidoId', 'titulo', 'tipo', 'negocioNome'])
                      && get(/databases/$(database)/documents/pedidos/$(d.avaliacao.pedidoId)).data.vendedorId == request.auth.uid
                      && valorTextoAte(d.avaliacao.get('titulo', ''), 120)
                      && valorTextoAte(d.avaliacao.get('negocioNome', ''), 80)))
              // anexo: só do próprio site (dentro da mensagem ou Firebase Storage)
              && (!('arquivoUrl' in d)
                  || (d.arquivoUrl is string
                      && d.arquivoUrl.matches('^(data:(image|video|audio)/[^,]{1,80},|https://firebasestorage[.]googleapis[.]com/).*')))
              && (!('nomeArquivo' in d) || valorTextoAte(d.nomeArquivo, 120))
              && (!('cartao' in d) || (d.cartao is map && d.get('tipoArquivo', '') == 'cartao'))
              && (!('resposta' in d) || respostaOk(d.resposta));
        }

        // Versão anterior guardada ao editar: o texto e a hora que ele tinha (envio ou última edição).
        function versaoOk(v) {
          return v is map && v.keys().hasOnly(['texto', 'em'])
              && v.texto == resource.data.get('texto', '')
              && v.em == resource.data.get('editadaEm', resource.data.criadoEm);
        }

        allow read: if logado() && participante();
        // A moderação abre só a mensagem denunciada (uma por vez, sem listar a conversa).
        allow get: if podeModerar();

        allow create: if logado()
                      && emailOk()
                      && participante()
                      && semBloqueio()
                      && outroAtivo()
                      && ritmoOk('msg')
                      && request.resource.data.remetenteId == request.auth.uid
                      && request.resource.data.criadoEm == request.time
                      && textoAte('texto', 4000)
                      && mensagemOk();

        // Editar (só quem enviou): o texto novo e, na lista "edicoes", a versão anterior (quem
        // conversa pode ver o original). Cartões de pedido/avaliação e mensagens apagadas não se editam.
        // Editar e apagar só até 10 minutos depois de enviar.
        function noPrazoDeMexer() { return request.time < resource.data.criadoEm + duration.value(10, 'm'); }

        allow update: if logado()
                      && request.auth.uid == resource.data.remetenteId
                      && noPrazoDeMexer()
                      && !('apagada' in resource.data)
                      && resource.data.get('tipoArquivo', '') in ['', 'imagem', 'video', 'audio']
                      && mudou().hasOnly(['texto', 'editadaEm', 'edicoes'])
                      && request.resource.data.texto is string && request.resource.data.texto.size() <= 4000
                      && (request.resource.data.texto.size() > 0 || 'arquivoUrl' in resource.data)
                      && request.resource.data.editadaEm == request.time
                      && request.resource.data.edicoes is list
                      && request.resource.data.edicoes.size() == resource.data.get('edicoes', []).size() + 1
                      && request.resource.data.edicoes.size() <= 20
                      // as versões antigas ficam como estavam; a nova é o texto de antes, com a hora dele
                      && request.resource.data.edicoes == resource.data.get('edicoes', []).concat([request.resource.data.edicoes[resource.data.get('edicoes', []).size()]])
                      && versaoOk(request.resource.data.edicoes[resource.data.get('edicoes', []).size()]);

        // Apagar para todos (só quem enviou): no lugar fica "apagou esta mensagem"; o conteúdo vai,
        // na mesma gravação, para apagadas/{id}, que os dois ainda podem abrir por 24 horas.
        allow update: if logado()
                      && request.auth.uid == resource.data.remetenteId
                      && noPrazoDeMexer()
                      && !('apagada' in resource.data)
                      && request.resource.data.keys().hasOnly(['remetenteId', 'criadoEm', 'apagada'])
                      && request.resource.data.remetenteId == resource.data.remetenteId
                      && request.resource.data.criadoEm == resource.data.criadoEm
                      && request.resource.data.apagada is map
                      && request.resource.data.apagada.keys().hasOnly(['por', 'em'])
                      && request.resource.data.apagada.por == request.auth.uid
                      && request.resource.data.apagada.em == request.time
                      && existsAfter(/databases/$(database)/documents/conversas/$(conversaId)/apagadas/$(mensagemId));

        allow delete: if logado()
                      && request.auth.uid == resource.data.remetenteId
                      && noPrazoDeMexer();
      }

      // ===================================================
      // CONTEÚDO DE MENSAGEM APAGADA PARA TODOS: conversas/{c}/apagadas/{idDaMensagem}
      // Os dois da conversa abrem por 24 horas ("Ver mensagem apagada", e dá para denunciar);
      // depois as regras não deixam mais ler e qualquer um dos dois apaga de vez.
      // A moderação abre enquanto existir (denúncias).
      // ===================================================
      match /apagadas/{mensagemId} {
        function daConversa() {
          return request.auth.uid in get(/databases/$(database)/documents/conversas/$(conversaId)).data.participantes;
        }
        function original() {
          return get(/databases/$(database)/documents/conversas/$(conversaId)/mensagens/$(mensagemId)).data;
        }
        function noPrazo() { return request.time < resource.data.apagadaEm + duration.value(24, 'h'); }
        allow get: if logado() && daConversa() && noPrazo();
        allow get: if podeModerar();
        allow create: if logado()
                      && request.resource.data.keys().hasOnly(['remetenteId', 'conteudo', 'apagadaEm'])
                      && request.resource.data.remetenteId == request.auth.uid
                      && original().remetenteId == request.auth.uid
                      && getAfter(/databases/$(database)/documents/conversas/$(conversaId)/mensagens/$(mensagemId)).data.apagada.por == request.auth.uid
                      && request.resource.data.apagadaEm == request.time
                      && request.resource.data.conteudo is map
                      && request.resource.data.conteudo.keys().hasOnly(['texto', 'tipoArquivo', 'arquivoUrl', 'nomeArquivo', 'cartao', 'avaliacao', 'resposta', 'edicoes'])
                      // o conteúdo guardado é o da mensagem de verdade (não dá para inventar outro)
                      && request.resource.data.conteudo.get('texto', '') == original().get('texto', '')
                      && request.resource.data.conteudo.get('tipoArquivo', '') == original().get('tipoArquivo', '')
                      && request.resource.data.conteudo.get('arquivoUrl', '') == original().get('arquivoUrl', '');
        // Quem recebeu e denunciou (dentro das 24 horas) marca "denunciada": aí só a moderação apaga.
        allow update: if logado() && daConversa() && noPrazo()
                      && request.auth.uid != resource.data.remetenteId
                      && mudou().hasOnly(['denunciada'])
                      && request.resource.data.denunciada == true;
        // Denunciada (pela marca ou porque existe a denúncia, mesmo feita antes de apagar): fica para a moderação.
        function temDenuncia() {
          let p = get(/databases/$(database)/documents/conversas/$(conversaId)).data.participantes;
          let quemRecebeu = p[0] == resource.data.remetenteId ? p[1] : p[0];
          return exists(/databases/$(database)/documents/denuncias/$('mensagem_' + mensagemId + '_' + quemRecebeu));
        }
        allow delete: if logado() && daConversa() && !noPrazo() && resource.data.get('denunciada', false) != true && !temDenuncia();
        allow delete: if podeModerar();
      }
    }
