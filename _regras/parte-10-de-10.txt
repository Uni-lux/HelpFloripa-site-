      // Quem pode comentar (escolha do autor): o autor sempre pode.
      function segue(a, b) { return exists(/databases/$(database)/documents/relacoes/$(a + '_' + b)); }
      function podeComentar(p) {
        let q = p.get('quemComenta', 'todos');
        return p.autorId == request.auth.uid
            || q == 'todos'
            || (q == 'seguidores' && segue(request.auth.uid, p.autorId))
            || (q == 'mutuos' && segue(request.auth.uid, p.autorId) && segue(p.autorId, request.auth.uid));
      }

      function respostaOk(d) {
        return (!('respostaA' in d) && !('respostaAutorId' in d))
            || (d.respostaA is string
                && exists(/databases/$(database)/documents/comentarios/$(d.respostaA))
                && get(/databases/$(database)/documents/comentarios/$(d.respostaA)).data.postId == d.postId
                && d.respostaAutorId == get(/databases/$(database)/documents/comentarios/$(d.respostaA)).data.autorId);
      }

      allow create: if logado()
                    && emailOk()
                    && request.resource.data.keys().hasOnly(['postId', 'postAutorId', 'autorId', 'nome', 'foto', 'texto', 'respostaA', 'respostaAutorId', 'criadoEm'])
                    && request.resource.data.autorId == request.auth.uid
                    && textoComentarioOk(request.resource.data)
                    && valorTextoAte(request.resource.data.get('nome', ''), 80)
                    && fotoOk(request.resource.data.get('foto', ''))
                    && exists(/databases/$(database)/documents/diario/$(request.resource.data.postId))
                    && get(/databases/$(database)/documents/diario/$(request.resource.data.postId)).data.autorId == request.resource.data.postAutorId
                    && get(/databases/$(database)/documents/diario/$(request.resource.data.postId)).data.get('comentariosOcultos', false) != true
                    && podeComentar(get(/databases/$(database)/documents/diario/$(request.resource.data.postId)).data)
                    && respostaOk(request.resource.data)
                    && !bloqueioEntre(request.auth.uid, request.resource.data.postAutorId)
                    && ritmoOk('com')
                    && request.resource.data.criadoEm == request.time;

      allow update: if logado()
                    && emailOk()
                    && (
                      // quem comentou corrige o texto
                      (resource.data.autorId == request.auth.uid
                        && mudou().hasOnly(['texto', 'editadoEm'])
                        && textoComentarioOk(request.resource.data)
                        && request.resource.data.editadoEm == request.time)
                      ||
                      // o dono da publicação oculta ou mostra o comentário
                      (resource.data.postAutorId == request.auth.uid
                        && mudou().hasOnly(['oculto'])
                        && request.resource.data.oculto is bool)
                    );

      allow delete: if logado()
                    && (resource.data.autorId == request.auth.uid || resource.data.postAutorId == request.auth.uid);
      allow delete: if podeModerar();
    }

    // =====================================================
    // CURTIDAS EM COMENTÁRIOS: curtidas_comentarios/{comentarioId}_{uid}
    // =====================================================
    match /curtidas_comentarios/{id} {

      allow read: if logado();

      // Reação (emoji); sem o campo = curtida antiga (conta como ❤️).
      function reacaoOk(d) { return !('reacao' in d) || d.reacao in ['❤️', '👍', '😂', '😮', '😢', '👏']; }

      allow create: if logado()
                    && emailOk()
                    && request.resource.data.keys().hasOnly(['comentarioId', 'uid', 'criadoEm', 'reacao', 'comentarioAutorId'])
                    // quem escreveu o comentário (para avisar da reação)
                    && (!('comentarioAutorId' in request.resource.data)
                        || request.resource.data.comentarioAutorId == get(/databases/$(database)/documents/comentarios/$(request.resource.data.comentarioId)).data.autorId)
                    && request.resource.data.uid == request.auth.uid
                    && id == request.resource.data.comentarioId + '_' + request.auth.uid
                    && exists(/databases/$(database)/documents/comentarios/$(request.resource.data.comentarioId))
                    && reacaoOk(request.resource.data)
                    && request.resource.data.criadoEm == request.time;

      // Trocar a reação (só o emoji).
      allow update: if logado()
                    && resource.data.uid == request.auth.uid
                    && mudou().hasOnly(['reacao'])
                    && reacaoOk(request.resource.data);
      allow delete: if logado() && resource.data.uid == request.auth.uid;
    }


    // =====================================================
    // DIÁRIO / PUBLICAÇÕES
    // =====================================================
    match /diario/{postId} {

      allow read: if logado();

      function publicacaoOk() {
        let d = request.resource.data;
        return textoAte('texto', 1000)
            && textoAte('nome', 80)
            && campoFotoOk('fotoPerfil')
            && (!('mediaTipo' in d) || d.mediaTipo in ['', 'image', 'video'])
            && (!('mediaUrl' in d) || d.mediaUrl == ''
                || (d.mediaUrl is string
                    && d.mediaUrl.matches('^(data:image/(jpeg|png|webp);base64,|https://firebasestorage[.]googleapis[.]com/).*')))
            // Várias fotos (2 a 4): a primeira é a mesma de mediaUrl; cada uma do próprio site.
            && (!('midias' in d)
                || (d.midias is list && d.midias.size() >= 2 && d.midias.size() <= 4
                    && d.get('mediaTipo', '') == 'image' && d.midias[0] == d.get('mediaUrl', '')
                    && fotoPubOk(d.midias[1])
                    && (d.midias.size() < 3 || fotoPubOk(d.midias[2]))
                    && (d.midias.size() < 4 || fotoPubOk(d.midias[3]))));
      }
      // Hashtags (até 10, minúsculas, sem #) e pessoas marcadas (até 10 uids): uma expressão cada (join).
      function tagsOk(d) {
        return !('tags' in d) || (d.tags is list && d.tags.size() <= 10
            && (d.tags.size() == 0 || d.tags.join(' ').matches('^[\\p{Ll}\\p{N}_]{1,30}( [\\p{Ll}\\p{N}_]{1,30}){0,9}$')));
      }
      function marcadosOk(d) {
        return !('marcados' in d) || (d.marcados is list && d.marcados.size() <= 10
            && (d.marcados.size() == 0 || d.marcados.join(' ').matches('^[A-Za-z0-9]{1,64}( [A-Za-z0-9]{1,64}){0,9}$')));
      }
      function fotoPubOk(u) {
        return u is string && u.matches('^(data:image/(jpeg|png|webp);base64,|https://firebasestorage[.]googleapis[.]com/).*');
      }

      allow create: if logado()
                    && emailOk()
                    && request.resource.data.autorId == request.auth.uid
                    && request.resource.data.keys().hasOnly(['autorId', 'nome', 'fotoPerfil', 'texto', 'mediaUrl', 'mediaTipo', 'midias', 'tags', 'marcados', 'agendadoPara', 'enquete', 'criadoEm'])
                    && (!('enquete' in request.resource.data) || enqueteOk(request.resource.data.enquete))
                    && ritmoOk('pub')
                    && publicacaoOk()
                    && tagsOk(request.resource.data) && marcadosOk(request.resource.data)
                    // agendada: aparece para os outros a partir de agendadoPara (até 60 dias)
                    && (!('agendadoPara' in request.resource.data)
                        || (request.resource.data.agendadoPara is timestamp
                            && request.resource.data.agendadoPara > request.time
                            && request.resource.data.agendadoPara < request.time + duration.value(60, 'd')));

      allow update: if logado()
                    && emailOk()
                    && resource.data.autorId == request.auth.uid
                    && request.resource.data.autorId == resource.data.autorId
                    && gravou().hasOnly(['texto', 'nome', 'fotoPerfil', 'comentariosOcultos', 'ocultoAte', 'editadoEm', 'quemComenta', 'comentarioFixo', 'tags', 'marcados', 'agendadoPara'])
                    && tagsOk(request.resource.data) && marcadosOk(request.resource.data)
                    // agendada: publicar agora (agendadoPara = agora) ou mudar a data (até 60 dias)
                    && (!mudou().hasAny(['agendadoPara'])
                        || (request.resource.data.agendadoPara is timestamp
                            && request.resource.data.agendadoPara >= request.time
                            && request.resource.data.agendadoPara < request.time + duration.value(60, 'd')))
                    // quem pode comentar: todos, seguidores, seguidores que o autor segue de volta, ninguém
                    && (!('quemComenta' in request.resource.data) || request.resource.data.quemComenta in ['todos', 'seguidores', 'mutuos', 'ninguem'])
                    // comentário fixado no topo (id) ou nulo
                    && (!('comentarioFixo' in request.resource.data) || request.resource.data.comentarioFixo == null
                        || (request.resource.data.comentarioFixo is string && request.resource.data.comentarioFixo.size() <= 40))
                    && dataOuNulo('ocultoAte')
                    && travaModeracao()
                    // texto editado: marca a data da edição
                    && (!mudou().hasAny(['texto']) || request.resource.data.get('editadoEm', null) == request.time)
                    && (!mudou().hasAny(['editadoEm']) || request.resource.data.editadoEm == request.time)
                    && (!('comentariosOcultos' in request.resource.data) || request.resource.data.comentariosOcultos is bool)
                    && publicacaoOk();

      allow delete: if logado()
                    && resource.data.autorId == request.auth.uid;

      // Alcance: cada conta que viu a publicação (uma vez). Só o autor lista/conta.
      match /vistas/{uid} {
        allow create: if logado() && uid == request.auth.uid
                      && request.resource.data.keys().hasOnly(['criadoEm', 'origem'])
                      && request.resource.data.criadoEm == request.time
                      && (!('origem' in request.resource.data) || request.resource.data.origem in ['principais', 'seguindo', 'amigos', 'perfil', 'link', 'tag', 'salvos']);
        allow read: if logado() && get(/databases/$(database)/documents/diario/$(postId)).data.autorId == request.auth.uid;
        allow update, delete: if false;
      }

      // A equipe esconde ou remove publicações que violam os Termos.
      allow update: if podeModerar()
                    && moderacaoOk();
      allow delete: if podeModerar();
    }
  }
