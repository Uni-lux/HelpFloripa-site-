    match /avisos/{id} {
      allow read: if logado() && resource.data.uid == request.auth.uid;
      allow read: if admin();
      allow create: if admin()
                    && request.resource.data.keys().hasOnly(['uid', 'titulo', 'texto', 'tipo', 'por', 'em', 'lidoEm'])
                    && request.resource.data.uid is string
                    && valorTextoAte(request.resource.data.get('titulo', ''), 120)
                    && request.resource.data.texto is string && request.resource.data.texto.size() <= 2000
                    && request.resource.data.tipo in ['info', 'alerta', 'grave']
                    && request.resource.data.por == request.auth.uid
                    && request.resource.data.em == request.time
                    && request.resource.data.lidoEm == null;
      // quem recebeu marca como lido
      allow update: if logado()
                    && resource.data.uid == request.auth.uid
                    && mudou().hasOnly(['lidoEm'])
                    && request.resource.data.lidoEm == request.time;
      allow delete: if false;
    }

    match /comunicados/{id} {
      allow read: if logado();
      allow create, update: if podeModerar()
                    && request.resource.data.keys().hasOnly(['titulo', 'texto', 'tipo', 'ativo', 'link', 'por', 'em', 'ate'])
                    && valorTextoAte(request.resource.data.get('titulo', ''), 120)
                    && valorTextoAte(request.resource.data.get('texto', ''), 600)
                    && request.resource.data.tipo in ['info', 'novidade', 'alerta']
                    && request.resource.data.ativo is bool
                    && request.resource.data.por == request.auth.uid;
      allow delete: if podeModerar();
    }

    match /suporte/{id} {
      function camposChamado() {
        return ['uid', 'nome', 'email', 'assunto', 'categoria', 'status', 'criadoEm', 'atualizadoEm', 'ultimaDe', 'atendidoPor', 'responsavel', 'responsavelNome', 'assumidoEm'];
      }
      allow read: if logado() && resource.data.uid == request.auth.uid;
      allow read: if admin();
      // Abrir chamado: não exige e-mail confirmado nem conta livre (quem foi suspenso pode contestar).
      allow create: if logado()
                    && request.resource.data.keys().hasOnly(camposChamado())
                    && request.resource.data.uid == request.auth.uid
                    && request.resource.data.assunto is string && request.resource.data.assunto.size() >= 3 && request.resource.data.assunto.size() <= 120
                    && request.resource.data.categoria in ['conta', 'pagamento', 'golpe', 'denuncia', 'contestacao', 'negocio', 'sugestao', 'outro']
                    && valorTextoAte(request.resource.data.get('nome', ''), 80)
                    && valorTextoAte(request.resource.data.get('email', ''), 120)
                    && request.resource.data.status == 'aberto'
                    && request.resource.data.criadoEm == request.time
                    && request.resource.data.atualizadoEm == request.time
                    && request.resource.data.ultimaDe == 'usuario'
                    && !request.resource.data.keys().hasAny(['atendidoPor', 'responsavel', 'responsavelNome', 'assumidoEm']);
      // A equipe abre uma conversa de suporte com qualquer pessoa ("Suporte Help Floripa"):
      // o chamado já nasce respondido, com a primeira mensagem da equipe gravada junto.
      allow create: if admin()
                    && request.resource.data.keys().hasOnly(camposChamado().concat(['abertoPor']))
                    && request.resource.data.uid is string
                    && exists(/databases/$(database)/documents/usuarios/$(request.resource.data.uid))
                    && request.resource.data.abertoPor == request.auth.uid
                    && request.resource.data.assunto is string && request.resource.data.assunto.size() >= 3 && request.resource.data.assunto.size() <= 120
                    && request.resource.data.categoria in ['conta', 'pagamento', 'golpe', 'denuncia', 'contestacao', 'negocio', 'sugestao', 'outro']
                    && valorTextoAte(request.resource.data.get('nome', ''), 80)
                    && valorTextoAte(request.resource.data.get('email', ''), 120)
                    && request.resource.data.status == 'respondido'
                    && request.resource.data.criadoEm == request.time
                    && request.resource.data.atualizadoEm == request.time
                    && request.resource.data.ultimaDe == 'equipe'
                    // quem abre pode já ficar como responsável pelo chamado
                    && (!('responsavel' in request.resource.data)
                        || (request.resource.data.responsavel == request.auth.uid
                            && request.resource.data.get('assumidoEm', null) == request.time
                            && valorTextoAte(request.resource.data.get('responsavelNome', ''), 80)))
                    && existsAfter(/databases/$(database)/documents/suporte/$(id)/mensagens/inicio);
      allow update: if logado()
                    && ((resource.data.uid == request.auth.uid
                         && mudou().hasOnly(['status', 'atualizadoEm', 'ultimaDe'])
                         && request.resource.data.status in ['aberto', 'fechado']
                         && request.resource.data.ultimaDe == 'usuario')
                        || (admin()
                         && mudou().hasOnly(['status', 'atualizadoEm', 'ultimaDe', 'atendidoPor', 'responsavel', 'responsavelNome', 'assumidoEm'])
                         && request.resource.data.status in ['aberto', 'respondido', 'fechado']
                         && responsavelOk()))
                    && request.resource.data.atualizadoEm == request.time;
      // A equipe só assume ou passa o chamado (sem mexer na data da conversa).
      allow update: if admin()
                    && mudou().hasOnly(['responsavel', 'responsavelNome', 'assumidoEm'])
                    && responsavelOk();
      allow delete: if false;

      match /mensagens/{msgId} {
        function chamado() { return get(/databases/$(database)/documents/suporte/$(id)).data; }
        // getAfter: o chamado e a primeira mensagem são gravados juntos (tudo ou nada).
        function chamadoDepois() { return getAfter(/databases/$(database)/documents/suporte/$(id)).data; }
        allow read: if logado() && (chamado().uid == request.auth.uid || admin());
        allow create: if logado()
                      && (chamadoDepois().uid == request.auth.uid || admin())
                      && request.resource.data.keys().hasOnly(['autorId', 'equipe', 'texto', 'em'])
                      && request.resource.data.autorId == request.auth.uid
                      // "equipe" (resposta oficial) só quem é da equipe; no próprio chamado, todos escrevem como usuário
                      && (request.resource.data.equipe == false || (request.resource.data.equipe == true && admin()))
                      && request.resource.data.texto is string && request.resource.data.texto.size() >= 1 && request.resource.data.texto.size() <= 2000
                      && request.resource.data.em == request.time;
        allow update, delete: if false;
      }
    }

    // Pedido do selo de verificado: a pessoa e a moderação leem. A foto do documento é apagada na decisão.
    match /verificacoes/{uid} {
      function pedidoOk(d) {
        return d.keys().hasOnly(['uid', 'categoria', 'nomeLegal', 'conhecidoComo', 'documentoTipo', 'documento', 'links', 'motivo', 'status', 'criadoEm', 'atualizadoEm'])
            && d.uid == uid
            && d.categoria is string && d.categoria.matches('^(pessoa|profissional|negocio|publica|organizacao)$')
            && d.nomeLegal is string && d.nomeLegal.size() >= 5 && d.nomeLegal.size() <= 120
            && valorTextoAte(d.get('conhecidoComo', ''), 80)
            && d.documentoTipo is string && d.documentoTipo.matches('^(rg|cnh|passaporte|rne|cnpj)$')
            && d.documento is string && d.documento.size() <= 700000 && d.documento.matches('^data:image/jpeg;base64,[A-Za-z0-9+/=]+$')
            && (!('links' in d) || (d.links is list && d.links.size() <= 3))
            && valorTextoAte(d.get('motivo', ''), 500)
            && d.status == 'pendente'
            && d.criadoEm == request.time && d.atualizadoEm == request.time;
      }
      allow read: if (logado() && request.auth.uid == uid) || podeModerar();
      // Pedir (ou pedir de novo depois de não aprovado ou cancelado).
      allow create: if logado() && request.auth.uid == uid && pedidoOk(request.resource.data);
      allow update: if logado() && request.auth.uid == uid
                    && resource.data.status in ['recusada', 'cancelada']
                    && pedidoOk(request.resource.data);
      // Cancelar o próprio pedido (a foto do documento sai junto).
      allow update: if logado() && request.auth.uid == uid
                    && resource.data.status in ['pendente', 'em_analise']
                    && request.resource.data.status == 'cancelada'
                    && mudou().hasOnly(['status', 'atualizadoEm', 'documento'])
                    && !('documento' in request.resource.data)
                    && request.resource.data.atualizadoEm == request.time;
      // A moderação analisa e decide; ao aprovar ou recusar, a foto do documento é apagada.
      allow update: if podeModerar()
                    && mudou().hasOnly(['status', 'motivoRecusa', 'analisadaPor', 'analisadaEm', 'documento', 'atualizadoEm'])
                    && request.resource.data.status in ['em_analise', 'aprovada', 'recusada']
                    && request.resource.data.analisadaPor == request.auth.uid
                    && (request.resource.data.status == 'em_analise' || !('documento' in request.resource.data))
                    && valorTextoAte(request.resource.data.get('motivoRecusa', ''), 500);
      allow delete: if (logado() && request.auth.uid == uid) || podeModerar();
    }

    // Reporte de atendimento do suporte (só a moderação e os donos leem; o suporte não).
    //   atendimento_{chamado}_{uid}: quem abriu o chamado reporta como foi atendido (demora, grosseria...).
    //   usuario_{chamado}_{uid}: alguém da equipe reporta a pessoa do chamado (ofensas, ameaça, golpe...).
    match /reportes/{id} {
      function chamadoDe(c) { return get(/databases/$(database)/documents/suporte/$(c)).data; }
      allow get: if logado() && resource.data.autorId == request.auth.uid;
      allow read: if podeModerar();
      allow create: if logado()
                    && request.resource.data.keys().hasOnly(['tipo', 'chamadoId', 'assunto', 'autorId', 'alvoId', 'alvoNome', 'motivo', 'detalhe', 'status', 'criadoEm'])
                    && request.resource.data.autorId == request.auth.uid
                    && request.resource.data.chamadoId is string
                    && id == request.resource.data.tipo + '_' + request.resource.data.chamadoId + '_' + request.auth.uid
                    && request.resource.data.motivo.matches('^(demora|grosseria|nao_resolveu|errado|dados|ofensa|ameaca|golpe|spam|outro)$')
                    && valorTextoAte(request.resource.data.get('detalhe', ''), 1000)
                    && valorTextoAte(request.resource.data.get('assunto', ''), 120)
                    && valorTextoAte(request.resource.data.get('alvoNome', ''), 80)
                    && request.resource.data.status == 'novo'
                    && request.resource.data.criadoEm == request.time
                    && ((request.resource.data.tipo == 'atendimento'
                         && chamadoDe(request.resource.data.chamadoId).uid == request.auth.uid
                         && (!('alvoId' in request.resource.data) || request.resource.data.alvoId is string))
                        || (request.resource.data.tipo == 'usuario'
                         && admin()
                         && request.resource.data.alvoId == chamadoDe(request.resource.data.chamadoId).uid));
      allow update: if podeModerar()
                    && mudou().hasOnly(['status', 'nota', 'analisadaEm', 'analisadaPor', 'responsavel', 'responsavelNome', 'assumidoEm'])
                    && request.resource.data.status in ['novo', 'em_analise', 'resolvido', 'descartado']
                    && (!mudou().hasAny(['status', 'nota', 'analisadaEm', 'analisadaPor']) || request.resource.data.analisadaPor == request.auth.uid)
                    && responsavelOk()
                    && valorTextoAte(request.resource.data.get('nota', ''), 1000);
      allow delete: if false;
    }
