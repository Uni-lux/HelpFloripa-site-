      // Qualquer um dos dois pode cancelar, recusar ou desfazer.
      allow delete: if logado()
                    && request.auth.uid in resource.data.participantes;
    }


    // =====================================================
    // PERFIS DE NEGÓCIO (serviços, delivery, lojinha, imóveis)
    // Documento: negocios/{donoId}_{tipo}
    // =====================================================
    match /negocios/{id} {

      // Campos que o site grava. Qualquer outro campo é recusado.
      function camposNegocio() {
        return ['donoId', 'tipo', 'nome', 'descricao', 'cidade', 'uf', 'cidadeIbge', 'bairro', 'rua', 'numero', 'complemento', 'cep',
                'whatsapp', 'foto', 'fotos', 'itens',
                'categoria', 'categorias', 'categoriaPersonalizada', 'criadoEm', 'atualizadoEm',
                'precoDesde', 'horario', 'atendimento', 'dias', 'regiao', 'experiencia', 'diferenciais', 'pagamentos', 'cnh', 'veiculo',
                'horaAbre', 'horaFecha', 'tempoMin', 'tempoMax', 'taxaEntrega', 'pedidoMinimo', 'opcoesEntrega',
                'entrega', 'tipoAnunciante', 'creci',
                'instagram', 'site', 'capa', 'cnpj', 'promocao', 'semana', 'geo',
                'saude',       // perfil de Saúde: profissão, registro no conselho, especialidades, convênios...
                'busca',       // palavras e começos de palavras para a busca nacional (busca.js), até 100
                'principal',   // sem plano, só o perfil deste tipo fica no ar (o dono escolhe; o site esconde os outros)
                'ocultoAte'];  // escondido enquanto a conta do dono estiver desativada
      }

      // Saúde: um mapa só, sempre com os 15 campos (o site grava todos), conferido com dois regex
      // (gasta pouco do limite de 1.000 expressões: a gravação de um negócio completo já chega perto).
      function saudeOk(s) {
        return s.keys().hasOnly(['profissao', 'profissaoOutra', 'especialidades', 'registro', 'registroUf', 'rqe', 'modalidades',
                                 'convenios', 'particular', 'valorConsulta', 'duracao', 'publico', 'idiomas', 'acessivel', 'formacao'])
            && [s.profissao, s.profissaoOutra, s.registro, s.registroUf, s.rqe, s.formacao, s.modalidades.join(','), s.publico.join(','),
                string(s.particular), string(s.acessivel), string(s.valorConsulta), string(s.duracao)].join('\n')
                 .matches('^(medico|enfermeiro|tecEnfermagem|psicologo|veterinario|dentista|fisioterapeuta|nutricionista|fonoaudiologo|terapeutaOcupacional|farmaceutico|educadorFisico|cuidador|outros)\n[^\n]{0,40}\n[0-9A-Za-z./ -]{0,20}\n([A-Z]{2})?\n[0-9A-Za-z./-]{0,12}\n[^\n]{0,200}\n((consultorio|online|domicilio)(,(consultorio|online|domicilio)){0,2})?\n([a-z]{3,14}(,[a-z]{3,14}){0,5})?\n(true|false)\n(true|false)\n(null|[0-9]{1,5}([.][0-9]{1,2})?)\n(null|[1-9][0-9]{0,2})$')
            && [s.especialidades.join('\n'), s.convenios.join('\n'), s.idiomas.join('\n')].join('\n').matches('^[^\n]{0,40}(\n[^\n]{0,40}){0,29}$');
      }

      // Promoção/cupom com prazo (até 62 dias): título, texto, código opcional (A-Z e 0-9).
      function promocaoOk(p) {
        return p is map
            && p.keys().hasOnly(['titulo', 'texto', 'codigo', 'ate'])
            && p.titulo is string && p.titulo.size() >= 3 && p.titulo.size() <= 60
            && valorTextoAte(p.get('texto', ''), 200)
            && (p.get('codigo', '') == '' || (p.codigo is string && p.codigo.matches('^[A-Z0-9]{3,20}$')))
            && p.ate is timestamp && p.ate < request.time + duration.value(62, 'd');
      }

      // Instagram (só o @, sem link), site (http/https, até 120) e CNPJ (14 números) num regex só;
      // capa (foto do site); horário de cada dia (7 textos "HH:MM-HH:MM" ou "", domingo a sábado) e promoção.
      // Só confere o que mudou: a gravação completa ficava perto do limite de 1.000 expressões.
      function extrasNegocioOk() {
        let d = request.resource.data;
        let m = resource == null ? d.keys() : mudou();
        return (!m.hasAny(['instagram', 'site', 'cnpj'])
                || [d.get('instagram', ''), d.get('site', ''), d.get('cnpj', '')].join('\n')
                     .matches('^[A-Za-z0-9._]{0,30}\n(https?://[^ \n]{3,112})?\n([0-9]{14})?$'))
            && (!m.hasAny(['capa']) || fotoOk(d.get('capa', null)))
            && (!m.hasAny(['semana']) || d.get('semana', null) == null
                || (d.semana is list && d.semana.size() == 7
                    && d.semana.join(';').matches('^(([01][0-9]|2[0-3]):[0-5][0-9]-([01][0-9]|2[0-3]):[0-5][0-9])?(;(([01][0-9]|2[0-3]):[0-5][0-9]-([01][0-9]|2[0-3]):[0-5][0-9])?){6}$')))
            && (!m.hasAny(['promocao']) || d.get('promocao', null) == null || promocaoOk(d.promocao))
            && (!m.hasAny(['geo']) || geoOk(d.get('geo', null)))
            && (!m.hasAny(['saude']) || saudeOk(d.get('saude', null)))
            && (!m.hasAny(['principal']) || d.get('principal', null) == null
                || d.principal in ['servicos', 'delivery', 'lojinha', 'imoveis', 'saude', 'automotivo']);
      }

      function negocioOk() {
        return textoAte('nome', 80)
            && textoAte('descricao', 600)
            && textoAte('cidade', 60)
            && localOk()
            && enderecoOk()
            && textoAte('whatsapp', 20)
            && textoAte('regiao', 100)
            && textoAte('horario', 60)
            && textoAte('categoriaPersonalizada', 30)
            && listaAte('itens', 12)
            && listaAte('fotos', 5)
            && listaAte('busca', 100)
            && dataOuNulo('ocultoAte')
            && ((resource != null && !mudou().hasAny(['foto'])) || campoFotoOk('foto'))
            && extrasNegocioOk();
      }

      allow read: if logado();

      allow create: if logado()
                    && emailOk()
                    && request.resource.data.donoId == request.auth.uid
                    && request.resource.data.tipo in ['servicos', 'delivery', 'lojinha', 'imoveis', 'saude', 'automotivo']
                    && id == request.auth.uid + '_' + request.resource.data.tipo
                    && request.resource.data.keys().hasOnly(camposNegocio())
                    && naoTocaCamposProtegidosNoCreate()
                    && !banidoDoPrograma(request.auth.uid, request.resource.data.tipo)
                    && contaVerificada()
                    && (request.resource.data.tipo != 'saude' || request.resource.data.get('saude', null) is map)
                    && negocioOk();

      allow update: if logado()
                    && emailOk()
                    && resource.data.donoId == request.auth.uid
                    && request.resource.data.donoId == resource.data.donoId
                    && request.resource.data.tipo == resource.data.tipo
                    && gravou().hasOnly(camposNegocio())
                    && naoTocaCamposProtegidosNoUpdate()
                    && travaModeracao()
                    && negocioOk();

      allow delete: if logado()
                    && resource.data.donoId == request.auth.uid;

      // A equipe esconde (ocultoAte) o conteúdo de contas suspensas/banidas.
      allow update: if podeModerar()
                    && moderacaoOk();
    }


    // =====================================================
    // ANÚNCIOS (vários imóveis por pessoa)
    // Documento: anuncios/{id}
    // =====================================================
    match /anuncios/{id} {

      function camposAnuncio() {
        return ['donoId', 'tipo', 'ativo', 'titulo', 'finalidade', 'categoria', 'preco', 'condominio', 'iptu',
                'quartos', 'suites', 'banheiros', 'vagas', 'area', 'andar', 'bairro', 'cidade', 'mobiliado',
                'uf', 'cidadeIbge', 'rua', 'numero', 'complemento', 'cep',
                'disponivel', 'hospedes', 'caracteristicas', 'condicoes', 'descricao', 'fotos', 'criadoEm', 'atualizadoEm',
                'categoriaPersonalizada',  // tipo escrito pela pessoa quando escolhe "Outro" (Chácara, Galpão...)
                'geo',                     // ponto do endereço (Perto de você)
                'busca',                   // palavras para a busca nacional (busca.js), até 100
                'ocultoAte'];
      }

      function anuncioOk() {
        return textoAte('titulo', 100)
            && textoAte('descricao', 1000)
            && textoAte('cidade', 60)
            && localOk()
            && enderecoOk()
            && textoAte('andar', 12)
            && textoAte('disponivel', 20)
            && textoAte('categoriaPersonalizada', 30)
            && geoOk(request.resource.data.get('geo', null))
            && listaAte('fotos', 5)
            && listaAte('busca', 100)
            && listaAte('caracteristicas', 20)
            && listaAte('condicoes', 12)
            && dataOuNulo('ocultoAte');
      }

      allow read: if logado();

      allow create: if logado()
                    && emailOk()
                    && request.resource.data.donoId == request.auth.uid
                    && request.resource.data.tipo in ['imovel']
                    && request.resource.data.keys().hasOnly(camposAnuncio())
                    && naoTocaCamposProtegidosNoCreate()
                    && contaVerificada()
                    && anuncioOk();

      allow update: if logado()
                    && emailOk()
                    && resource.data.donoId == request.auth.uid
                    && request.resource.data.donoId == resource.data.donoId
                    && request.resource.data.tipo == resource.data.tipo
                    && gravou().hasOnly(camposAnuncio())
                    && naoTocaCamposProtegidosNoUpdate()
                    && travaModeracao()
                    && anuncioOk();

      allow delete: if logado()
                    && resource.data.donoId == request.auth.uid;

      // A equipe esconde (ocultoAte) o conteúdo de contas suspensas/banidas.
      allow update: if podeModerar()
                    && moderacaoOk();
    }


    // =====================================================
    // COMUNIDADE: doações, trocas, pedidos de ajuda, vagas em quarto compartilhado e achados e perdidos.
    // Documento: comunidade/{id}. É PROIBIDO vender ou cobrar: não aceita preço, "R$" nem "vendo/venda" no texto
    // (o quarto compartilhado só mostra a divisão das despesas, no campo próprio). Quem descumpre é banido do
    // programa (programa_banidos/{uid}_comunidade) e pode ter a conta suspensa.
    // =====================================================
    match /comunidade/{id} {
      function camposComunidade() {
        return ['donoId', 'categoria', 'tipoItem', 'titulo', 'descricao', 'fotos', 'estado', 'entrega', 'status',
                'bairro', 'cidade', 'uf', 'cidadeIbge', 'geo', 'quarto', 'achado', 'aceiteRegras', 'busca',
                'criadoEm', 'atualizadoEm', 'ocultoAte', 'ocultoPelaEquipe'];
      }
      // Vaga em quarto/moradia compartilhada: só a divisão das despesas entre moradores.
      function quartoOk(q) {
        return q is map
            && q.keys().hasOnly(['vaga', 'rateio', 'contasInclusas', 'mobiliado', 'vagas', 'para', 'disponivelEm', 'regras'])
            && q.get('vaga', 'quarto').matches('^(individual|compartilhado|republica)$')
            && (q.get('rateio', null) == null || (q.rateio is number && q.rateio >= 0 && q.rateio <= 5000))
            && q.get('contasInclusas', false) is bool && q.get('mobiliado', false) is bool
            && (q.get('vagas', null) == null || (q.vagas is int && q.vagas >= 1 && q.vagas <= 10))
            && [q.get('para', 'todos'), q.get('disponivelEm', ''), q.get('regras', '')].join('\n')
                 .matches('^(todos|mulheres|homens|lgbt)\n([0-9]{4}-[0-9]{2}-[0-9]{2})?\n[^\n]{0,300}$');
      }
      // Achados e perdidos: perdi / encontrei, onde e quando.
      function achadoOk(a) {
        return a is map && a.keys().hasOnly(['tipo', 'data', 'local'])
            && [a.get('tipo', ''), a.get('data', ''), a.get('local', '')].join('\n')
                 .matches('^(perdido|encontrado)\n([0-9]{4}-[0-9]{2}-[0-9]{2})?\n[^\n]{0,80}$');
      }
