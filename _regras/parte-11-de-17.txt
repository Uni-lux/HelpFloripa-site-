    // Plano ativo da conta: público (selo e destaque); só os donos do site mudam.
    // Primeiro acesso: a própria pessoa começa 30 dias grátis no plano que escolher (só se a conta nunca teve plano).
    match /planos/{uid} {
      function tiposPerfisOk(l) {
        return l is list && l.size() >= 1 && l.size() <= 5
            && l.join(',').matches('^(servicos|delivery|lojinha|imoveis|saude)(,(servicos|delivery|lojinha|imoveis|saude)){0,4}$');
      }
      allow read: if logado();
      // Os 30 dias grátis agora vêm junto com a primeira assinatura paga (o dono do site soma ao confirmar o Pix):
      // ninguém grava o próprio plano.
      allow create, update: if donoDoSite()
                    && request.resource.data.keys().hasOnly(['plano', 'ate', 'ativadoEm', 'ativadoPor', 'pedidoId', 'negociacaoId', 'meses', 'teste', 'dias', 'tipos', 'semSelo'])
                    && request.resource.data.plano in ['plus', 'pro', 'premium', 'gratis']
                    // Plano combinado pode liberar só alguns perfis (tipos) e esconder o selo (semSelo).
                    && (!('tipos' in request.resource.data) || tiposPerfisOk(request.resource.data.tipos))
                    && request.resource.data.get('semSelo', false) is bool
                    && (!('dias' in request.resource.data) || (request.resource.data.dias is int && request.resource.data.dias >= 1 && request.resource.data.dias <= 800))
                    && request.resource.data.ate is timestamp
                    && request.resource.data.ativadoPor == request.auth.uid
                    && request.resource.data.ativadoEm == request.time;
      allow delete: if donoDoSite();
    }
    // Condição especial (baixa renda) ou parceria pelos planos: um pedido por pessoa (id = uid). A pessoa envia,
    // cancela enquanto está aberto e pede de novo depois de decidido; só os donos do site leem e negociam.
    // Condição especial: a pessoa diz qual documento prova a baixa renda (ou que manda depois), declara que é verdade e
    // pode anexar a foto/PDF; o anexo sai do documento quando os donos decidem (ou quando ela cancela).
    match /planos_negociacoes/{uid} {
      function comprovanteOk(x) {
        return x is map
            && x.keys().hasOnly(['tipo', 'url', 'nome'])
            && x.tipo in ['imagem', 'arquivo']
            && valorTextoAte(x.get('nome', ''), 120)
            && x.url is string && x.url.size() <= 750000
            && x.url.matches('^(data:(image/jpeg|application/pdf);base64,|https://firebasestorage[.]googleapis[.]com/).*');
      }
      function respNegOk(d) {
        return (d.get('responsavel', null) == null || d.responsavel is string)
            && valorTextoAte(d.get('responsavelNome', ''), 80)
            && (!('assumidoEm' in d) || d.assumidoEm == request.time || d.assumidoEm == resource.data.get('assumidoEm', null));
      }
      function pedidoNegOk() {
        let d = request.resource.data;
        return d.keys().hasOnly(['uid', 'nome', 'email', 'tipo', 'plano', 'valorProposto', 'contato', 'mensagem', 'status', 'criadoEm', 'comprovanteTipo', 'comprovante', 'declaracao', 'perfisPedidos', 'mesesPedidos'])
          && (!('perfisPedidos' in d) || (d.perfisPedidos is int && d.perfisPedidos >= 1 && d.perfisPedidos <= 5))
          && (!('mesesPedidos' in d) || (d.mesesPedidos is int && d.mesesPedidos >= 1 && d.mesesPedidos <= 12))
          // Baixa renda: declaração + qual documento + o documento anexado (não dá mais para "mandar depois").
          && (d.tipo != 'apoio' || (d.get('declaracao', false) == true
              && d.get('comprovanteTipo', '') is string
              && d.get('comprovanteTipo', '').matches('^(cadunico|bolsa|bpc|cnis|renda|desemprego|tarifa|mei|isento)$')
              && 'comprovante' in d))
          && (!('comprovante' in d) || (d.tipo == 'apoio' && comprovanteOk(d.comprovante)))
          && d.uid == request.auth.uid && uid == request.auth.uid
          && d.tipo in ['apoio', 'parceria']
          && d.plano in ['plus', 'pro', 'premium']
          && d.valorProposto is number && d.valorProposto >= 0 && d.valorProposto <= 1000
          && d.mensagem is string && d.mensagem.size() >= 20 && d.mensagem.size() <= 1500
          && valorTextoAte(d.get('contato', ''), 40)
          && valorTextoAte(d.get('nome', ''), 80)
          && valorTextoAte(d.get('email', ''), 120)
          && d.status == 'novo'
          && d.criadoEm == request.time;
      }
      allow read: if (logado() && uid == request.auth.uid) || donoDoSite();
      allow create: if logado() && emailOk() && pedidoNegOk();
      // Pedir de novo: só depois que o anterior foi decidido ou cancelado (substitui o documento inteiro).
      allow update: if logado() && emailOk() && uid == request.auth.uid
                    && resource.data.status in ['aprovado', 'recusado', 'cancelado']
                    && pedidoNegOk();
      allow update: if logado() && uid == request.auth.uid
                    && resource.data.status in ['novo', 'em_conversa', 'aguardando_pagamento']
                    && mudou().hasOnly(['status', 'comprovante'])
                    && !('comprovante' in request.resource.data)
                    && request.resource.data.status == 'cancelado';
      // Responsável pelo pedido (só os donos do site): assumir, passar ou liberar.
      allow update: if donoDoSite()
                    && mudou().hasOnly(['responsavel', 'responsavelNome', 'assumidoEm'])
                    && respNegOk(request.resource.data);
      allow update: if donoDoSite()
                    && resource.data.status in ['novo', 'em_conversa']
                    && mudou().hasOnly(['status', 'resposta', 'decididoPor', 'decididoEm', 'planoConcedido', 'meses', 'valorAcordado', 'comprovante', 'dias', 'tipos', 'semSelo', 'responsavel', 'responsavelNome'])
                    && respNegOk(request.resource.data)
                    && (!('tipos' in request.resource.data) || (request.resource.data.tipos is list && request.resource.data.tipos.size() <= 5 && request.resource.data.tipos.join(',').matches('^(servicos|delivery|lojinha|imoveis|saude)(,(servicos|delivery|lojinha|imoveis|saude)){0,4}$')))
                    && request.resource.data.get('semSelo', false) is bool
                    && (!('dias' in request.resource.data) || (request.resource.data.dias is int && request.resource.data.dias >= 1 && request.resource.data.dias <= 800))
                    && request.resource.data.status in ['em_conversa', 'aprovado', 'aguardando_pagamento', 'recusado']
                    && (request.resource.data.status == 'em_conversa' ? !mudou().hasAny(['comprovante']) : !('comprovante' in request.resource.data))
                    && request.resource.data.decididoPor == request.auth.uid
                    && request.resource.data.decididoEm == request.time
                    && valorTextoAte(request.resource.data.get('resposta', ''), 600)
                    // Aprovado com valor: fica esperando o pagamento (o plano só liga quando o Pix/cartão é confirmado).
                    && (!(request.resource.data.status in ['aprovado', 'aguardando_pagamento'])
                        || (request.resource.data.planoConcedido in ['plus', 'pro', 'premium']
                            && request.resource.data.meses is int && request.resource.data.meses >= 1 && request.resource.data.meses <= 24
                            && request.resource.data.valorAcordado is number && request.resource.data.valorAcordado >= 0 && request.resource.data.valorAcordado <= 5000
                            && (request.resource.data.status == 'aprovado' || request.resource.data.valorAcordado > 0)));
      // Esperando o pagamento: o dono do site confirma (pago, com o pedido) ou cancela a cobrança.
      allow update: if donoDoSite()
                    && resource.data.status == 'aguardando_pagamento'
                    && mudou().hasOnly(['status', 'pedidoId', 'pagoEm', 'resposta', 'decididoPor', 'decididoEm'])
                    && request.resource.data.status in ['aprovado', 'recusado']
                    && (!('pedidoId' in request.resource.data) || request.resource.data.pedidoId is string)
                    && request.resource.data.decididoPor == request.auth.uid
                    && request.resource.data.decididoEm == request.time
                    && valorTextoAte(request.resource.data.get('resposta', ''), 600);
      allow delete: if false;
    }

    match /parcerias/{id} {
      // Formulário de Sócios e parcerias: só quem tem conta (e-mail confirmado), até 3 por dia; só a equipe lê.
      allow create: if logado()
                    && emailOk()
                    && request.resource.data.uid == request.auth.uid
                    && ritmoOk('parc')
                    && request.resource.data.keys().hasOnly(['nome', 'empresa', 'email', 'telefone', 'tipo', 'mensagem', 'site', 'cidade', 'uid', 'status', 'criadoEm'])
                    && request.resource.data.nome is string && request.resource.data.nome.size() >= 2 && request.resource.data.nome.size() <= 80
                    && request.resource.data.email is string && request.resource.data.email.matches('^[^@\\s]+@[^@\\s]+[.][^@\\s]+$') && request.resource.data.email.size() <= 120
                    && valorTextoAte(request.resource.data.get('empresa', ''), 100)
                    && valorTextoAte(request.resource.data.get('telefone', ''), 30)
                    && valorTextoAte(request.resource.data.get('site', ''), 200)
                    && valorTextoAte(request.resource.data.get('cidade', ''), 80)
                    && request.resource.data.tipo in ['socio', 'investidor', 'parceiro', 'anunciante', 'imprensa', 'outro']
                    && request.resource.data.mensagem is string && request.resource.data.mensagem.size() >= 10 && request.resource.data.mensagem.size() <= 2000
                    && request.resource.data.status == 'novo'
                    && request.resource.data.criadoEm == request.time;
      allow read: if admin();
      allow update: if podeComercial()
                    && mudou().hasOnly(['status', 'notas', 'atualizadoEm', 'responsavel', 'responsavelNome', 'assumidoEm'])
                    && request.resource.data.status in ['novo', 'em_conversa', 'proposta', 'fechado', 'recusado']
                    && (!mudou().hasAny(['responsavelNome', 'assumidoEm']) || responsavelOk())
                    && valorTextoAte(request.resource.data.get('notas', ''), 3000);
      allow delete: if podeComercial();
    }

    // Presença da equipe no painel (quem está online) e chat interno da equipe.
    match /equipe_presenca/{uid} {
      allow read: if admin();
      allow create, update: if admin()
                    && request.auth.uid == uid
                    && request.resource.data.keys().hasOnly(['nome', 'em', 'secao', 'item', 'itemNome', 'status'])
                    && request.resource.data.get('status', 'disponivel') in ['disponivel', 'pausa', 'fora']
                    && valorTextoAte(request.resource.data.get('nome', ''), 80)
                    && valorTextoAte(request.resource.data.get('secao', ''), 40)
                    && valorTextoAte(request.resource.data.get('item', null), 160)
                    && valorTextoAte(request.resource.data.get('itemNome', null), 120)
                    && request.resource.data.em == request.time;
      allow delete: if admin() && request.auth.uid == uid;
    }
