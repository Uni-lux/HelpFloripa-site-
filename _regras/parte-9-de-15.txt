    match /verificacoes/{uid} {
      function pedidoOk(d) {
        return d.keys().hasOnly(['uid', 'categoria', 'nomeLegal', 'conhecidoComo', 'documentoTipo', 'documento', 'links', 'motivo', 'status', 'criadoEm', 'atualizadoEm'])
            && d.uid == uid
            && d.categoria is string && d.categoria.matches('^(pessoa|profissional|negocio|publica|organizacao)$')
            && d.nomeLegal is string && d.nomeLegal.size() >= 5 && d.nomeLegal.size() <= 120
            && valorTextoAte(d.get('conhecidoComo', ''), 80)
            && d.documentoTipo is string && d.documentoTipo.matches('^(rg|cnh|passaporte|rne|cnpj)$')
            && d.documento is string && d.documento.size() <= 700000 && d.documento.matches('^data:image/jpeg;base64,[A-Za-z0-9+/=]+$')
            && (!('links' in d) || (d.links is list && d.links.size() <= 3))
            && valorTextoAte(d.get('motivo', ''), 500)
            && d.status == 'pendente'
            && d.criadoEm == request.time && d.atualizadoEm == request.time;
      }
      allow read: if (logado() && request.auth.uid == uid) || podeModerar();
      // Pedir (ou pedir de novo depois de não aprovado ou cancelado).
      allow create: if logado() && request.auth.uid == uid && pedidoOk(request.resource.data);
      allow update: if logado() && request.auth.uid == uid
                    && resource.data.status in ['recusada', 'cancelada']
                    && pedidoOk(request.resource.data);
      // Cancelar o próprio pedido (a foto do documento sai junto).
      allow update: if logado() && request.auth.uid == uid
                    && resource.data.status in ['pendente', 'em_analise']
                    && request.resource.data.status == 'cancelada'
                    && mudou().hasOnly(['status', 'atualizadoEm', 'documento'])
                    && !('documento' in request.resource.data)
                    && request.resource.data.atualizadoEm == request.time;
      // A moderação analisa e decide um pedido em aberto; ao aprovar ou recusar, a foto do documento é apagada.
      allow update: if podeModerar()
                    && resource.data.status in ['pendente', 'em_analise']
                    && mudou().hasOnly(['status', 'motivoRecusa', 'analisadaPor', 'analisadaEm', 'documento', 'atualizadoEm'])
                    && request.resource.data.status in ['em_analise', 'aprovada', 'recusada']
                    && request.resource.data.analisadaPor == request.auth.uid
                    && (request.resource.data.status == 'em_analise' || !('documento' in request.resource.data))
                    && valorTextoAte(request.resource.data.get('motivoRecusa', ''), 500);
      allow delete: if (logado() && request.auth.uid == uid) || podeModerar();
    }

    // Reporte de atendimento do suporte (só a moderação e os donos leem; o suporte não).
    //   atendimento_{chamado}_{uid}: quem abriu o chamado reporta como foi atendido (demora, grosseria...).
    //   usuario_{chamado}_{uid}: alguém da equipe reporta a pessoa do chamado (ofensas, ameaça, golpe...).
    match /reportes/{id} {
      function chamadoDe(c) { return get(/databases/$(database)/documents/suporte/$(c)).data; }
      allow get: if logado() && resource.data.autorId == request.auth.uid;
      allow read: if podeModerar();
      allow create: if logado()
                    && request.resource.data.keys().hasOnly(['tipo', 'chamadoId', 'assunto', 'autorId', 'alvoId', 'alvoNome', 'motivo', 'detalhe', 'status', 'criadoEm'])
                    && request.resource.data.autorId == request.auth.uid
                    && request.resource.data.chamadoId is string
                    && id == request.resource.data.tipo + '_' + request.resource.data.chamadoId + '_' + request.auth.uid
                    && request.resource.data.motivo.matches('^(demora|grosseria|nao_resolveu|errado|dados|ofensa|ameaca|golpe|spam|outro)$')
                    && valorTextoAte(request.resource.data.get('detalhe', ''), 1000)
                    && valorTextoAte(request.resource.data.get('assunto', ''), 120)
                    && valorTextoAte(request.resource.data.get('alvoNome', ''), 80)
                    && request.resource.data.status == 'novo'
                    && request.resource.data.criadoEm == request.time
                    && ((request.resource.data.tipo == 'atendimento'
                         && chamadoDe(request.resource.data.chamadoId).uid == request.auth.uid
                         && (!('alvoId' in request.resource.data) || request.resource.data.alvoId is string))
                        || (request.resource.data.tipo == 'usuario'
                         && admin()
                         && request.resource.data.alvoId == chamadoDe(request.resource.data.chamadoId).uid));
      allow update: if podeModerar()
                    && mudou().hasOnly(['status', 'nota', 'analisadaEm', 'analisadaPor', 'responsavel', 'responsavelNome', 'assumidoEm'])
                    && request.resource.data.status in ['novo', 'em_analise', 'resolvido', 'descartado']
                    && (!mudou().hasAny(['status', 'nota', 'analisadaEm', 'analisadaPor']) || request.resource.data.analisadaPor == request.auth.uid)
                    && responsavelOk()
                    && valorTextoAte(request.resource.data.get('nota', ''), 1000);
      allow delete: if false;
    }

    // Nota do atendimento (1 a 5) dada por quem abriu o chamado, uma vez; a equipe lê.
    match /suporte_notas/{chamadoId} {
      allow read: if admin() || (logado() && resource.data.uid == request.auth.uid);
      allow create: if logado()
                    && request.resource.data.keys().hasOnly(['uid', 'nota', 'comentario', 'atendidoPor', 'criadoEm'])
                    && request.resource.data.uid == request.auth.uid
                    && get(/databases/$(database)/documents/suporte/$(chamadoId)).data.uid == request.auth.uid
                    && request.resource.data.nota is int && request.resource.data.nota >= 1 && request.resource.data.nota <= 5
                    && valorTextoAte(request.resource.data.get('comentario', ''), 500)
                    && (!('atendidoPor' in request.resource.data) || request.resource.data.atendidoPor is string)
                    && request.resource.data.criadoEm == request.time;
      allow update, delete: if false;
    }

    // Planos dos negócios (Grátis + Plus, Pro e Premium, Pix manual). A pessoa cria o pedido (pendente) e pode
    // cancelar; só os donos do site conferem o pagamento, marcam pago/recusado e ativam planos/{uid}.
    match /planos_pedidos/{id} {
      allow read: if (logado() && resource.data.uid == request.auth.uid) || donoDoSite();
      allow create: if logado() && emailOk()
                    && request.resource.data.keys().hasOnly(['uid', 'nome', 'email', 'plano', 'meses', 'valor', 'txid', 'status', 'criadoEm'])
                    && request.resource.data.uid == request.auth.uid
                    && request.resource.data.plano in ['plus', 'pro', 'premium']
                    && request.resource.data.meses in [1, 3, 6, 12]
                    && request.resource.data.valor is number && request.resource.data.valor > 0 && request.resource.data.valor <= 1000
                    && request.resource.data.txid is string && request.resource.data.txid.matches('^[A-Z0-9]{4,25}$')
                    && valorTextoAte(request.resource.data.get('nome', ''), 80)
                    && valorTextoAte(request.resource.data.get('email', ''), 120)
                    && request.resource.data.status == 'pendente'
                    && request.resource.data.criadoEm == request.time;
      allow update: if logado() && resource.data.uid == request.auth.uid
                    && resource.data.status == 'pendente'
                    && mudou().hasOnly(['status'])
                    && request.resource.data.status == 'cancelado';
      allow update: if donoDoSite()
                    && resource.data.status == 'pendente'
                    && mudou().hasOnly(['status', 'decididoPor', 'decididoEm', 'motivo'])
                    && request.resource.data.status in ['pago', 'recusado']
                    && request.resource.data.decididoPor == request.auth.uid
                    && request.resource.data.decididoEm == request.time
                    && valorTextoAte(request.resource.data.get('motivo', ''), 300);
      allow delete: if false;
    }
    // Plano ativo da conta: público (selo e destaque); só os donos do site mudam.
    // Primeiro acesso: a própria pessoa começa 30 dias grátis no plano que escolher (só se a conta nunca teve plano).
    match /planos/{uid} {
      function tiposPerfisOk(l) {
        return l is list && l.size() >= 1 && l.size() <= 5
            && l.join(',').matches('^(servicos|delivery|lojinha|imoveis|saude)(,(servicos|delivery|lojinha|imoveis|saude)){0,4}$');
      }
      allow read: if logado();
      // Os 30 dias grátis agora vêm junto com a primeira assinatura paga (o dono do site soma ao confirmar o Pix):
      // ninguém grava o próprio plano.
      allow create, update: if donoDoSite()
                    && request.resource.data.keys().hasOnly(['plano', 'ate', 'ativadoEm', 'ativadoPor', 'pedidoId', 'negociacaoId', 'meses', 'teste', 'dias', 'tipos', 'semSelo'])
                    && request.resource.data.plano in ['plus', 'pro', 'premium', 'gratis']
                    // Plano combinado pode liberar só alguns perfis (tipos) e esconder o selo (semSelo).
                    && (!('tipos' in request.resource.data) || tiposPerfisOk(request.resource.data.tipos))
                    && request.resource.data.get('semSelo', false) is bool
                    && (!('dias' in request.resource.data) || (request.resource.data.dias is int && request.resource.data.dias >= 1 && request.resource.data.dias <= 800))
                    && request.resource.data.ate is timestamp
                    && request.resource.data.ativadoPor == request.auth.uid
                    && request.resource.data.ativadoEm == request.time;
      allow delete: if donoDoSite();
    }
    // Condição especial (baixa renda) ou parceria pelos planos: um pedido por pessoa (id = uid). A pessoa envia,
    // cancela enquanto está aberto e pede de novo depois de decidido; só os donos do site leem e negociam.
    // Condição especial: a pessoa diz qual documento prova a baixa renda (ou que manda depois), declara que é verdade e
    // pode anexar a foto/PDF; o anexo sai do documento quando os donos decidem (ou quando ela cancela).
    match /planos_negociacoes/{uid} {
      function comprovanteOk(x) {
        return x is map
            && x.keys().hasOnly(['tipo', 'url', 'nome'])
            && x.tipo in ['imagem', 'arquivo']
            && valorTextoAte(x.get('nome', ''), 120)
            && x.url is string && x.url.size() <= 750000
            && x.url.matches('^(data:(image/jpeg|application/pdf);base64,|https://firebasestorage[.]googleapis[.]com/).*');
      }
      function respNegOk(d) {
        return (d.get('responsavel', null) == null || d.responsavel is string)
            && valorTextoAte(d.get('responsavelNome', ''), 80)
            && (!('assumidoEm' in d) || d.assumidoEm == request.time || d.assumidoEm == resource.data.get('assumidoEm', null));
      }
