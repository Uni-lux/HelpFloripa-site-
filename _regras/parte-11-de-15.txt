      allow read: if admin() && request.auth.uid in resource.data.participantes;
      allow create, update: if membro()
                    && request.resource.data.keys().hasOnly(['participantes', 'ultimaEm', 'ultimaTexto', 'ultimoAutor', 'lidoEm'])
                    && request.resource.data.participantes == par.split('_')
                    && valorTextoAte(request.resource.data.get('ultimaTexto', ''), 140)
                    && (!('ultimoAutor' in request.resource.data) || request.resource.data.ultimoAutor in par.split('_'))
                    && request.resource.data.get('lidoEm', {}) is map;
      allow delete: if false;
      match /mensagens/{id} {
        allow read: if admin() && request.auth.uid in par.split('_');
        allow create: if admin() && par.split('_').size() == 2 && request.auth.uid in par.split('_') && msgEquipeOk(par);
        allow update: if admin() && request.auth.uid in par.split('_') && reacaoValida();
        allow update: if admin() && request.auth.uid in par.split('_') && (edicaoEquipeOk() || apagarEquipeOk(id));
        allow delete: if false;
      }
    }
    // Conteúdo de mensagem do chat da equipe apagada para todos (abre por 24 horas, só quem participa;
    // depois disso qualquer um da conversa apaga de vez).
    match /equipe_apagadas/{id} {
      function daConversaEq(c) { return c == 'equipe' || c.matches('^canal-[a-z]+$') || request.auth.uid in c.split('_'); }
      function msgEq(c) {
        return c == 'equipe' ? /databases/$(database)/documents/equipe_chat/$(id)
             : c.matches('^canal-[a-z]+$') ? /databases/$(database)/documents/equipe_canais/$(c.split('-')[1])/mensagens/$(id)
             : /databases/$(database)/documents/equipe_dm/$(c)/mensagens/$(id);
      }
      allow get: if admin() && daConversaEq(resource.data.conversa) && request.time < resource.data.apagadaEm + duration.value(24, 'h');
      allow create: if admin()
                    && request.resource.data.keys().hasOnly(['autorId', 'conversa', 'conteudo', 'apagadaEm'])
                    && request.resource.data.autorId == request.auth.uid
                    && request.resource.data.conversa is string
                    && daConversaEq(request.resource.data.conversa)
                    && get(msgEq(request.resource.data.conversa)).data.autorId == request.auth.uid
                    && getAfter(msgEq(request.resource.data.conversa)).data.apagada.por == request.auth.uid
                    && request.resource.data.apagadaEm == request.time
                    && request.resource.data.conteudo is map
                    && request.resource.data.conteudo.keys().hasOnly(['texto', 'tipoArquivo', 'anexo', 'nomeArquivo', 'tamanho', 'miniatura', 'duracao', 'resposta', 'mencoes', 'edicoes'])
                    && request.resource.data.conteudo.get('texto', '') == get(msgEq(request.resource.data.conversa)).data.get('texto', '')
                    && request.resource.data.conteudo.get('anexo', '') == get(msgEq(request.resource.data.conversa)).data.get('anexo', '');
      allow update: if false;
      allow delete: if admin() && daConversaEq(resource.data.conversa) && request.time >= resource.data.apagadaEm + duration.value(24, 'h');
    }

    // Conteúdo dos anexos do chat da equipe (data: URL de até ~900 mil caracteres).
    // Só quem participa da conversa lê; ninguém edita.
    match /equipe_anexos/{id} {
      allow read: if admin()
                  && (resource.data.conversa == 'equipe' || resource.data.conversa.matches('^canal-[a-z]+$')
                      || request.auth.uid in resource.data.conversa.split('_'));
      allow create: if admin()
                    && request.resource.data.keys().hasOnly(['autorId', 'conversa', 'dados', 'em'])
                    && request.resource.data.autorId == request.auth.uid
                    && request.resource.data.em == request.time
                    && request.resource.data.conversa is string
                    && (request.resource.data.conversa == 'equipe'
                        || request.resource.data.conversa.matches('^canal-(avisos|seguranca|moderacao|pagamentos|negocios|conta|comercial)$')
                        || (request.resource.data.conversa.split('_').size() == 2 && request.auth.uid in request.resource.data.conversa.split('_')))
                    && request.resource.data.dados is string
                    && request.resource.data.dados.size() <= 950000
                    && request.resource.data.dados.matches('^data:[a-z]+/[-+.a-z0-9]{1,80}(;[^,]{0,80})?,.*');
      allow update, delete: if false;
    }

    // Foto do grupo "Equipe" no chat da equipe: equipe_info/grupo. Qualquer pessoa da equipe troca.
    match /equipe_info/{id} {
      allow read: if admin();
      allow create, update: if admin()
                    && id == 'grupo'
                    && request.resource.data.keys().hasOnly(['foto', 'por', 'em'])
                    && (request.resource.data.foto == null
                        || (request.resource.data.foto is string && request.resource.data.foto.size() <= 250000 && fotoOk(request.resource.data.foto)))
                    && request.resource.data.por == request.auth.uid
                    && request.resource.data.em == request.time;
      allow delete: if false;
    }

    // Quadro de tarefas da equipe: quem faz o quê.
    match /equipe_tarefas/{id} {
      function tarefaOk() {
        let d = request.resource.data;
        return d.keys().hasOnly(['titulo', 'descricao', 'status', 'prioridade', 'responsavel', 'responsavelNome', 'prazo', 'ligacao', 'ligacaoNome', 'criadoPor', 'criadoPorNome', 'criadoEm', 'atualizadoEm', 'concluidaEm'])
            && d.titulo is string && d.titulo.size() >= 2 && d.titulo.size() <= 140
            && valorTextoAte(d.get('descricao', null), 2000)
            && d.status in ['a_fazer', 'fazendo', 'feito']
            && d.prioridade in ['baixa', 'normal', 'alta', 'urgente']
            && (d.get('responsavel', null) == null || (d.responsavel is string && exists(/databases/$(database)/documents/admins/$(d.responsavel))))
            && valorTextoAte(d.get('responsavelNome', null), 80)
            && (d.get('prazo', null) == null || d.prazo is timestamp)
            && valorTextoAte(d.get('ligacao', null), 160)
            && valorTextoAte(d.get('ligacaoNome', null), 120)
            && valorTextoAte(d.get('criadoPorNome', null), 80)
            && d.atualizadoEm == request.time;
      }
      allow read: if admin();
      allow create: if admin() && tarefaOk()
                    && request.resource.data.criadoPor == request.auth.uid
                    && request.resource.data.criadoEm == request.time;
      allow update: if admin() && tarefaOk()
                    && request.resource.data.criadoPor == resource.data.criadoPor
                    && request.resource.data.criadoEm == resource.data.criadoEm;
      allow delete: if admin() && (resource.data.criadoPor == request.auth.uid || papel() == 'dono');
    }

    // Notas internas da equipe sobre um usuário (só a equipe vê; ninguém edita).
    match /notas_internas/{id} {
      allow read: if admin();
      allow create: if admin()
                    && request.resource.data.keys().hasOnly(['alvo', 'texto', 'autorId', 'autorNome', 'em'])
                    && request.resource.data.alvo is string && request.resource.data.alvo.size() <= 128
                    && request.resource.data.autorId == request.auth.uid
                    && valorTextoAte(request.resource.data.get('autorNome', null), 80)
                    && request.resource.data.texto is string && request.resource.data.texto.size() >= 1 && request.resource.data.texto.size() <= 2000
                    && request.resource.data.em == request.time;
      allow update: if false;
      allow delete: if admin() && (resource.data.autorId == request.auth.uid || papel() == 'dono');
    }

    match /admin_log/{id} {
      // Só os donos do site leem o registro; o resto da equipe só vê as cobranças das reclamações.
      allow read: if donoDoSite() || (admin() && resource.data.acao == 'cobrar_reclamacao');
      allow create: if admin()
                    && request.resource.data.por == request.auth.uid
                    && request.resource.data.em == request.time
                    && request.resource.data.acao is string && request.resource.data.acao.size() <= 60
                    && valorTextoAte(request.resource.data.get('detalhe', ''), 1000);
      allow update, delete: if false;
    }

    match /estatisticas/{id} {
      allow read: if admin();
      allow write: if false;
    }

    match /relatorios/{id} {
      allow read: if admin();
      allow write: if false;
    }


    // =====================================================
    // LIMITE DIÁRIO (anti-spam): ritmo/{uid}
    // Janela de 24 h a partir da primeira ação. Por janela: até 500 mensagens,
    // 20 publicações e 200 comentários. Cada ação soma exatamente 1 no seu tipo.
    // =====================================================
    match /ritmo/{uid} {
      function limite(t) {
        return t == 'msg' ? 500 : (t == 'pub' ? 20 : (t == 'parc' ? 3 : 200));
      }
      function janelaNova() {
        return resource == null || resource.data.inicio < request.time - duration.value(1, 'd');
      }
      function contaOk(k, t) {
        let v = request.resource.data.get(k, 0);
        return v is int
            && v == (k == t ? (janelaNova() ? 1 : resource.data.get(k, 0) + 1)
                            : (janelaNova() ? 0 : resource.data.get(k, 0)));
      }
      allow get: if logado() && request.auth.uid == uid;
      allow read: if admin();
      allow create, update: if logado()
                    && request.auth.uid == uid
                    && request.resource.data.keys().hasOnly(['inicio', 'msg', 'pub', 'com', 'parc', 'ultimaEm', 'ultimoTipo'])
                    && request.resource.data.ultimoTipo in ['msg', 'pub', 'com', 'parc']
                    && request.resource.data.ultimaEm == request.time
                    && request.resource.data.inicio == (janelaNova() ? request.time : resource.data.inicio)
                    && contaOk('msg', request.resource.data.ultimoTipo)
                    && contaOk('pub', request.resource.data.ultimoTipo)
                    && contaOk('com', request.resource.data.ultimoTipo)
                    && contaOk('parc', request.resource.data.ultimoTipo)
                    && request.resource.data.get(request.resource.data.ultimoTipo, 0) <= limite(request.resource.data.ultimoTipo);
      allow delete: if false;
    }


    // =====================================================
    // PEDIDOS DE EXCLUSÃO DE CONTA: exclusoes/{uid}
    // A pessoa pede a exclusão; durante 30 dias pode recuperar a conta (apaga este
    // documento). Depois disso a limpeza automática (Admin SDK, fora das regras)
    // apaga a conta e os dados e anonimiza avaliações e reclamações.
    // =====================================================
    match /exclusoes/{uid} {
      allow get: if logado() && request.auth.uid == uid;
      allow read: if admin();
      allow create: if logado()
                    && request.auth.uid == uid
                    && request.resource.data.keys().hasOnly(['pedidaEm', 'motivo'])
                    && request.resource.data.pedidaEm == request.time
                    && valorTextoAte(request.resource.data.get('motivo', ''), 300);
      allow update: if false;
      allow delete: if logado() && request.auth.uid == uid;
    }


    // =====================================================
    // ESTRELAS NAS PUBLICAÇÕES: curtidas/{postId}_{uid}
    // Nota de 1 a 5, uma por pessoa em cada publicação. A pessoa pode
    // trocar a nota ou tirar. Ninguém dá estrelas na própria publicação.
    // =====================================================
    match /curtidas/{id} {

      allow read: if logado();

      function notaOk(d) {
        return d.nota is int && d.nota >= 1 && d.nota <= 5;
      }
      // Cada nota muda junto o resumo da publicação (diario_notas/{postId}); as regras do resumo conferem a conta.
      function resumoJunto(postId) {
        let r = /databases/$(database)/documents/diario_notas/$(postId);
        return existsAfter(r)
            && getAfter(r).data.get('soma', 0) != (exists(r) ? get(r).data.get('soma', 0) : 0);
      }
