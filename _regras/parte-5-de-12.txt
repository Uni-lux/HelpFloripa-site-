    // =====================================================
    // PEDIDOS / ATENDIMENTOS
    // O negócio registra o atendimento para pedir avaliação ao cliente.
    // Só vale entre pessoas que já conversam no chat.
    // =====================================================
    match /pedidos/{id} {

      allow read: if logado()
                  && (resource.data.vendedorId == request.auth.uid
                      || resource.data.clienteId == request.auth.uid);
      allow read: if admin();

      allow create: if logado()
                    && emailOk()
                    && request.resource.data.keys().hasOnly(['vendedorId', 'clienteId', 'negocioId', 'tipo', 'titulo', 'conversaId', 'criadoEm'])
                    && request.resource.data.vendedorId == request.auth.uid
                    && request.resource.data.clienteId is string
                    && request.resource.data.clienteId != request.auth.uid
                    && request.resource.data.tipo in ['servicos', 'delivery', 'lojinha', 'imoveis']
                    && request.resource.data.negocioId == request.auth.uid + '_' + request.resource.data.tipo
                    && exists(/databases/$(database)/documents/negocios/$(request.resource.data.negocioId))
                    && request.auth.uid in get(/databases/$(database)/documents/conversas/$(request.resource.data.conversaId)).data.participantes
                    && request.resource.data.clienteId in get(/databases/$(database)/documents/conversas/$(request.resource.data.conversaId)).data.participantes
                    && valorTextoAte(request.resource.data.get('titulo', ''), 120)
                    && request.resource.data.criadoEm == request.time
                    && !bloqueioEntre(request.auth.uid, request.resource.data.clienteId);

      allow update, delete: if false;
    }


    // =====================================================
    // AVALIAÇÕES
    //   n_{negocio}_{cliente}: cliente avalia o negócio — UMA por cliente em cada negócio
    //   c_{negocio}_{cliente}: negócio avalia o cliente — UMA por cliente em cada negócio
    //   d_{post}_{uid}       : alguém avalia uma publicação (1 vez por publicação)
    //   p_{pedido}_c / _v    : modelo antigo (1 por pedido). Continuam valendo, mas não
    //                          podem mais ser criados: pedir avaliação várias vezes para a
    //                          mesma pessoa não gera notas extras.
    // Para avaliar um negócio: e-mail confirmado, conta com 3 dias, os dois já escreveram
    // na conversa, não é o dono e sem bloqueio. Quem tem vínculo aceito no Social (amigos,
    // família...) também avalia: continua valendo uma avaliação por pessoa em cada negócio.
    // n_ e c_ podem ser EDITADAS pelo autor (troca a nota no resumo, sem contar de novo).
    // =====================================================
    match /avaliacoes/{id} {

      function base() {
        let d = request.resource.data;
        return d.keys().hasOnly(['tipo', 'autorId', 'autorNome', 'autorFoto', 'alvoId', 'negocioId', 'pedidoId', 'postId', 'nota', 'comentario', 'chaves', 'criadoEm'])
            && d.autorId == request.auth.uid
            && d.nota is int && d.nota >= 1 && d.nota <= 5
            && valorTextoAte(d.get('comentario', ''), 500)
            && valorTextoAte(d.get('autorNome', ''), 80)
            && fotoOk(d.get('autorFoto', ''))
            && d.criadoEm == request.time;
      }

      function resumoAtualizado(chave) {
        return getAfter(/databases/$(database)/documents/notas/$(chave)).data.ultima == id;
      }

      // "Atendimento confirmado": pedido real entre os dois, do mesmo negócio.
      function pedidoOk(d, cliente, negocioId) {
        return !('pedidoId' in d)
            || (get(/databases/$(database)/documents/pedidos/$(d.pedidoId)).data.clienteId == cliente
                && get(/databases/$(database)/documents/pedidos/$(d.pedidoId)).data.negocioId == negocioId);
      }

      function doCliente() {
        let d = request.resource.data;
        let dono = donoDoNegocio(d.negocioId);
        let eu = request.auth.uid;
        return id == 'n_' + d.negocioId + '_' + eu
            && d.alvoId == dono
            && dono != eu
            && d.chaves == ['neg_' + d.negocioId]
            && contaAntiga(eu)
            && falou(eu, dono, eu)
            && falou(eu, dono, dono)
            && !bloqueioEntre(eu, dono)
            && pedidoOk(d, eu, d.negocioId)
            && resumoAtualizado('neg_' + d.negocioId);
      }

      function doVendedor() {
        let d = request.resource.data;
        let eu = request.auth.uid;
        return id == 'c_' + d.negocioId + '_' + d.alvoId
            && donoDoNegocio(d.negocioId) == eu
            && d.alvoId != eu
            && d.chaves == ['cli_' + d.alvoId]
            && falou(eu, d.alvoId, eu)
            && falou(eu, d.alvoId, d.alvoId)
            && !bloqueioEntre(eu, d.alvoId)
            && pedidoOk(d, d.alvoId, d.negocioId)
            && resumoAtualizado('cli_' + d.alvoId);
      }

      function daPublicacao() {
        let d = request.resource.data;
        let post = get(/databases/$(database)/documents/diario/$(d.postId)).data;
        return id == 'd_' + d.postId + '_' + request.auth.uid
            && post.autorId != request.auth.uid
            && d.alvoId == post.autorId
            && d.chaves == ['post_' + d.postId, 'pub_' + post.autorId]
            && resumoAtualizado('post_' + d.postId)
            && resumoAtualizado('pub_' + post.autorId);
      }

      allow read: if logado();

      allow create: if logado()
                    && emailOk()
                    && base()
                    && ((request.resource.data.tipo == 'negocio' && doCliente())
                        || (request.resource.data.tipo == 'cliente' && doVendedor())
                        || (request.resource.data.tipo == 'publicacao' && daPublicacao()));

      // Edição de n_ / c_ pelo próprio autor: só nota, comentário e (uma vez) o pedido confirmado.
      allow update: if logado()
                    && emailOk()
                    && resource.data.autorId == request.auth.uid
                    && (id.matches('^n_.*') || id.matches('^c_.*'))
                    && mudou().hasOnly(['nota', 'comentario', 'atualizadoEm', 'pedidoId'])
                    && request.resource.data.nota is int && request.resource.data.nota >= 1 && request.resource.data.nota <= 5
                    && valorTextoAte(request.resource.data.get('comentario', ''), 500)
                    && request.resource.data.atualizadoEm == request.time
                    && (!mudou().hasAny(['pedidoId'])
                        || (!('pedidoId' in resource.data)
                            && pedidoOk(request.resource.data, id.matches('^n_.*') ? request.auth.uid : resource.data.alvoId, resource.data.negocioId)))
                    && !bloqueioEntre(request.auth.uid, resource.data.alvoId)
                    // nota mudou: o resumo precisa trocar a nota antiga pela nova NESTA mesma gravação
                    && (request.resource.data.nota == resource.data.nota
                        || (getAfter(/databases/$(database)/documents/notas/$(resource.data.chaves[0])).data.ultima == id
                            && getAfter(/databases/$(database)/documents/notas/$(resource.data.chaves[0])).data.soma
                               == get(/databases/$(database)/documents/notas/$(resource.data.chaves[0])).data.soma
                                  - resource.data.nota + request.resource.data.nota));

      allow delete: if false;
    }


    // =====================================================
    // RESUMO DAS NOTAS (média e quantas de cada estrela)
    //   neg_{negocioId} · cli_{uid} · pub_{uid} · post_{postId}
    // Só aceita somar exatamente a avaliação nova gravada junto.
    // =====================================================
    match /notas/{chave} {

      function caminhoNova() {
        return /databases/$(database)/documents/avaliacoes/$(request.resource.data.ultima);
      }

      function avaliacaoNova() {
        return request.resource.data.ultima is string
            && !exists(caminhoNova())
            && existsAfter(caminhoNova())
            && getAfter(caminhoNova()).data.autorId == request.auth.uid
            && chave in getAfter(caminhoNova()).data.chaves;
      }

      function estrela() {
        return 'n' + string(getAfter(caminhoNova()).data.nota);
      }

      allow read: if logado();

      allow create: if logado()
                    && avaliacaoNova()
                    && request.resource.data.keys().hasOnly(['total', 'soma', 'n1', 'n2', 'n3', 'n4', 'n5', 'ultima', 'atualizadoEm'])
                    && request.resource.data.keys().hasAll(['total', 'soma', 'n1', 'n2', 'n3', 'n4', 'n5', 'ultima'])
                    && request.resource.data.total == 1
                    && request.resource.data.soma == getAfter(caminhoNova()).data.nota
                    && request.resource.data[estrela()] == 1
                    && request.resource.data.n1 + request.resource.data.n2 + request.resource.data.n3
                       + request.resource.data.n4 + request.resource.data.n5 == 1;

      allow update: if logado()
                    && avaliacaoNova()
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['total', 'soma', 'ultima', 'atualizadoEm', estrela()])
                    && request.resource.data.total == resource.data.total + 1
                    && request.resource.data.soma == resource.data.soma + getAfter(caminhoNova()).data.nota
                    && request.resource.data[estrela()] == resource.data.get(estrela(), 0) + 1;

      // Edição de uma avaliação n_/c_: troca a nota antiga pela nova (o total não muda).
      function edicao() {
        let antes = get(caminhoNova()).data;
        let depois = getAfter(caminhoNova()).data;
        return request.resource.data.ultima is string
            && exists(caminhoNova())
            && depois.autorId == request.auth.uid
            && chave in depois.chaves
            && antes.nota != depois.nota
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['soma', 'ultima', 'atualizadoEm', 'n' + string(antes.nota), 'n' + string(depois.nota)])
            && request.resource.data.total == resource.data.total
            && request.resource.data.soma == resource.data.soma - antes.nota + depois.nota
            && resource.data.get('n' + string(antes.nota), 0) >= 1
            && request.resource.data['n' + string(antes.nota)] == resource.data.get('n' + string(antes.nota), 0) - 1
            && request.resource.data['n' + string(depois.nota)] == resource.data.get('n' + string(depois.nota), 0) + 1;
      }

      allow update: if logado() && edicao();

      allow delete: if false;
    }


    // =====================================================
    // RECLAMAÇÕES: resposta pública do negócio
    // Reclamação = avaliação de negócio com nota 1 ou 2 (avaliacoes/{id}).
    // reclamacoes/{id da avaliação}: o negócio avaliado responde uma vez
    // (pode editar o texto) e o cliente que reclamou marca como resolvida.
    // =====================================================
    match /reclamacoes/{avalId} {

      allow read: if logado();

      function aval() {
        return get(/databases/$(database)/documents/avaliacoes/$(avalId)).data;
      }

      function respostaOk(d) {
        return d.texto is string && d.texto.size() >= 2 && d.texto.size() <= 1000;
      }

      allow create: if logado()
                    && emailOk()
                    && exists(/databases/$(database)/documents/avaliacoes/$(avalId))
                    && aval().tipo == 'negocio'
                    && aval().nota <= 2
                    && aval().alvoId == request.auth.uid
                    && request.resource.data.keys().hasOnly(['texto', 'autorId', 'criadoEm'])
                    && request.resource.data.autorId == request.auth.uid
                    && respostaOk(request.resource.data)
                    && request.resource.data.criadoEm == request.time;
