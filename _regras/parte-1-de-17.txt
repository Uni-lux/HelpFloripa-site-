rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function logado() {
      return request.auth != null;
    }

    // E-mail confirmado (login com Google já vem confirmado).
    // Exigido para ações que chegam em outras pessoas: mensagens, seguir,
    // vínculos, negócios, anúncios e publicações. Freia contas falsas e robôs.
    // Conta suspensa ou banida pela equipe (sancoes/{uid}) também não age.
    function emailOk() {
      return request.auth.token.email_verified == true && !sancionado(request.auth.uid);
    }

    // Para anunciar (criar perfil de negócio ou anúncio de imóvel), a conta precisa ser verificada:
    // usuarios/{uid}.verificado só vira true pela moderação com o pedido aprovado (seloLiberado) ou pelo dono do site.
    function contaVerificada() {
      return get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.get('verificado', false) == true;
    }

    // Administrador: documento admins/{uid} criado à mão no console do Firebase.
    function admin() {
      return request.auth != null
          && exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    // Papel na equipe (campo "papel" no documento admins/{uid}, definido no console).
    // Sem o campo = "dono" (acesso total). Outros: moderacao, suporte, comercial.
    function papel() {
      return get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.get('papel', 'dono');
    }
    // Moderação: selo, esconder/remover conteúdo, suspender/banir, denúncias, comunicados.
    function podeModerar() {
      return admin() && papel() in ['dono', 'moderacao'];
    }
    // Selo de verificado: tirar é livre para a moderação; dar, só com o pedido aprovado
    // (verificacoes/{uid}) ou pelo dono do site (casos especiais, com motivo no registro).
    function seloLiberado(uid) {
      return request.resource.data.get('verificado', false) != true
          || resource.data.get('verificado', false) == true
          || papel() == 'dono'
          || (exists(/databases/$(database)/documents/verificacoes/$(uid))
              && get(/databases/$(database)/documents/verificacoes/$(uid)).data.status == 'aprovada');
    }
    // Comercial: propostas de sócios e parcerias.
    function podeComercial() {
      return admin() && papel() in ['dono', 'comercial'];
    }
    // Só os donos do site: planos (pagamentos, negociações, receita) e o registro de ações da equipe.
    function donoDoSite() {
      return admin() && papel() == 'dono';
    }

    // Suspensão (até uma data) ou banimento (sem data).
    function sancionado(uid) {
      let s = /databases/$(database)/documents/sancoes/$(uid);
      return exists(s)
          && (get(s).data.tipo == 'banimento'
              || (get(s).data.get('ate', null) is timestamp && get(s).data.ate > request.time));
    }

    // Banido de um programa (um tipo de perfil de negócio ou a Comunidade): programa_banidos/{uid}_{tipo},
    // sem data = para sempre; com `ate`, até essa data.
    function banidoDoPrograma(uid, tipo) {
      let b = /databases/$(database)/documents/programa_banidos/$(uid + '_' + tipo);
      return exists(b) && (get(b).data.get('ate', null) == null || get(b).data.ate > request.time);
    }

    // Campos que só podem ser alterados pelo painel/admin (Admin SDK ignora as regras).
    function camposProtegidos() {
      return ['verificado', 'avaliacaoMedia', 'totalAvaliacoes'];
    }

    function naoTocaCamposProtegidosNoCreate() {
      return !request.resource.data.keys().hasAny(camposProtegidos());
    }

    // No update também ficam travados plano/valorPlanoMensal/tipoUsuario:
    // são definidos no cadastro e nunca mudam pelo site.
    function naoTocaCamposProtegidosNoUpdate() {
      return !request.resource.data.diff(resource.data).affectedKeys()
               .hasAny(camposProtegidos().concat(['plano', 'valorPlanoMensal', 'tipoUsuario']));
    }

    // Campo de texto opcional com tamanho máximo (evita documentos gigantes).
    // (Lê o campo uma vez só com get: gasta menos do limite de 1.000 expressões por gravação.)
    function textoAte(campo, max) {
      return valorTextoAte(request.resource.data.get(campo, null), max);
    }

    // Mesmo limite, para um valor qualquer (ex.: campo dentro de um mapa).
    function valorTextoAte(v, max) {
      return v == null || (v is string && v.size() <= max);
    }

    // Enquete (publicação ou flash): pergunta (até 120) e 2 a 4 opções (até 40 cada).
    // No flash, x e y dizem onde a enquete fica na tela (0 a 1).
    // fimEm: quando a votação acaba (depois de inicio, até maxMin minutos depois; sem fimEm = enquete antiga, sem fim).
    function enqueteOk(e, inicio, maxMin) {
      return e is map && e.keys().hasOnly(['pergunta', 'opcoes', 'x', 'y', 'k', 'fimEm', 'anonima', 'cor', 'certa'])
          && e.get('anonima', false) is bool
          // Teste (quiz): a resposta certa = índice de uma das opções
          && (!('certa' in e) || (e.certa is int && e.certa >= 0 && e.certa < e.opcoes.size()))
          && (!('cor' in e) || corFigOk(e.cor))
          && (!('fimEm' in e) || (e.fimEm is timestamp && e.fimEm > inicio && e.fimEm <= inicio + duration.value(maxMin, 'm')))
          && e.pergunta.size() >= 1 && e.pergunta.size() <= 120
          // 2 a 4 opções de 1 a 40 letras, conferidas numa expressão só (opção não tem quebra de linha; o regex já
          // garante de 2 a 4 linhas)
          && e.opcoes.size() <= 4 && e.opcoes.join('\n').matches('^[^\n]{1,40}(\n[^\n]{1,40}){1,3}$')
          // posição x e y entre 0 e 1, conferidas num regex só (economiza expressões: o flash com tudo junto
          // estava no limite de 1.000). O tamanho k (dois dedos, 09/11/2026) não é conferido aqui: não cabia no limite e
          // o visor já prende entre 0,6 e 1,3 ao desenhar.
          && (string(e.get('x', 0)) + ' ' + string(e.get('y', 0))).matches('^(0|1|0[.][0-9]+|1[.]0+) (0|1|0[.][0-9]+|1[.]0+)$');
    }
    // Cor da enquete / caixa de perguntas no flash (branco é o padrão e não vai gravado).
    function corFigOk(c) {
      return c is string && c.matches('^(escuro|azul|rosa|laranja|verde|roxo)$');
    }
    // Caixa de perguntas (publicação ou flash): o autor pergunta e quem vê responde com texto.
    // anonima: quem responde não aparece para o autor (só a moderação sabe, em caixa_autoria).
    function caixaOk(c) {
      return c is map && c.keys().hasOnly(['pergunta', 'anonima', 'x', 'y', 'k', 'cor'])
          && (!('cor' in c) || corFigOk(c.cor))
          && c.pergunta.size() >= 1 && c.pergunta.size() <= 120
          && c.anonima is bool
          // posição x e y entre 0 e 1 num regex só (como na enquete), para caber no limite de 1.000 (k: veja a enquete)
          && (string(c.get('x', 0)) + ' ' + string(c.get('y', 0))).matches('^(0|1|0[.][0-9]+|1[.]0+) (0|1|0[.][0-9]+|1[.]0+)$');
    }
    // Flash: figurinha que leva ao perfil de um negócio (qualquer negócio que exista).
    function linkFlashOk(l) {
      return l is map && l.keys().hasOnly(['negocioId', 'nome', 'tipo', 'x', 'y', 'k'])
          && l.negocioId is string && l.negocioId.matches('^[A-Za-z0-9_-]{1,100}$')
          && exists(/databases/$(database)/documents/negocios/$(l.negocioId))
          && valorTextoAte(l.get('nome', ''), 80)
          && l.tipo in ['servicos', 'delivery', 'lojinha', 'imoveis', 'saude']
          // x e y num regex só (como na caixa): sobra espaço no limite de 1.000 para o limite diário do flash
          && (string(l.get('x', 0)) + ' ' + string(l.get('y', 0))).matches('^(0|1|0[.][0-9]+|1[.]0+) (0|1|0[.][0-9]+|1[.]0+)$');
    }
    // Flash repostado: só quem foi mencionado no flash original (que ainda existe) reposta.
    function repostOk(r) {
      let orig = /databases/$(database)/documents/stories/$(r.storyId);
      return r is map && r.keys().hasOnly(['storyId', 'autorId', 'nome'])
          && r.storyId is string && r.storyId.matches('^[A-Za-z0-9]{1,64}$')
          && exists(orig)
          && get(orig).data.autorId == r.autorId
          && get(orig).data.get('mencoes', []).hasAny([request.auth.uid])
          && valorTextoAte(r.get('nome', ''), 80);
    }
    // Flash nos perfis de negócio da própria pessoa (ids = uid_tipo, até 4) e se aparece também no perfil pessoal.
    function negociosFlashOk(d) {
      let n = '(' + request.auth.uid + '_(servicos|delivery|lojinha|imoveis|saude))';
      return (!('negocios' in d) || (d.negocios is list && d.negocios.size() >= 1 && d.negocios.size() <= 4
                                      && d.negocios.join(' ').matches('^' + n + '( ' + n + '){0,3}$')))
          && (!('perfil' in d) || (d.perfil is bool && (d.perfil || 'negocios' in d)));
    }
    // Publicação do Diário compartilhada no flash: a publicação existe, o autor confere e ele não desligou
    // "Permitir que compartilhem minhas publicações no Flash" (pref_publicas.permiteFlash; sem o documento, pode).
    function postFlashOk(r) {
      let orig = /databases/$(database)/documents/diario/$(r.postId);
      let pref = /databases/$(database)/documents/pref_publicas/$(r.autorId);
      return r is map && r.keys().hasOnly(['postId', 'autorId', 'nome'])
          && r.postId is string && r.postId.matches('^[A-Za-z0-9]{1,64}$')
          && r.autorId is string
          && exists(orig) && get(orig).data.autorId == r.autorId
          && (r.autorId == request.auth.uid || !exists(pref) || get(pref).data.get('permiteFlash', true) != false)
          && valorTextoAte(r.get('nome', ''), 80);
    }
    // Flash: figurinhas tocáveis (até 8), conferidas numa expressão só (a gravação "com tudo" do flash já estava no
    // limite de 1.000): "h|hashtag|x|y|w|h", "m|uid|x|y|w|h" e a localização "l|nome|lat|lng|x|y|w|h" (lat/lng no Brasil).
    // x, y, w, h = onde a figurinha ficou no quadro (fração de 0 a 1).
    function toquesOk(t) {
      let lugar = 'l[|][^|\n]{2,60}[|](-(3[0-3]|[12]?[0-9])|[0-5])([.][0-9]{1,6})?[|]-(7[0-4]|[3-6][0-9]|2[89])([.][0-9]{1,6})?';
      let item = '(h[|][^|\n]{1,28}|m[|][A-Za-z0-9]{1,64}|' + lugar + ')([|][01]([.][0-9]{1,3})?){4}';
      return t is list && t.size() <= 8
          && t.join('\n').matches('^' + item + '(\n' + item + '){0,7}$');
    }
    function extrasFlashOk(d) {
      return negociosFlashOk(d) && flashNovos2Ok(d)
          && (!('toques' in d) || toquesOk(d.toques))
          // menção oculta: a pessoa foi mencionada (recebe o aviso), mas não aparece para quem vê
          && (!('mencoesOcultas' in d) || uidsOk(d.mencoesOcultas, 10))
          && (!('caixa' in d) || caixaOk(d.caixa))
          && (!('link' in d) || linkFlashOk(d.link))
          && (!('repostDe' in d) || repostOk(d.repostDe))
          && (!('postDe' in d) || postFlashOk(d.postDe));
    }
// ===== FIM DA PARTE 1 DE 17 =====
