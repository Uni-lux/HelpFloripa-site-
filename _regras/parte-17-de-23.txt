    // ===================================================
    // PREFERÊNCIAS PÚBLICAS: pref_publicas/{uid} — o que os outros precisam saber antes de interagir
    // (quem pode marcar/mencionar a pessoa). Só o dono grava; qualquer pessoa logada lê.
    // ===================================================
    match /pref_publicas/{uid} {
      allow read: if logado();
      allow create, update: if logado() && uid == request.auth.uid
                    && request.resource.data.keys().hasOnly(['permiteMarcar', 'permiteFlash', 'presentes', 'atualizadoEm', 'msgPessoal', 'msgNegocios', 'negociosNoPerfil', 'perfil', 'pronomes', 'profissao'])
                    // pronomes e profissão/ramo (13/11/2026, opcionais, aparecem no perfil)
                    && request.resource.data.get('pronomes', '') is string && request.resource.data.get('pronomes', '').size() <= 24
                    && request.resource.data.get('profissao', '') is string && request.resource.data.get('profissao', '').size() <= 40
                    // quem pode mandar mensagem (pessoal e negócios) e se os atalhos dos negócios aparecem no perfil
                    && request.resource.data.get('msgPessoal', 'seguindo') in ['todos', 'seguindo', 'fechado']
                    && request.resource.data.get('msgNegocios', 'todos') in ['todos', 'pedidos']
                    && request.resource.data.get('negociosNoPerfil', true) is bool
                    // o que aparece no perfil para quem visita (12/11/2026): cidade, avaliação e destaques do Flash
                    && (!('perfil' in request.resource.data) || (request.resource.data.perfil is map
                        && request.resource.data.perfil.keys().hasOnly(['cidade', 'nota', 'destaques'])
                        && request.resource.data.perfil.get('cidade', true) is bool && request.resource.data.perfil.get('nota', true) is bool
                        && request.resource.data.perfil.get('destaques', true) is bool))
                    && (!('presentes' in request.resource.data) || request.resource.data.presentes in ['todos', 'seguindo', 'vinculos', 'ninguem'])
                    && (!('permiteMarcar' in request.resource.data) || request.resource.data.permiteMarcar in ['todos', 'seguindo', 'ninguem'])
                    && (!('permiteFlash' in request.resource.data) || request.resource.data.permiteFlash is bool);
      allow delete: if logado() && uid == request.auth.uid;
    }

    // LINKS DO PERFIL (13/11/2026): contatos_perfil/{uid}_{instagram|whatsapp|tiktok|x|youtube} {uid, tipo, valor, ver, em}.
    // Quem vê: todos (com login), só quem segue a pessoa ou ninguém; o dono e a moderação sempre. Bloqueio esconde.
    match /contatos_perfil/{id} {
      allow get: if logado() && resource != null
                    && (resource.data.uid == request.auth.uid || admin()
                        || (!bloqueioEntre(request.auth.uid, resource.data.uid)
                            && (resource.data.ver == 'todos'
                                || (resource.data.ver == 'seguidores'
                                    && exists(/databases/$(database)/documents/relacoes/$(request.auth.uid + '_' + resource.data.uid))))));
      allow list: if false;
      allow create, update: if logado()
                    && request.resource.data.keys().hasOnly(['uid', 'tipo', 'valor', 'ver', 'em'])
                    && request.resource.data.uid == request.auth.uid
                    && id == request.auth.uid + '_' + request.resource.data.tipo
                    && request.resource.data.ver in ['todos', 'seguidores', 'ninguem']
                    && request.resource.data.valor is string
                    && ((request.resource.data.tipo == 'instagram' && request.resource.data.valor.matches('^[A-Za-z0-9._]{1,30}$'))
                        || (request.resource.data.tipo == 'whatsapp' && request.resource.data.valor.matches('^[0-9]{10,13}$'))
                        || (request.resource.data.tipo == 'tiktok' && request.resource.data.valor.matches('^[A-Za-z0-9._]{2,24}$'))
                        || (request.resource.data.tipo == 'x' && request.resource.data.valor.matches('^[A-Za-z0-9_]{1,15}$'))
                        || (request.resource.data.tipo == 'youtube' && request.resource.data.valor.matches('^[A-Za-z0-9._-]{3,30}$')))
                    && request.resource.data.em == request.time;
      allow delete: if logado() && (resource.data.uid == request.auth.uid || admin());
    }

    // NOTAS (12/11/2026): notas/{uid}/itens/{id} — anotações pessoais, só a própria pessoa lê e grava.
    match /notas/{uid}/itens/{id} {
      allow read, delete: if logado() && uid == request.auth.uid;
      allow create, update: if logado() && uid == request.auth.uid
                    && request.resource.data.keys().hasOnly(['titulo', 'texto', 'cor', 'pasta', 'fixada', 'arquivada', 'lixeira', 'prazo', 'criadoEm', 'em'])
                    && request.resource.data.titulo is string && request.resource.data.titulo.size() <= 120
                    && request.resource.data.texto is string && request.resource.data.texto.size() <= 20000
                    && request.resource.data.get('pasta', '') is string && request.resource.data.get('pasta', '').size() <= 40
                    && request.resource.data.get('cor', '') is string && request.resource.data.get('cor', '').size() <= 12
                    && request.resource.data.get('prazo', '') is string && request.resource.data.get('prazo', '').size() <= 10
                    && request.resource.data.get('fixada', false) is bool && request.resource.data.get('arquivada', false) is bool
                    && request.resource.data.get('lixeira', 0) is number && request.resource.data.em is number;
    }

    match /rascunhos/{uid}/itens/{id} {
      allow read, delete: if logado() && uid == request.auth.uid;
      allow create, update: if logado() && uid == request.auth.uid
                    && request.resource.data.keys().hasOnly(['texto', 'fotos', 'atualizadoEm'])
                    && valorTextoAte(request.resource.data.get('texto', ''), 1000)
                    && (!('fotos' in request.resource.data) || (request.resource.data.fotos is list && request.resource.data.fotos.size() <= 4));
    }

    // ===================================================
    // Ranking do Diário (Etapa A, com o plano Blaze): ranking_diario/br e ranking_diario/uf_XX, montados a cada 15 min
    // pela Cloud Function montarRankingDiario (functions/diario-ranking.js). Só o servidor grava; quem está logado lê.
    // servidor_atendimento/{conversa} (espera do tempo de resposta) é só do servidor: sem regra, ninguém lê nem grava.
    // ===================================================
    match /ranking_diario/{id} {
      allow read: if logado();
      allow write: if false;
    }

    // ===================================================
    // RESUMO DAS ESTRELAS DE UMA PUBLICAÇÃO: diario_notas/{postId} { total, soma, n1..n5 }
    // Uma leitura mostra a média e quantas pessoas deram 5, 4, 3, 2 e 1 estrela, com mil ou um milhão
    // de notas. Só muda junto com a nota de quem grava (curtidas/{postId}_{uid}), na conta exata:
    // a nota de antes sai e a nova entra. A moderação pode recalcular.
    // ===================================================
    match /diario_notas/{postId} {
      allow read: if logado();

      function minhaNota() { return /databases/$(database)/documents/curtidas/$(postId + '_' + request.auth.uid); }
      function contaCerta() {
        let a = exists(minhaNota()) ? get(minhaNota()).data.nota : 0;
        let d = existsAfter(minhaNota()) ? getAfter(minhaNota()).data.nota : 0;
        let r = resource == null ? {} : resource.data;
        let n = request.resource.data;
        return a != d
            && n.keys().hasOnly(['total', 'soma', 'n1', 'n2', 'n3', 'n4', 'n5'])
            && n.get('total', 0) == r.get('total', 0) + (d > 0 ? 1 : 0) - (a > 0 ? 1 : 0)
            && n.get('soma', 0) == r.get('soma', 0) + d - a
            && n.get('n1', 0) == r.get('n1', 0) + (d == 1 ? 1 : 0) - (a == 1 ? 1 : 0)
            && n.get('n2', 0) == r.get('n2', 0) + (d == 2 ? 1 : 0) - (a == 2 ? 1 : 0)
            && n.get('n3', 0) == r.get('n3', 0) + (d == 3 ? 1 : 0) - (a == 3 ? 1 : 0)
            && n.get('n4', 0) == r.get('n4', 0) + (d == 4 ? 1 : 0) - (a == 4 ? 1 : 0)
            && n.get('n5', 0) == r.get('n5', 0) + (d == 5 ? 1 : 0) - (a == 5 ? 1 : 0);
      }
      allow create, update: if logado() && contaCerta();
      // Recalcular (moderação): números inteiros, sem campo estranho.
      allow create, update: if podeModerar()
                    && request.resource.data.keys().hasOnly(['total', 'soma', 'n1', 'n2', 'n3', 'n4', 'n5'])
                    && request.resource.data.total is int && request.resource.data.soma is int
                    && request.resource.data.n1 is int && request.resource.data.n2 is int && request.resource.data.n3 is int
                    && request.resource.data.n4 is int && request.resource.data.n5 is int;
      allow delete: if podeModerar();
    }

    // =====================================================
    // COMENTÁRIOS: comentarios/{id}
    // - Resposta: respostaA = id do comentário respondido (mesma publicação)
    //   e respostaAutorId = autor dele (para o aviso).
    // - Quem comentou edita o texto. O dono da publicação oculta um comentário.
    // - Quem comentou ou o dono da publicação apaga.
    // - Publicação com comentários ocultos não recebe comentários novos.
    // =====================================================
    match /comentarios/{id} {

      allow read: if logado();

      // Texto até 500; pode ficar vazio quando o comentário é uma foto, um áudio, uma figurinha ou uma enquete.
      function textoComentarioOk(d) {
        return d.texto is string && d.texto.size() <= 500 && (d.texto.size() >= 1 || 'midia' in d || 'enquete' in d);
      }
      // Foto, áudio (até 60 s) ou figurinha no comentário: do próprio site (dentro do documento ou Storage).
      function midiaComentOk(d) {
        return !('midia' in d)
            || (d.midia is map && d.midia.keys().hasOnly(['tipo', 'url', 'dur'])
                && d.midia.tipo in ['imagem', 'audio', 'figurinha']
                && d.midia.url is string && d.midia.url.size() <= 700000
                && d.midia.url.matches(d.midia.tipo == 'audio'
                     ? '^(data:audio/[a-z0-9.+-]{1,20}(;codecs=[a-z0-9.]{1,12})?;base64,|https://firebasestorage[.]googleapis[.]com/).*'
                     : '^(data:image/(jpeg|png|webp);base64,|https://firebasestorage[.]googleapis[.]com/).*')
                && (!('dur' in d.midia) || (d.midia.dur is number && d.midia.dur > 0 && d.midia.dur <= 61)));
      }
      // Enquete no comentário (como no Instagram): só o autor da publicação; pergunta e 2 a 4 opções.
      function enqueteComentOk(d) {
        return !('enquete' in d)
            || (d.postAutorId == request.auth.uid
                && d.enquete is map && d.enquete.keys().hasOnly(['pergunta', 'opcoes'])
                && d.enquete.pergunta is string && d.enquete.pergunta.size() >= 1 && d.enquete.pergunta.size() <= 120
                && d.enquete.opcoes is list && d.enquete.opcoes.size() >= 2 && d.enquete.opcoes.size() <= 4
                && d.enquete.opcoes.join('\n').matches('^[^\n]{1,60}(\n[^\n]{1,60}){1,3}$'));
      }

      // Quem pode comentar (escolha do autor): o autor sempre pode.
      function segue(a, b) { return exists(/databases/$(database)/documents/relacoes/$(a + '_' + b)); }
      function podeComentar(p) {
        let q = p.get('quemComenta', 'todos');
        return p.autorId == request.auth.uid
            || q == 'todos'
            || (q == 'seguidores' && segue(request.auth.uid, p.autorId))
            || (q == 'mutuos' && segue(request.auth.uid, p.autorId) && segue(p.autorId, request.auth.uid));
      }

      function respostaOk(d) {
        return (!('respostaA' in d) && !('respostaAutorId' in d))
            || (d.respostaA is string
                && exists(/databases/$(database)/documents/comentarios/$(d.respostaA))
                && get(/databases/$(database)/documents/comentarios/$(d.respostaA)).data.postId == d.postId
                && d.respostaAutorId == get(/databases/$(database)/documents/comentarios/$(d.respostaA)).data.autorId);
      }
