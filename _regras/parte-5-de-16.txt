    // Ligações de voz e vídeo (chamadas.js): só entre as duas pessoas de uma conversa que já existe (do site ou do chat
    // da equipe) e sem bloqueio.
    // A chamada guarda a "campainha" (oferta/resposta do WebRTC) e o resultado; áudio e vídeo nunca passam por aqui.
    match /chamadas/{id} {
      function daChamada() { return logado() && (request.auth.uid == resource.data.de || request.auth.uid == resource.data.para); }
      function sdpOk(x) { return x is map && x.keys().hasOnly(['type', 'sdp']) && x.type is string && x.sdp is string && x.sdp.size() <= 30000; }
      allow get, list: if daChamada();
      allow create: if logado()
                    && request.resource.data.keys().hasOnly(['de', 'para', 'conv', 'tipo', 'status', 'oferta', 'criadoEm'])
                    && request.resource.data.de == request.auth.uid
                    && request.resource.data.para is string && request.resource.data.para != request.auth.uid
                    && request.resource.data.conv == (request.auth.uid < request.resource.data.para
                         ? request.auth.uid + '_' + request.resource.data.para : request.resource.data.para + '_' + request.auth.uid)
                    // conversa do site ou conversa individual do chat da equipe (equipe_dm)
                    && (exists(/databases/$(database)/documents/conversas/$(request.resource.data.conv))
                        || exists(/databases/$(database)/documents/equipe_dm/$(request.resource.data.conv)))
                    && !bloqueioEntre(request.auth.uid, request.resource.data.para)
                    && request.resource.data.tipo in ['audio', 'video']
                    && request.resource.data.status == 'chamando'
                    && sdpOk(request.resource.data.oferta)
                    && request.resource.data.criadoEm == request.time;
      // Quem recebe atende (com a resposta) enquanto toca.
      allow update: if daChamada() && request.auth.uid == resource.data.para
                    && resource.data.status == 'chamando'
                    && mudou().hasOnly(['status', 'resposta', 'aceitaEm'])
                    && request.resource.data.status == 'aceita'
                    && sdpOk(request.resource.data.resposta)
                    && request.resource.data.aceitaEm == request.time;
      // Qualquer um dos dois termina: recusar, cancelar, perder (não atendeu) ou encerrar.
      allow update: if daChamada()
                    && resource.data.status in ['chamando', 'aceita']
                    && mudou().hasOnly(['status', 'fimEm', 'encerradaPor'])
                    && request.resource.data.status in ['recusada', 'cancelada', 'perdida', 'encerrada']
                    && request.resource.data.encerradaPor == request.auth.uid
                    && request.resource.data.fimEm == request.time;
      allow delete: if false;
      match /candidatos/{c} {
        function chamada() { return get(/databases/$(database)/documents/chamadas/$(id)).data; }
        allow read: if logado() && request.auth.uid in [chamada().de, chamada().para];
        allow create: if logado() && request.auth.uid in [chamada().de, chamada().para]
                      && chamada().status in ['chamando', 'aceita']
                      && request.resource.data.keys().hasOnly(['de', 'c', 'em'])
                      && request.resource.data.de == request.auth.uid
                      && request.resource.data.c is map && request.resource.data.c.keys().hasOnly(['candidate', 'sdpMid', 'sdpMLineIndex'])
                      && request.resource.data.c.candidate is string && request.resource.data.c.candidate.size() <= 1000
                      && request.resource.data.em == request.time;
        allow update, delete: if false;
      }
    }

    match /conversas/{conversaId} {

      // Cada pessoa só mexe na própria marca de leitura / "digitando".
      function soMinhaChave(campo) {
        return !mudou().hasAny([campo])
            || (request.resource.data[campo] is map
                && request.resource.data[campo].diff(resource.data.get(campo, {})).affectedKeys().hasOnly([request.auth.uid]));
      }

      function contextoOk() {
        let c = request.resource.data.get('contexto', null);
        return c == null
            || (c is map && c.keys().hasOnly(['tipo', 'titulo'])
                && c.tipo in ['servicos', 'delivery', 'lojinha', 'imoveis', 'saude', 'comunidade']
                && valorTextoAte(c.get('titulo', ''), 120));
      }

      // Conversa ainda não criada: só quem faz parte dela (o id tem os dois uids) pode consultar.
      allow get: if logado()
                 && ((resource == null && request.auth.uid in conversaId.split('_'))
                     || (resource != null && request.auth.uid in resource.data.participantes));

      allow list: if logado()
                  && request.auth.uid in resource.data.participantes;
      // A moderação vê quem conversa com quem (para a ficha e denúncias), mas não lista mensagens.
      allow read: if podeModerar();

      allow create: if logado()
                    && request.resource.data.participantes is list
                    && request.resource.data.participantes.size() == 2
                    && request.auth.uid in request.resource.data.participantes
                    && request.resource.data.participantes[0] < request.resource.data.participantes[1]
                    && conversaId == request.resource.data.participantes[0]
                                     + '_' + request.resource.data.participantes[1]
                    && !bloqueioEntre(request.resource.data.participantes[0],
                                      request.resource.data.participantes[1])
                    && emailOk()
                    && request.resource.data.keys().hasOnly(['participantes', 'ultimaMensagem', 'atualizadoEm', 'contexto', 'criadoEm'])
                    && contextoOk()
                    && textoAte('ultimaMensagem', 200);

      // Com bloqueio entre os dois, ninguém mexe na conversa (evita "recado"
      // pelo resumo da última mensagem).
      allow update: if logado()
                    && request.auth.uid in resource.data.participantes
                    && request.resource.data.participantes == resource.data.participantes
                    && !bloqueioEntre(resource.data.participantes[0], resource.data.participantes[1])
                    && mudou().hasOnly(['ultimaMensagem', 'ultimaMensagemRemetenteId', 'atualizadoEm', 'lidoEm', 'vistoEm', 'digitando', 'contexto', 'arquivadaPara', 'ocultaPara', 'falaram', 'fixadaPara', 'silenciadaPara', 'mensagemFixada'])
                    // o resumo da última mensagem só pode ser escrito por quem mandou
                    && (!mudou().hasAny(['ultimaMensagem', 'ultimaMensagemRemetenteId'])
                        || request.resource.data.get('ultimaMensagemRemetenteId', '') == request.auth.uid)
                    && soMinhaChave('lidoEm')
                    && soMinhaChave('vistoEm')
                    && soMinhaChave('digitando')
                    // arquivar / apagar "só para mim": cada um mexe só na própria marca
                    && soMinhaChave('arquivadaPara')
                    && soMinhaChave('ocultaPara')
                    // fixar no topo / silenciar avisos: cada um só a própria marca
                    && soMinhaChave('fixadaPara')
                    && soMinhaChave('silenciadaPara')
                    // falaram: cada um marca só que ELE escreveu (usado para liberar avaliação e reclamação)
                    && soMinhaChave('falaram')
                    && (!mudou().hasAny(['falaram']) || request.resource.data.falaram[request.auth.uid] == true)
                    && (!mudou().hasAny(['contexto']) || contextoOk())
                    // mensagem fixada no topo da conversa (qualquer um dos dois fixa ou tira)
                    && (!mudou().hasAny(['mensagemFixada'])
                        || request.resource.data.mensagemFixada == null
                        || (request.resource.data.mensagemFixada is map
                            && request.resource.data.mensagemFixada.keys().hasOnly(['id', 'texto', 'por', 'em'])
                            && request.resource.data.mensagemFixada.id is string
                            && request.resource.data.mensagemFixada.por == request.auth.uid
                            && valorTextoAte(request.resource.data.mensagemFixada.get('texto', ''), 200)))
                    && textoAte('ultimaMensagem', 200);

      allow delete: if logado()
                    && request.auth.uid in resource.data.participantes;


      // ===================================================
      // MENSAGENS DA CONVERSA
      // ===================================================
      match /mensagens/{mensagemId} {

        function participante() {
          return request.auth.uid in
                 get(/databases/$(database)/documents/conversas/$(conversaId)).data.participantes;
        }

        function semBloqueio() {
          let p = get(/databases/$(database)/documents/conversas/$(conversaId)).data.participantes;
          return !bloqueioEntre(p[0], p[1]);
        }

        // A outra pessoa da conversa não pode estar com a conta desativada.
        function outroAtivo() {
          let p = get(/databases/$(database)/documents/conversas/$(conversaId)).data.participantes;
          return !desativada(p[0] == request.auth.uid ? p[1] : p[0]);
        }

        // Resposta (citação): precisa apontar para uma mensagem real desta conversa,
        // com o autor verdadeiro. Impede citar algo que a outra pessoa nunca disse.
        function respostaOk(r) {
          let caminho = /databases/$(database)/documents/conversas/$(conversaId)/mensagens/$(r.id);
          return r is map
              && r.keys().hasOnly(['id', 'texto', 'remetenteId'])
              && r.id is string
              && valorTextoAte(r.get('texto', ''), 200)
              && exists(caminho)
              && get(caminho).data.remetenteId == r.remetenteId;
        }

        function mensagemOk() {
          let d = request.resource.data;
          return d.keys().hasOnly(['remetenteId', 'texto', 'criadoEm', 'resposta', 'tipoArquivo', 'arquivoUrl', 'nomeArquivo', 'cartao', 'avaliacao'])
              && (!('tipoArquivo' in d) || d.tipoArquivo in ['imagem', 'video', 'audio', 'cartao', 'avaliacao'])
              // pedido de avaliação: aponta para um atendimento criado por quem envia
              && (!('avaliacao' in d)
                  || (d.avaliacao is map && d.get('tipoArquivo', '') == 'avaliacao'
                      && d.avaliacao.keys().hasOnly(['pedidoId', 'titulo', 'tipo', 'negocioNome'])
                      && get(/databases/$(database)/documents/pedidos/$(d.avaliacao.pedidoId)).data.vendedorId == request.auth.uid
                      && valorTextoAte(d.avaliacao.get('titulo', ''), 120)
                      && valorTextoAte(d.avaliacao.get('negocioNome', ''), 80)))
              // anexo: só do próprio site (dentro da mensagem ou Firebase Storage)
              && (!('arquivoUrl' in d)
                  || (d.arquivoUrl is string
                      && d.arquivoUrl.matches('^(data:(image|video|audio)/[^,]{1,80},|https://firebasestorage[.]googleapis[.]com/).*')))
              && (!('nomeArquivo' in d) || valorTextoAte(d.nomeArquivo, 120))
              && (!('cartao' in d) || (d.cartao is map && d.get('tipoArquivo', '') == 'cartao'))
              && (!('resposta' in d) || respostaOk(d.resposta));
        }

        // Versão anterior guardada ao editar: o texto e a hora que ele tinha (envio ou última edição).
        function versaoOk(v) {
          return v is map && v.keys().hasOnly(['texto', 'em'])
              && v.texto == resource.data.get('texto', '')
              && v.em == resource.data.get('editadaEm', resource.data.criadoEm);
        }

        allow read: if logado() && participante();
        // A moderação abre só a mensagem denunciada (uma por vez, sem listar a conversa).
        allow get: if podeModerar();

        allow create: if logado()
                      && emailOk()
                      && participante()
                      && semBloqueio()
                      && outroAtivo()
                      && ritmoOk('msg')
                      && request.resource.data.remetenteId == request.auth.uid
                      && request.resource.data.criadoEm == request.time
                      && textoAte('texto', 4000)
                      && mensagemOk();
